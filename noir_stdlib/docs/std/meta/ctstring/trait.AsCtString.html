Trait AsCtString

pub trait AsCtString {
    // Required methods
    pub comptime fn as_ctstring(self) -> CtString;
}

Converts a string into a CtString at compile time. It is implemented for str<N> and fmtstr<N, T>.

Example:

use std::meta::ctstring::AsCtString;

comptime {
    let x = 7_u32;
    assert_eq(f"x = {x}".as_ctstring(), "x = 7".as_ctstring());
}
