63    #[test]
64    fn signature_does_not_verify_when_hashed_msg_exceeds_curve_order() {
65        // All 0xFF bytes is larger than the secp256r1 curve order
66        // (0xFFFFFFFF00000000FFFFFFFFFFFFFFFFBCE6FAADA7179E84F3B9CAC2FC632551).
67        // The function should reduce it modulo the order, not panic.
68        let oversized_hash: [u8; 32] = [0xff; 32];
69
70        // The result will be false (signature doesn't match the reduced hash), but no panic.
71        let result = verify_signature(&oversized_hash, &PUB_KEY_X, &PUB_KEY_Y, &SIGNATURE).unwrap();
72        assert!(!result);
73    }
