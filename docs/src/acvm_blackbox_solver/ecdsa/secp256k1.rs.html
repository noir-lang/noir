70    #[test]
71    fn signature_does_not_verify_when_hashed_msg_exceeds_curve_order() {
72        // All 0xFF bytes is larger than the secp256k1 curve order
73        // (0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141).
74        // The function should reduce it modulo the order, not panic.
75        let oversized_hash: [u8; 32] = [0xff; 32];
76
77        // The result will be false (signature doesn't match the reduced hash), but no panic.
78        let result = verify_signature(&oversized_hash, &PUB_KEY_X, &PUB_KEY_Y, &SIGNATURE).unwrap();
79        assert!(!result);
80    }
