97    #[test]
98    fn verifies_signature_whose_nonce_point_x_coordinate_exceeds_curve_order() {
99        // A signature where `R.x` lies in `[n, p)`: `R.x = n + 2`, so `r = R.x mod n = 2`.
100        // Per SEC 1 section 4.1.4 the verifier compares `R.x mod n` against `r`, so this is valid.
101        // 0xdeadbeef
102        let hashed_msg: [u8; 32] = [
103            0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
104            0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
105            0xde, 0xad, 0xbe, 0xef,
106        ];
107        // 0x4ed0a0862ad5f9f33c99f1a04251bbb777b17671f25c1462d9c38314fc721c11
108        let pub_key_x: [u8; 32] = [
109            0x4e, 0xd0, 0xa0, 0x86, 0x2a, 0xd5, 0xf9, 0xf3, 0x3c, 0x99, 0xf1, 0xa0, 0x42, 0x51,
110            0xbb, 0xb7, 0x77, 0xb1, 0x76, 0x71, 0xf2, 0x5c, 0x14, 0x62, 0xd9, 0xc3, 0x83, 0x14,
111            0xfc, 0x72, 0x1c, 0x11,
112        ];
113        // 0x1690aa2e4797fbbcce6c1e45f2c026a40c073d39634a05b23cc87c8760538676
114        let pub_key_y: [u8; 32] = [
115            0x16, 0x90, 0xaa, 0x2e, 0x47, 0x97, 0xfb, 0xbc, 0xce, 0x6c, 0x1e, 0x45, 0xf2, 0xc0,
116            0x26, 0xa4, 0x0c, 0x07, 0x3d, 0x39, 0x63, 0x4a, 0x05, 0xb2, 0x3c, 0xc8, 0x7c, 0x87,
117            0x60, 0x53, 0x86, 0x76,
118        ];
119        // r = 2, s = 0x3039
120        let mut signature = [0u8; 64];
121        signature[31] = 0x02;
122        signature[62] = 0x30;
123        signature[63] = 0x39;
124
125        let order = k256::Secp256k1::ORDER.get();
126        let nonce_point_x =
127            nonce_point_x::<k256::Secp256k1>(&hashed_msg, &pub_key_x, &pub_key_y, &signature);
128        assert_eq!(nonce_point_x, order.wrapping_add(&U256::from_u8(2)));
129
130        let valid = verify_signature(&hashed_msg, &pub_key_x, &pub_key_y, &signature).unwrap();
131        assert!(valid);
132
133        // Any other `r` must still be rejected.
134        signature[31] = 0x03;
135        let valid = verify_signature(&hashed_msg, &pub_key_x, &pub_key_y, &signature).unwrap();
136        assert!(!valid);
137    }
