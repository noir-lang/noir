1//! Rewrites a chain of `array_set`s that rebuilds an array into a single `make_array`.
2//!
3//! A loop that fills an array (`for i in 0..N { a[i] = f(i) }`) unrolls into `N` sequential
4//! `array_set` instructions, each producing a fresh `N`-element array value:
5//!
6//! ```text
7//! v3 = make_array [Field 0, ...]   : [Field; N]
8//! v5 = array_set v3, index u32 0, value v1
9//! v9 = array_set v5, index u32 1, value v7
10//! ...
11//! ```
12//!
13//! Every later pass that reasons about array contents then walks `N` distinct array values of
14//! `N` elements each, which is quadratic in `N`. Collapsing the chain to one `make_array` of the
15//! final elements keeps the same semantics and leaves a single array value behind.
16//!
17//! A chain is only rewritten when it is safe to evaluate every write unconditionally:
18//!   - it is rooted at a `make_array`, so all initial elements are known;
19//!   - every intermediate array value has exactly one use, the next `array_set` in the chain, so
20//!     no other instruction can observe a partially built array;
21//!   - every index is an in-bounds constant, so no write can trap.
22//!
23//! Element types wider than one value need no special handling: an array of tuples flattens to one
24//! `make_array` element per field, and `array_set` indexes those flat slots directly, so writing
25//! `elements[index]` is right for `[(Field, Field); N]` just as it is for `[Field; N]`.
26//!
27//! # Ordering
28//!
29//! This must run after unrolling, which is what creates the chains, and before flattening, which
30//! is what introduces `enable_side_effects`. A write under a predicate is conditional and cannot
31//! be folded into an unconditional `make_array`, so the pass takes the absence of
32//! `enable_side_effects` as a precondition rather than checking for it. It must also run before
33//! `mutable_array_set_optimization`, whose in-place writes mutate the root array rather than
34//! copying it, and before `brillig_array_get_and_set`, which shifts constant indices past the
35//! in-memory array header so that an index no longer names the slot it writes.
36

51/// Minimum number of element visits a rewrite has to save before it is worth doing.
52///
53/// A chain of `n` writes over a `len`-element array makes later passes walk `n` array values of
54/// `len` elements; the `make_array` it collapses to is one value of `len` elements, so the rewrite
55/// saves roughly `(n - 1) * len` element visits. Gating on that rather than on `n` alone is what
56/// tracks the cost being avoided: 4 writes over a 5-element array is not the shape that hurts,
57/// 4,096 writes over a 4,096-element array is.
58const MIN_COLLAPSE_WORK: usize = 64;
59
60impl Ssa {
61    /// See the [module docs][self] for more information.
62    #[tracing::instrument(level = "trace", skip(self))]
63    pub(crate) fn lower_array_initializations(self) -> Self {
64        self.lower_array_initializations_saving_at_least(MIN_COLLAPSE_WORK)
65    }
66
67    /// [`Ssa::lower_array_initializations`] with an explicit threshold, so tests can exercise the
68    /// rewrite on arrays small enough to read in a snapshot.
69    fn lower_array_initializations_saving_at_least(mut self, min_work: usize) -> Self {
70        for func in self.functions.values_mut() {
71            #[cfg(debug_assertions)]
72            array_initialization_pre_check(func);
73
74            func.lower_array_initializations(min_work);
75
76            #[cfg(debug_assertions)]
77            array_initialization_post_check(func, min_work);
78        }
79        self
80    }
81}

83/// Pre-check condition for [`Function::lower_array_initializations`].
84///
85/// Panics if the function contains an `enable_side_effects`, which would make a write conditional,
86/// an `array_set` already marked `mutable`, which writes through to the root array instead of
87/// copying it, or Brillig array indices that have been offset, which would stop a constant index
88/// naming the slot it writes. See the [module docs][self] on ordering for why none can be present.
89#[cfg(debug_assertions)]
90fn array_initialization_pre_check(func: &Function) {
91    super::checks::assert_no_brillig_array_offsets(func);
92    super::checks::for_each_instruction(func, |instruction, _dfg| {
93        super::checks::assert_not_enable_side_effects(instruction);
94        super::checks::assert_not_mutable_array_set(instruction);
95    });
96}
97
98/// Post-check condition for [`Function::lower_array_initializations`].
99///
100/// Panics if a chain worth collapsing survives, which is what running to a fixed point is for.
101#[cfg(debug_assertions)]
102fn array_initialization_post_check(func: &Function, min_work: usize) {
103    assert!(
104        collapsible_chains(func, min_work).is_empty(),
105        "array initialization lowering left a collapsible chain behind"
106    );
107}

109/// One `array_set` in a chain.
110struct Link {
111    instruction: InstructionId,
112    block: BasicBlockId,
113    array: ValueId,
114    index: ValueId,
115    value: ValueId,
116}
117
118/// A chain that is worth rewriting, and the array it was found to build.
119struct Collapse {
120    /// The chain's last write, reused in place as the `make_array`. Its result already has the
121    /// array type, so every existing use of the finished array stays valid.
122    tail: InstructionId,
123    /// The writes leading up to it, which the rewrite makes dead.
124    superseded: Vec<(BasicBlockId, InstructionId)>,
125    elements: imbl::Vector<ValueId>,
126    typ: Type,
127}
128
129impl Function {
130    fn lower_array_initializations(&mut self, min_work: usize) {
131        // Collapsing a chain can expose another: a chain rooted at an `array_set` is left alone
132        // because the elements it starts from are unknown, and that root may have just become a
133        // `make_array`. Every round removes at least one `array_set`, so this terminates.
134        loop {
135            let collapses = collapsible_chains(self, min_work);
136            if collapses.is_empty() {
137                return;
138            }
139            for collapse in collapses {
140                self.apply_collapse(collapse);
141            }
142        }
143    }
144
145    /// Rewrites the chain's last write into the `make_array` it builds and drops the rest.
146    fn apply_collapse(&mut self, collapse: Collapse) {
147        let Collapse { tail, superseded, elements, typ } = collapse;
148        self.dfg[tail] = Instruction::MakeArray { elements, typ };
149
150        let mut dead: HashMap<BasicBlockId, HashSet<InstructionId>> = HashMap::default();
151        for (block, instruction) in superseded {
152            dead.entry(block).or_default().insert(instruction);
153        }
154        for (block, dead) in dead {
155            self.dfg[block].instructions_mut().retain(|instruction| !dead.contains(instruction));
156        }
157    }
158}

160/// Finds every chain in `func` worth collapsing. Chains are disjoint, because an intermediate
161/// belongs to the one chain that consumes it, so the results can all be applied.
162fn collapsible_chains(func: &Function, min_work: usize) -> Vec<Collapse> {
163    let uses = count_uses(func);
164    let links = collect_links(func);
165
166    // An array value consumed only by the next write is interior to a chain, so any `array_set`
167    // result that is not one of those ends a chain.
168    let interior: HashSet<ValueId> = links
169        .values()
170        .filter(|link| links.contains_key(&link.array) && uses[&link.array] == 1)
171        .map(|link| link.array)
172        .collect();
173
174    let mut tails: Vec<ValueId> =
175        links.keys().copied().filter(|result| !interior.contains(result)).collect();
176    // `links` is a hash map, so sort for a deterministic rewrite order.
177    tails.sort_unstable();
178
179    tails
180        .into_iter()
181        .filter_map(|tail| {
182            let (chain, root) = chain_ending_at(tail, &links, &uses);
183            let (elements, typ) = collapsed_array(func, &chain, root, min_work)?;
184            Some(Collapse {
185                tail: chain[0].instruction,
186                superseded: chain[1..].iter().map(|link| (link.block, link.instruction)).collect(),
187                elements,
188                typ,
189            })
190        })
191        .collect()
192}

194/// Counts how many times each value is used, so a chain can tell an array it alone consumes from
195/// one something else can still read.
196fn count_uses(func: &Function) -> HashMap<ValueId, u32> {
197    let mut uses: HashMap<ValueId, u32> = HashMap::default();
198    for block in func.reachable_blocks() {
199        for instruction in func.dfg[block].instructions() {
200            func.dfg[*instruction].for_each_value(|value| {
201                *uses.entry(value).or_default() += 1;
202            });
203        }
204        if let Some(terminator) = func.dfg[block].terminator() {
205            terminator.for_each_value(|value| {
206                *uses.entry(value).or_default() += 1;
207            });
208        }
209    }
210    uses
211}

213/// Indexes every non-mutable `array_set` in `func` by the array value it produces.
214fn collect_links(func: &Function) -> HashMap<ValueId, Link> {
215    let mut links = HashMap::default();
216    for block in func.reachable_blocks() {
217        for instruction in func.dfg[block].instructions() {
218            if let Instruction::ArraySet { array, index, value, mutable: false } =
219                func.dfg[*instruction]
220            {
221                let result = func.dfg.instruction_results(*instruction)[0];
222                links
223                    .insert(result, Link { instruction: *instruction, block, array, index, value });
224            }
225        }
226    }
227    links
228}

230/// Walks back from `tail` along the `array` operand for as long as each array value is used only
231/// by the next write. Returns the chain, last write first, and the array it starts from.
232fn chain_ending_at<'links>(
233    tail: ValueId,
234    links: &'links HashMap<ValueId, Link>,
235    uses: &HashMap<ValueId, u32>,
236) -> (Vec<&'links Link>, ValueId) {
237    let mut chain = Vec::new();
238    let mut current = tail;
239    loop {
240        let Some(link) = links.get(&current) else { return (chain, current) };
241        chain.push(link);
242        if uses[&link.array] != 1 {
243            return (chain, link.array);
244        }
245        current = link.array;
246    }
247}

249/// Applies `chain` to the elements of `root`, giving the array the chain builds, or `None` if the
250/// chain cannot be collapsed or is not worth collapsing.
251fn collapsed_array(
252    func: &Function,
253    chain: &[&Link],
254    root: ValueId,
255    min_work: usize,
256) -> Option<(imbl::Vector<ValueId>, Type)> {
257    // A single write cannot be shortened, and the saving is only known once the array's length is.
258    if chain.len() < 2 {
259        return None;
260    }
261    let (mut elements, typ) = func.dfg.get_array_constant(root)?;
262    if (chain.len() - 1) * elements.len() < min_work {
263        return None;
264    }
265
266    // `chain` runs last-write-first; the writes take effect starting from the root.
267    for link in chain.iter().rev() {
268        let index = func.dfg.get_numeric_constant(link.index)?.try_to_u32()?;
269        // Out of bounds: the write would trap, which `make_array` would not reproduce.
270        let slot = elements.get_mut(index as usize)?;
271        *slot = link.value;
272    }
273    Some((elements, typ))
274}
