1//! Post-flattening validation pass that ensures predicated values escaping
2//! their enable-side-effect definition are properly predicated.
3//!
4//! In ACIR, the result of a
5//! [`requires_acir_gen_predicate`][crate::ssa::ir::instruction::Instruction::requires_acir_gen_predicate]
6//! instruction depends on the active `enable_side_effects` predicate. For
7//! side-effecting *arithmetic* (the overflow-checked `Binary` operations) the
8//! result is multiplied by the predicate, so it is `0` whenever the predicate is
9//! false. The other predicated instructions consume the predicate differently
10//! (for example `ArrayGet` clamps its index and a `Call` forwards the predicate
11//! to the callee), so their result is not necessarily `0` when the predicate is
12//! false; this pass conservatively tracks all of them and still requires them to
13//! be guarded before they can escape.
14//!
15//! Some optimization passes (such as `checked_to_unchecked`) can remove
16//! *side-effect* checks and rewrite instructions into ones having
17//! `requires_acir_gen_predicate` set to `false`. The rewritten instruction's
18//! result is then no longer `0` when the predicate is false. This stays sound
19//! only because flattening adds a predicate multiplication, so the escaping
20//! value is still `0` when the predicate is false.
21//!
22//! This pass makes that an invariant and validates that no `requires_acir_gen_predicate`
23//! value can escape its enclosing `enable_side_effects` without a predicate
24//! multiplication (or an equivalent `IfElse` merge, which gates each branch by
25//! its own condition).
26//!
27//! As a result, any optimization on `requires_acir_gen_predicate` done after
28//! flattening is ensured to be sound.
29//!
30//! This is a post-flattening check, so each ACIR function is a single block. The
31//! pass asserts this rather than assuming it, which makes the validation a simple
32//! iteration over the instructions:
33//! - mark `requires_acir_gen_predicate` instructions,
34//! - mark also their uses,
35//! - report a violation if used outside `enable_side_effects`,
36//!   unless multiplied with the predicate.
37

54pub(crate) fn verify_side_effect_predicates(ssa: &Ssa) -> RtResult<()> {
55    for function in ssa.functions.values() {
56        verify_function(function)?;
57    }
58    Ok(())
59}

61fn verify_function(function: &Function) -> RtResult<()> {
62    // Brillig functions do not have `enable_side_effects` instructions
63    if function.runtime().is_brillig() {
64        return Ok(());
65    }
66
67    // The single-instruction-iteration below only inspects the entry block, which
68    // is only complete because flattening reduces every ACIR function to one block.
69    // Assert the invariant instead of trusting it: it is a cheap check when it holds
70    // and turns a future regression into a clear failure rather than silently
71    // skipping the instructions of any other block.
72    assert_eq!(
73        function.reachable_blocks().len(),
74        1,
75        "ACIR function {} should be a single block after flattening",
76        function.name()
77    );
78
79    let dfg = &function.dfg;
80    let block = function.entry_block();
81
82    // Values resulting from `requires_acir_gen_predicate` instructions, or
83    // using such predicated values. They are mapped to their predicate.
84    // These are the 'unsafe' values that we want to track for a non-predicated
85    // use outside their enable-side-effect definition.
86    let mut predicated_values: HashMap<ValueId, ValueId> = HashMap::default();
87
88    // Predicates that hold in every satisfying assignment need no guard, see
89    // `is_constrained_to_one`.
90    let constrained_to_one = collect_values_constrained_to_one(dfg, block);
91
92    // The active predicate: `None` if no predicate (or `1`).
93    let mut current: Option<ValueId> = None;
94
95    for instruction_id in dfg[block].instructions() {
96        let instruction = &dfg[*instruction_id];
97
98        if let Instruction::EnableSideEffectsIf { condition } = instruction {
99            current = (!is_one(function, *condition)).then_some(*condition);
100            continue;
101        }
102
103        // Match instructions for
104        // - using predicated operands
105        // - using predicate operands outside enable-side-effect context
106        // - predicate guards: predicate * value
107        let mut guard_a_predicated_value = false;
108        let mut use_a_predicated_value = None;
109        let mut violation = None;
110        instruction.for_each_value(|operand| {
111            // If the instruction uses a `predicated_value`
112            let Some(&p) = predicated_values.get(&operand) else {
113                return;
114            };
115            if is_predicate_guard(dfg, instruction, operand, p) {
116                guard_a_predicated_value = true;
117            } else {
118                // Propagate the predicate to the current instruction
119                use_a_predicated_value.get_or_insert(p);
120                if current.is_none() {
121                    // The `predicated_value` operand is not used under a predicate,
122                    // flag it as an error.
123                    violation.get_or_insert(operand);
124                }
125            }
126        });
127        if let Some(operand) = violation {
128            let call_stack = function.dfg.get_instruction_call_stack(*instruction_id);
129            return Err(escape_error(function, operand, call_stack));
130        }
131
132        // `result_predicate` is the predicate of the current requires_acir_gen_predicate instruction,
133        // or the predicate of the instruction's operand, if any.
134        // When the instruction does not depend on a predicate, it is `None`
135        let result_predicate = if guard_a_predicated_value {
136            // A guard is safe
137            None
138        } else if use_a_predicated_value.is_some() {
139            use_a_predicated_value
140        } else if instruction.requires_acir_gen_predicate(dfg)
141            && !is_div_or_mod_by_nonzero_constant(dfg, instruction)
142        {
143            current
144        } else {
145            None
146        };
147        if let Some(p) = result_predicate {
148            // A predicate that is pinned to `1` is never false, so the results computed
149            // under it are never the disabled-branch value and need no guard to escape.
150            if !is_constrained_to_one(dfg, p, &constrained_to_one) {
151                for result in dfg.instruction_results(*instruction_id) {
152                    predicated_values.insert(*result, p);
153                }
154            }
155        }
156    }
157
158    // Error on returning an unsafe value
159    if let Some(TerminatorInstruction::Return { return_values, call_stack }) =
160        dfg[block].terminator()
161    {
162        for value in return_values {
163            if predicated_values.contains_key(value) {
164                let call_stack = function.dfg.get_call_stack(*call_stack);
165                return Err(escape_error(function, *value, call_stack));
166            }
167        }
168    }
169
170    Ok(())
171}

173/// Collect the values that an unconditional `constrain x == 1` pins to `1`.
174///
175/// A flattened ACIR function is a single block, so every `Constrain` in it is enforced
176/// on every execution. A value constrained to `1` here is therefore `1` in *every*
177/// satisfying assignment, no matter where in the block the constraint sits — which is
178/// why this is gathered up front rather than as the instructions are walked.
179fn collect_values_constrained_to_one(dfg: &DataFlowGraph, block: BasicBlockId) -> HashSet<ValueId> {
180    let mut constrained_to_one = HashSet::default();
181    for instruction_id in dfg[block].instructions() {
182        let Instruction::Constrain(lhs, rhs, _) = &dfg[*instruction_id] else {
183            continue;
184        };
185        for (value, expected) in [(lhs, rhs), (rhs, lhs)] {
186            if dfg.get_numeric_constant(*expected).is_some_and(|c| c.is_one()) {
187                constrained_to_one.insert(*value);
188            }
189        }
190    }
191    constrained_to_one
192}

194/// Whether predicate `p` is `1` in every satisfying assignment, in which case a value
195/// computed under it is never the disabled-branch value and may escape ungated.
196///
197/// This is what keeps the decomposition of a guarded assertion valid. `decompose_constrain`
198/// rewrites a boolean `constrain (p * value) == 1` into `constrain p == 1` and
199/// `constrain value == 1`, dropping the multiplication that guarded `value`. The two forms
200/// admit exactly the same assignments (`p = 1` and `value = 1`), so the bare use of `value`
201/// is not an over-constraint — but it is only faithful *because* `p` is pinned alongside it.
202///
203/// The proof may be spread over the factors rather than the product: flattening gates a
204/// nested branch by the conjunction of its enclosing conditions, and the decomposition
205/// recurses through that product, so `p = p_outer * p_inner` is proven by
206/// `constrain p_outer == 1` plus `constrain p_inner == 1`.
207///
208/// Casts are followed towards the source: casting `1` yields `1` in any numeric type, so a
209/// constrained source proves every cast of it.
210fn is_constrained_to_one(
211    dfg: &DataFlowGraph,
212    value: ValueId,
213    constrained_to_one: &HashSet<ValueId>,
214) -> bool {
215    if constrained_to_one.contains(&value) {
216        return true;
217    }
218    let value = strip_casts(dfg, value);
219    if constrained_to_one.contains(&value) {
220        return true;
221    }
222    if dfg.get_numeric_constant(value).is_some_and(|c| c.is_one()) {
223        return true;
224    }
225    if let Value::Instruction { instruction, .. } = &dfg[value]
226        && let Instruction::Binary(Binary { lhs, rhs, operator: BinaryOp::Mul { .. } }) =
227            &dfg[*instruction]
228    {
229        // A product is `1` exactly when both of its factors are.
230        return is_constrained_to_one(dfg, *lhs, constrained_to_one)
231            && is_constrained_to_one(dfg, *rhs, constrained_to_one);
232    }
233    false
234}

236/// True if `instruction` re-applies predicate `p` to `operand`, zeroing it
237/// whenever `p` is false: `mul p, operand` (either order), or a branch of an
238/// `IfElse` whose condition is `p` and whose value is `operand`. A merge gates
239/// each branch by its own condition (`then_condition * then_value +
240/// else_condition * else_value`), so both branches are checked.
241fn is_predicate_guard(
242    dfg: &DataFlowGraph,
243    instruction: &Instruction,
244    operand: ValueId,
245    p: ValueId,
246) -> bool {
247    match instruction {
248        Instruction::Binary(Binary { lhs, rhs, operator: BinaryOp::Mul { .. } }) => {
249            (*lhs == operand && is_predicate(dfg, *rhs, p))
250                || (*rhs == operand && is_predicate(dfg, *lhs, p))
251        }
252        Instruction::IfElse { then_condition, then_value, else_condition, else_value } => {
253            (*then_value == operand && is_predicate(dfg, *then_condition, p))
254                || (*else_value == operand && is_predicate(dfg, *else_condition, p))
255        }
256        _ => false,
257    }
258}

260/// Whether multiplying by `value` re-applies predicate `p`, i.e. `value` is `0`
261/// whenever `p` is `0`.
262///
263/// Predicates are `u1` conditions; casting one to a wider type preserves its
264/// zero/non-zero-ness, so casts on either side are ignored when comparing. This
265/// holds when `value` is `p` itself, but also when:
266/// - `value` is a product `p * r` (in any nesting or operand order): such a
267///   product is `0` whenever `p` is `0`, so `value * operand` is still `0` on the
268///   disabled branch. Flattening gates a nested branch by the *conjunction* of all
269///   enclosing conditions, so a value tainted by an outer predicate `p` is
270///   re-guarded by an inner predicate of the form `p * inner_condition`.
271/// - `value` and `p` are different casts of the same source: the signed-division
272///   expansion, for example, derives both the branch predicate and the merge
273///   multiplier as separate casts of one sign value.
274/// - `value` is an `IfElse` merge `then_condition * then_value + else_condition *
275///   else_value`: the sum is `0` whenever `p` is `0` exactly when each product term
276///   is, and a product term is `0` whenever either of its factors is.
277fn is_predicate(dfg: &DataFlowGraph, value: ValueId, p: ValueId) -> bool {
278    let value = strip_casts(dfg, value);
279    let p = strip_casts(dfg, p);
280    if value == p {
281        return true;
282    }
283    if let Value::Instruction { instruction, .. } = &dfg[value] {
284        match &dfg[*instruction] {
285            Instruction::Binary(Binary { lhs, rhs, operator: BinaryOp::Mul { .. } }) => {
286                return is_predicate(dfg, *lhs, p) || is_predicate(dfg, *rhs, p);
287            }
288            Instruction::IfElse { then_condition, then_value, else_condition, else_value } => {
289                return (is_predicate(dfg, *then_condition, p)
290                    || is_predicate(dfg, *then_value, p))
291                    && (is_predicate(dfg, *else_condition, p)
292                        || is_predicate(dfg, *else_value, p));
293            }
294            _ => {}
295        }
296    }
297    false
298}

300/// Follow a chain of `cast` instructions to the underlying source value.
301fn strip_casts(dfg: &DataFlowGraph, mut value: ValueId) -> ValueId {
302    while let Value::Instruction { instruction, .. } = &dfg[value] {
303        if let Instruction::Cast(src, _) = &dfg[*instruction] {
304            value = *src;
305        } else {
306            break;
307        }
308    }
309    value
310}

312fn is_one(function: &Function, value: ValueId) -> bool {
313    function.dfg.get_numeric_constant(value).is_some_and(|c| c.is_one())
314}
315
316fn is_div_or_mod_by_nonzero_constant(dfg: &DataFlowGraph, instruction: &Instruction) -> bool {
317    let Instruction::Binary(Binary { rhs, operator: BinaryOp::Div | BinaryOp::Mod, .. }) =
318        instruction
319    else {
320        return false;
321    };
322    dfg.get_numeric_constant(*rhs).is_some_and(|c| !c.is_zero())
323}

325fn escape_error(function: &Function, operand: ValueId, call_stack: CallStack) -> RuntimeError {
326    let message =
327        format!("Value {operand} escapes `enable_side_effects` in function {}", function.name());
328
329    RuntimeError::SsaValidationError { message, call_stack }
330}

338    #[test]
339    fn ifelse_on_the_multiplier_side_is_recognized() {
340        // Here the `IfElse` sits on the *predicate side*: it is the multiplier of a
341        // `mul` guard, not the merge of the tainted value. `is_predicate`'s `IfElse`
342        // arm sees that each branch of `if v1 then v0 else v0` is `0` whenever `v0`
343        // is, so the merge is `0` whenever `v0` is, and the guard is recognized.
344        let src = "
345        acir(inline) fn main f0 {
346          b0(v0: u1, v1: u1, v2: u16):
347            enable_side_effects v0
348            v3 = cast v2 as u32
349            v4 = add v3, u32 1
350            enable_side_effects u1 1
351            v5 = not v1
352            v6 = if v1 then v0 else (if v5) v0
353            v7 = cast v6 as u32
354            v8 = mul v7, v4
355            return v8
356        }
357        ";
358        let ssa = Ssa::from_str_no_validation(src).unwrap();
359        assert!(verify_side_effect_predicates(&ssa).is_ok());
360
361        // `v6 = v0`, so with `v0 = 0` the escaping value really is `0`.
362        let disabled =
363            ssa.interpret(vec![Value::bool(false), Value::bool(true), Value::u16(7)]).unwrap();
364        assert_eq!(disabled, vec![Value::u32(0)]);
365        let enabled =
366            ssa.interpret(vec![Value::bool(true), Value::bool(true), Value::u16(7)]).unwrap();
367        assert_eq!(enabled, vec![Value::u32(8)]);
368    }

589    #[test]
590    fn accepts_guard_by_conjoined_sub_predicate() {
591        // A checked add tainted by `v0` is re-guarded by `v0 * v1`, the
592        // conjunction flattening builds for a nested branch. `v0 * v1` is `0`
593        // whenever `v0` is `0`, so the guard zeroes the value on the disabled
594        // branch and the escape is spurious.
595        let src = "
596        acir(inline) fn main f0 {
597          b0(v0: u1, v1: u1, v2: u16):
598            enable_side_effects v0
599            v3 = cast v2 as u32
600            v4 = add v3, u32 1
601            v5 = unchecked_mul v0, v1
602            v6 = cast v5 as u32
603            v7 = mul v6, v4
604            enable_side_effects u1 1
605            return v7
606        }
607        ";
608        let ssa = Ssa::from_str_no_validation(src).unwrap();
609        assert!(verify_side_effect_predicates(&ssa).is_ok());
610    }

612    #[test]
613    fn accepts_guard_by_other_cast_of_predicate_source() {
614        // The branch predicate (`cast v0 as u1`) and the merge multiplier
615        // (`cast v0 as u32`) are different casts of the same source `v0`, as the
616        // signed-division expansion produces. Casts preserve zero-ness, so the
617        // multiplier still zeroes the tainted value when the predicate is `0`.

623            v3 = cast v1 as u32
624            v4 = add v3, u32 1
625            enable_side_effects u1 1
626            v5 = cast v0 as u32
627            v6 = mul v5, v4
628            return v6
629        }
630        ";
