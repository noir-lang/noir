1//! The call-argument reachable-aliased-use check.
2//!
3//! A callee may mutate an array argument in place at runtime — directly, or by
4//! returning an alias of it that the caller then mutates ([`super::Context`]'s
5//! alias analysis treats `Call` results as fresh, so that escape is invisible
6//! to the `array_set` scan). When the caller reuses such an argument without a
7//! protecting `inc_rc`, the mutation becomes observable through the alias —
8//! exactly the precondition `purity_analysis` relies on being absent.
9//!
10//! This verifier drives the same aliasing/coverage/forward-walk machinery as
11//! [`super::array_set`], seeded from call arguments instead of `array_set`
12//! sources, and gated on whether the callee can modify its arguments (mirroring
13//! `can_modify_args` in `ssa_gen`).
14//!
15//! It additionally checks a relation only visible at the call site: whether two
16//! argument positions of one call denote the same storage with no protecting
17//! `inc_rc` (see [`check_co_aliased_arguments`]).
18

38/// Verify the call-argument aliasing invariant on every Brillig function in
39/// `ssa`. See the module docs for the invariant.
40pub(crate) fn verify(ssa: &Ssa) -> RtResult<()> {
41    let may_mutate = compute_may_mutate_args(ssa);
42    let returns_arg_alias = compute_returns_arg_alias(ssa);
43
44    // A user-function call must have its array arguments checked if the callee
45    // may mutate one in place, *or* may hand back an alias of one that the
46    // caller then mutates (e.g. an identity function — see [#1443]). Either way,
47    // reusing the argument without a protecting `inc_rc` is a hazard. Both
48    // summaries cover every function, so indexing them is safe (and asserts we
49    // computed a value for every callee).
50    //
51    // [#1443]: https://github.com/noir-lang/noir-claude/issues/1443
52    let needs_check = |callee: FunctionId| may_mutate[&callee] || returns_arg_alias[&callee];
53
54    for function in ssa.functions.values() {
55        verify_function(function, &needs_check)?;
56    }
57    Ok(())
58}

60/// Per-function check. Skips ACIR functions (the invariant only applies to
61/// Brillig, where a callee may mutate an argument in place). For every `call`
62/// whose callee may mutate an argument or return an alias of one (`needs_check`),
63/// treats each array-typed argument as an all-index in-place mutation and runs
64/// the shared coverage + forward walk: a forward-reachable aliased read with no
65/// protecting `inc_rc` is a hazard.
66fn verify_function(function: &Function, needs_check: &impl Fn(FunctionId) -> bool) -> RtResult<()> {
67    if !function.runtime().is_brillig() {
68        return Ok(());
69    }
70
71    let ctx = Context::new(function);
72
73    for block_id in function.reachable_blocks() {
74        for (idx, instruction_id) in function.dfg[block_id].instructions().iter().enumerate() {
75            let instruction_id = *instruction_id;
76            let Instruction::Call { func, arguments } = &function.dfg[instruction_id] else {
77                continue;
78            };
79
80            // Mirror `ssa_gen`'s `can_modify_args`: a callee that provably
81            // cannot mutate an argument *or* hand back an alias of one (a
82            // foreign call, a pure builtin, or a function that neither mutates
83            // an argument nor returns an alias of one) has its argument clones
84            // elided by the ownership pass, so a reused argument legitimately
85            // carries no `inc_rc`. Skipping such calls is what keeps this check
86            // from flagging well-formed SSA.
87            if !callee_needs_arg_check(function, *func, needs_check) {
88                continue;
89            }
90
91            // Treat the call as an all-index mutation of each array-*value*
92            // argument: `derived` is empty (the callee has no in-place result
93            // to chain from here) and the write index is unknown (it may touch
94            // any position). `is_array` matches a top-level array/vector value
95            // and so excludes a reference argument (`&mut [T; N]`): mutation
96            // through a reference is the explicit, caller-visible pattern the
97            // frontend passes a `&mut` for, not the value-array copy-on-write
98            // hazard this check is about (and a reference param already makes
99            // the callee impure). A top-level array of references (`[&mut T;
100            // N]`) is still a value and is checked.
101            let array_args: Vec<ValueId> = arguments
102                .iter()
103                .copied()
104                .filter(|&arg| function.dfg.type_of_value(arg).is_array())
105                .collect();
106            for arg in array_args {
107                // See the same check in [`super::array_set`]: a global's storage outlives the
108                // function, so only an `inc_rc` can keep a callee's in-place mutation of it
109                // unobservable.
110                let alias_set = ctx.alias_set_for(arg, block_id, idx);
111                if let Some(global) =
112                    ctx.unprotected_global_for_source(&alias_set, arg, block_id, idx)
113                {
114                    let message = format!(
115                        "call in function {} passes array {arg}, whose storage may be that of \
116                         global {global}, to a callee that may mutate it, with no `inc_rc` \
117                         protecting it on that path; a global is live for the whole program, \
118                         so an in-place mutation would be observable",
119                        function.name(),
120                    );
121                    return Err(RuntimeError::CallArgAliasViolation {
122                        message,
123                        call_stack: function.dfg.get_instruction_call_stack(instruction_id),
124                        aliased_use_call_stack: function
125                            .dfg
126                            .get_instruction_call_stack(instruction_id),
127                    });
128                }
129
130                let Some(hit) = ctx.aliased_use_for_source(
131                    arg,
132                    block_id,
133                    idx,
134                    instruction_id,
135                    None,
136                    imbl::HashSet::new(),
137                ) else {
138                    continue;
139                };
140
141                let message = format!(
142                    "call in function {} passes array {arg} that is read again as {} on a \
143                     forward path with no preceding `inc_rc`; if the callee mutates the argument \
144                     in place, or returns an alias of it that is then mutated, the mutation would \
145                     be observable through that alias",
146                    function.name(),
147                    hit.value,
148                );
149                return Err(RuntimeError::CallArgAliasViolation {
150                    message,
151                    call_stack: function.dfg.get_instruction_call_stack(instruction_id),
152                    aliased_use_call_stack: function
153                        .dfg
154                        .get_instruction_call_stack(hit.instruction),
155                });
156            }
157
158            check_co_aliased_arguments(function, &ctx, arguments, block_id, idx, instruction_id)?;
159        }
160    }
161    Ok(())
162}

164/// Reject a `call` in which two argument positions may denote the same array
165/// storage with no protecting `inc_rc`.
166///
167/// The per-argument forward walk in [`verify_function`] looks for an aliased
168/// read *in the caller, after the call*, so it cannot see the hazard where the
169/// same buffer reaches the callee twice — as the same value in two positions
170/// (`f(v, v)`), or once as an array value and once as the pointee of a
171/// reference (the shape `ssa_gen` emits for `f(&mut x, x)`). The callee can
172/// then mutate one handle in place at reference count 1 and observe (e.g.
173/// return) the pre-mutation contents through the other, entirely inside the
174/// callee — where the two parameters are unrelated values and the shared-buffer
175/// relation is invisible to the intraprocedural alias engine. The relation is
176/// only established here, at the call site, so this is where it must be checked
177/// (noir-lang/noir-claude#1563).
178///
179/// Each argument position is resolved to the storage it denotes at the call:
180/// an array-typed argument to itself, and a reference argument to the value
181/// its in-block reaching `store` wrote, when there is one (after
182/// `mem2reg_brillig` the emitted shape keeps that `store` in the call's own
183/// block; a pointee established elsewhere is not traced — a false negative,
184/// never a false positive). Two *reference* arguments sharing a pointee are
185/// not flagged: a write through a reference is ordinary reference semantics,
186/// visible through every alias of the reference by design, not a copy-on-write
187/// violation — so a pair must include at least one array-value position.
188///
189/// Whether a pair denotes one buffer, and whether that buffer is protected, is
190/// decided per path by [`Context::pair_has_unprotected_shared_storage`]: a
191/// pair is flagged only when some backward path resolves both positions to the
192/// same storage with no `inc_rc` crossed on it. A whole-call relation is not
193/// path-sensitive enough — a branch-local `inc_rc` protecting the only sharing
194/// path, or two branches passing one buffer through *different* positions,
195/// must both be accepted.
196fn check_co_aliased_arguments(
197    function: &Function,
198    ctx: &Context,
199    arguments: &[ValueId],
200    block_id: BasicBlockId,
201    call_idx: usize,
202    call_id: InstructionId,
203) -> RtResult<()> {
204    // For each argument position: `is_value` (an array value, as opposed to a
205    // reference whose pointee was resolved) and the storage the position
206    // denotes. `None` when the position denotes no array storage we can
207    // resolve.
208    let storages: Vec<Option<(bool, ValueId)>> = arguments
209        .iter()
210        .map(|&arg| match function.dfg.type_of_value(arg).as_ref() {
211            typ if typ.is_array() => Some((true, arg)),
212            Type::Reference(element, _) if element.contains_an_array() => {
213                let pointee = in_block_reaching_store(function, block_id, call_idx, arg)?;
214                Some((false, pointee))
215            }
216            _ => None,
217        })
218        .collect();
219
220    for (i, i_storage) in storages.iter().enumerate() {
221        let Some((i_is_value, i_value)) = i_storage else { continue };
222        for (j, j_storage) in storages.iter().enumerate().skip(i + 1) {
223            let Some((j_is_value, j_value)) = j_storage else { continue };
224            if !i_is_value && !j_is_value {
225                continue;
226            }
227            if !ctx.pair_has_unprotected_shared_storage(*i_value, *j_value, block_id, call_idx) {
228                continue;
229            }
230            let message = format!(
231                "call in function {} passes the same array storage through two arguments \
232                 ({} and {}) with no preceding `inc_rc`; if the callee mutates it in place \
233                 through one of them, the mutation would be observable through the other",
234                function.name(),
235                arguments[i],
236                arguments[j],
237            );
238            let call_stack = function.dfg.get_instruction_call_stack(call_id);
239            return Err(RuntimeError::CallArgAliasViolation {
240                message,
241                call_stack: call_stack.clone(),
242                aliased_use_call_stack: call_stack,
243            });
244        }
245    }
246    Ok(())
247}

249/// The value most recently stored through `address` in `block_id` before the
250/// instruction at `call_idx` — the reference's in-block reaching definition —
251/// or `None` when the block contains no such store.
252fn in_block_reaching_store(
253    function: &Function,
254    block_id: BasicBlockId,
255    call_idx: usize,
256    address: ValueId,
257) -> Option<ValueId> {
258    let instructions = function.dfg[block_id].instructions();
259    instructions[..call_idx].iter().rev().find_map(|id| match &function.dfg[*id] {
260        Instruction::Store { address: a, value } if *a == address => Some(*value),
261        _ => None,
262    })
263}

265/// Whether a call to the callee referenced by `func` needs its array arguments
266/// checked — i.e. the callee may mutate an argument in place or return an alias
267/// of one. Mirrors `ssa_gen`'s `can_modify_args`: foreign calls only read their
268/// inputs and return fresh results; pure builtins that are safe for clone
269/// elision do neither; an unresolved/dynamic callee is assumed to need
270/// checking; and a known function is decided by `needs_check`, which combines
271/// the two call-graph summaries (`may_mutate || returns_arg_alias`).
272fn callee_needs_arg_check(
273    function: &Function,
274    func: ValueId,
275    needs_check: &impl Fn(FunctionId) -> bool,
276) -> bool {
277    match &function.dfg[func] {
278        Value::Function(callee) => needs_check(*callee),
279        Value::Intrinsic(intrinsic) => intrinsic_may_mutate_args(*intrinsic),
280        Value::ForeignFunction { .. } => false,
281        _ => true,
282    }
283}

285/// Whether a call to `intrinsic` may mutate an array argument in place,
286/// mirroring `is_pure_builtin_func` in `ssa_gen`: a pure intrinsic that is safe
287/// for clone elision in Brillig cannot, everything else conservatively can.
288fn intrinsic_may_mutate_args(intrinsic: Intrinsic) -> bool {
289    intrinsic.unsafe_for_clone_elision_in_brillig()
290        || !matches!(intrinsic.purity(), Purity::Pure | Purity::PureWithPredicate)
291}

293/// Populate per-function state with `init`, then run `update` over every
294/// function until a full round makes no change, and return the final state.
295///
296/// `init` receives each function and a mutable reference to the
297/// default-initialized state to fill in that function's entry. `update`
298/// likewise receives each function and the state, mutates it in place, and
299/// returns whether it changed anything; returning `false` short-circuits a
300/// function whose contribution is already settled. The caller keeps whichever
301/// part of the final state it needs.
302fn fixpoint<S: Default>(
303    ssa: &Ssa,
304    mut init: impl FnMut(&Function, &mut S),
305    mut update: impl FnMut(&Function, &mut S) -> bool,
306) -> S {
307    let mut state = S::default();
308    for function in ssa.functions.values() {
309        init(function, &mut state);
310    }
311    let mut changed = true;
312    while changed {
313        changed = false;
314        for function in ssa.functions.values() {
315            changed |= update(function, &mut state);
316        }
317    }
318    state
319}

321/// Compute, for every function, whether a call to it may mutate the storage of
322/// one of its array arguments observably to the caller.
323///
324/// A function may-mutate if it contains an in-place mutation (`array_set` or
325/// `store`), calls a may-mutate function, calls a mutating intrinsic, or calls
326/// an unresolved/dynamic target (assume the worst). Foreign calls contribute
327/// nothing — oracles only read their inputs. This is an over-approximation:
328/// the only callees marked *not* may-mutate are exactly those whose argument
329/// clones the ownership pass elides, so a reused argument with no `inc_rc`
330/// never trips the check on well-formed SSA. Propagated to a fixed point over
331/// the call graph.
332fn compute_may_mutate_args(ssa: &Ssa) -> HashMap<FunctionId, bool> {
333    let (may_mutate, _callees) = fixpoint(
334        ssa,
335        // Initialize the state to each function's own (non-propagated) may-mutate flag,
336        // plus its callee list. The callee list, built once by `init`, lets the propagation
337        // rounds avoid re-scanning instruction bodies.
338        |function,
339         (may_mutate, callees): &mut (
340            HashMap<FunctionId, bool>,
341            HashMap<FunctionId, Vec<FunctionId>>,
342        )| {
343            let mut base = false;
344            let mut calls = Vec::new();
345            for block_id in function.reachable_blocks() {
346                for instruction_id in function.dfg[block_id].instructions() {
347                    match &function.dfg[*instruction_id] {
348                        Instruction::ArraySet { .. } => base = true,
349                        // A `store` can only mutate an array argument's storage
350                        // when the stored value is (or contains) an array — e.g.
351                        // writing a new array through a `&mut [T]` parameter. A
352                        // store of a purely numeric value (such as a wrapper
353                        // materializing a scalar `&mut` argument) cannot touch any
354                        // array argument, so counting it would mark input-preserving
355                        // wrappers as may-mutate and reject SSA whose call-argument
356                        // clones the ownership pass legitimately elided.
357                        Instruction::Store { value, .. } => {
358                            base |= function.dfg.type_of_value(*value).contains_an_array();
359                        }
360                        Instruction::Call { func, .. } => match &function.dfg[*func] {
361                            Value::Function(callee) => calls.push(*callee),
362                            Value::Intrinsic(intrinsic) => {
363                                base |= intrinsic_may_mutate_args(*intrinsic);
364                            }
365                            // Foreign calls only read their inputs.
366                            Value::ForeignFunction { .. } => {}
367                            // An unresolved or dynamic callee: assume the worst.
368                            _ => base = true,
369                        },
370                        _ => {}
371                    }
372                }
373            }
374            may_mutate.insert(function.id(), base);
375            callees.insert(function.id(), calls);
376        },
377        // Propagate: a function may-mutate if any callee may-mutate.
378        |function, (may_mutate, callees)| {
379            let id = function.id();
380            if may_mutate[&id] {
381                return false;
382            }
383            let now = callees[&id].iter().any(|c| may_mutate[c]);
384            if now {
385                may_mutate.insert(id, true);
386            }
387            now
388        },
389    );
390    may_mutate
391}

393/// Compute, for every function, whether it may return an array value that
394/// aliases one of its array parameters — e.g. an identity function that returns
395/// its input unchanged. Such a call hands the caller an alias of the argument,
396/// so an in-place mutation of the *result* mutates the *argument's* storage; the
397/// call must be checked even when the callee does not itself mutate.
398///
399/// Distinct from "returns any array": a callee that returns a *fresh* array (a
400/// `make_array`, or a foreign-call result — the shape of an oracle wrapper that
401/// returns an array) is not flagged, so its caller's clone-elided arguments stay
402/// accepted. Propagated to a fixed point over the call graph because the alias
403/// property flows through `Value::Function` call results. (Alias-returning
404/// intrinsics such as the vector mutators are left to `may_mutate` — see
405/// [`function_returns_arg_alias`].)
406fn compute_returns_arg_alias(ssa: &Ssa) -> HashMap<FunctionId, bool> {
407    // Monotonic fixed point: every function starts `false`; one only ever flips
408    // to true, and `function_returns_arg_alias` reads the current map to resolve
409    // callee results.
410    fixpoint(
411        ssa,
412        |function, returns_arg_alias: &mut HashMap<FunctionId, bool>| {
413            returns_arg_alias.insert(function.id(), false);
414        },
415        |function, returns_arg_alias| {
416            let id = function.id();
417            if returns_arg_alias[&id] {
418                return false;
419            }
420            let now = function_returns_arg_alias(function, returns_arg_alias);
421            if now {
422                returns_arg_alias.insert(id, true);
423            }
424            now
425        },
426    )
427}

429/// Whether `function` returns an array value that may alias one of its array
430/// parameters, given the current `returns_arg_alias` summary for resolving
431/// callee results.
432///
433/// Computes the set of *parameter-derived* values to a fixed point: an array
434/// parameter, a block parameter threaded from one, an `array_set` or nested
435/// `array_get` of one, or a `Value::Function` call result whose callee
436/// `returns_arg_alias` and is fed a parameter-derived argument. `make_array` and
437/// foreign-call results stop the trace (genuinely fresh). Intrinsic results also
438/// stop the trace here — the alias-returning ones
439/// (`unsafe_for_clone_elision_in_brillig`) are instead covered by `may_mutate`,
440/// since calling one sets that flag. The function returns an arg alias iff any
441/// returned value is parameter-derived.
442fn function_returns_arg_alias(
443    function: &Function,
444    returns_arg_alias: &HashMap<FunctionId, bool>,
445) -> bool {
446    let dfg = &function.dfg;
447
448    // Incoming block-parameter arguments per destination block, to thread
449    // parameter-derived-ness across edges.
450    let mut incoming: HashMap<BasicBlockId, Vec<Vec<ValueId>>> = HashMap::default();
451    for block_id in function.reachable_blocks() {
452        match dfg[block_id].terminator() {
453            Some(TerminatorInstruction::Jmp { destination, arguments, .. }) => {
454                incoming.entry(*destination).or_default().push(arguments.clone());
455            }
456            Some(TerminatorInstruction::JmpIf {
457                then_destination,
458                then_arguments,
459                else_destination,
460                else_arguments,
461                ..
462            }) => {
463                incoming.entry(*then_destination).or_default().push(then_arguments.clone());
464                incoming.entry(*else_destination).or_default().push(else_arguments.clone());
465            }
466            _ => {}
467        }
468    }
469
470    // Seed with the function's array-value parameters (entry block parameters).
471    let entry = function.entry_block();
472    let mut param_derived: HashSet<ValueId> = dfg
473        .block_parameters(entry)
474        .iter()
475        .copied()
476        .filter(|&p| dfg.type_of_value(p).is_array())
477        .collect();
478
479    let mut changed = true;
480    while changed {
481        changed = false;
482
483        for block_id in function.reachable_blocks() {
484            // Block parameters fed a parameter-derived argument on some edge.
485            if let Some(edges) = incoming.get(&block_id) {
486                let params = dfg.block_parameters(block_id);
487                for (i, &param) in params.iter().enumerate() {
488                    if param_derived.contains(&param) {
489                        continue;
490                    }
491                    // Check any of the incoming edges for arguments which are derived from function inputs.
492                    let fed = edges
493                        .iter()
494                        .any(|args| args.get(i).is_some_and(|a| param_derived.contains(a)));
495                    if fed {
496                        param_derived.insert(param);
497                        changed = true;
498                    }
499                }
500            }
501
502            for instruction_id in dfg[block_id].instructions() {
503                let propagate = match &dfg[*instruction_id] {
504                    // An array_set result shares the operand's storage.
505                    Instruction::ArraySet { array, .. } => param_derived.contains(array),
506                    // A *nested* array_get returns a sub-array that shares the
507                    // source's storage (a brillig array_get on a nested array
508                    // aliases rather than copies). The `is_array` gate on the
509                    // result below restricts this to the nested case — a
510                    // non-nested get yields a scalar and propagates nothing.
511                    Instruction::ArrayGet { array, .. } => param_derived.contains(array),
512                    // A user-function call result aliases an argument only if
513                    // the callee returns an arg alias and is fed a
514                    // parameter-derived argument.
515                    //
516                    // Foreign-call results are genuinely fresh (oracles copy
517                    // across the boundary). Intrinsic results are *not* always
518                    // fresh — the `unsafe_for_clone_elision_in_brillig`
519                    // intrinsics (vector mutators, `str_as_bytes`,
520                    // `array_as_str_unchecked`) return an alias of their input —
521                    // but we deliberately don't trace through them here: calling
522                    // one already makes the function `may_mutate`
523                    // (`intrinsic_may_mutate_args`), so `needs_check` flags it
524                    // via that summary. This pass only has to cover the gap
525                    // `may_mutate` misses: a non-mutating function that passes an
526                    // input straight back.
527                    Instruction::Call { func, arguments } => match &dfg[*func] {
528                        Value::Function(callee) => {
529                            returns_arg_alias[callee]
530                                && arguments.iter().any(|a| param_derived.contains(a))
531                        }
532                        _ => false,
533                    },
534                    _ => false,
535                };
536                if propagate {
537                    for &result in dfg.instruction_results(*instruction_id) {
538                        if dfg.type_of_value(result).is_array() && param_derived.insert(result) {
539                            changed = true;
540                        }
541                    }
542                }
543            }
544        }
545    }
546
547    // Check if the return block contains a value that was derived from the inputs.
548    function.reachable_blocks().iter().any(|&block_id| {
549        matches!(
550            dfg[block_id].terminator(),
551            Some(TerminatorInstruction::Return { return_values, .. })
552                if return_values.iter().any(|v| param_derived.contains(v))
553        )
554    })
555}

562    /// Parse `src`, run the `call` verifier, and require it to reject the SSA
563    /// with a [`crate::errors::RuntimeError::CallArgAliasViolation`]. Panics on
564    /// any other outcome. Runs `call::verify` directly (not the combined check)
565    /// so the assertion proves the *call* verifier is the one that caught the
566    /// hazard.
567    fn assert_verifier_rejects(src: &str) {
568        let ssa = Ssa::from_str(src).expect("SSA parses");
569        let err = super::verify(&ssa).expect_err("expected the verifier to reject");
