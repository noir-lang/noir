40pub(crate) struct Interpreter<'ssa, W> {
41    /// The program's globals, established once by [`Interpreter::interpret_globals`].
42    ///
43    /// Deliberately *not* part of [`Interpreter::evaluation`]: the globals are
44    /// interpreted once and shared by every evaluation the interpreter performs,
45    /// which is what lets one interpreter be reused for many calls.
46    globals: GlobalScope,
47
48    /// The state of the evaluation currently in progress.
49    ///
50    /// [`Interpreter::interpret_function`] replaces this wholesale, so everything
51    /// scoped to one top-level evaluation is reset by construction rather than by
52    /// a list of fields to remember.
53    evaluation: Evaluation,
54
55    functions: &'ssa BTreeMap<FunctionId, Function>,
56
57    /// The options the interpreter was created with.
58    options: InterpreterOptions,
59
60    /// Print output.
61    output: W,
62}
63
64/// The program's globals: the value of each one, and the storage they own.
65///
66/// Held separately from the call stack because globals are not part of any one
67/// call's state — they are visible to every function, for the whole program.
68#[derive(Default)]
69struct GlobalScope {
70    /// The value of each global, by the [`ValueId`] that names it.
71    values: HashMap<ValueId, Value>,
72
73    /// Storage identities of every array or reference defined by global instructions.
74    /// Globals are visible everywhere, so mutating their storage inside a function
75    /// whose purity forbids caller-visible side effects is always a violation.
76    storage: HashSet<StorageIdentity>,
77}
78
79/// The state of one top-level evaluation ([`Interpreter::interpret_function`]).
80///
81/// Every field here is scoped to a single evaluation, and the whole struct is
82/// replaced at the start of the next one. Grouping them means a field added later
83/// is reset along with the rest instead of silently outliving the evaluation that
84/// produced it.
85#[derive(Default)]
86struct Evaluation {
87    /// One entry per call currently in progress, innermost last. Empty between
88    /// evaluations, and while global instructions are being interpreted it holds
89    /// the single frame those instructions are evaluated into.
90    call_stack: Vec<CallContext>,
91
92    /// Number of instructions and terminators executed.
93    step_counter: usize,
94}

96#[derive(Debug, Clone, Copy, Default)]
97pub struct InterpreterOptions {
98    /// If true, the interpreter will trace its execution.
99    pub trace: bool,
100    /// If true, the interpreter treats all foreign function calls (e.g., `print`) as unknown
101    pub no_foreign_calls: bool,
102    /// Optional limit on the number of executed instructions and terminators, to avoid infinite loops.
103    pub step_limit: Option<usize>,
104}

106struct CallContext {
107    /// The function that was called. This is `None` only for the top-level global
108    /// scope where global instructions are evaluated.
109    called_function: Option<FunctionId>,
110
111    /// Contains each value currently defined and visible to the current function.
112    scope: HashMap<ValueId, Value>,
113
114    /// This variable can be modified by `enable_side_effects_if` instructions and is
115    /// expected to have no effect if there are no such instructions or if the code
116    /// being executed is an unconstrained function.
117    side_effects_enabled: bool,
118
119    /// Present when the called function's recorded purity is [Purity::Pure] or
120    /// [Purity::PureWithPredicate], both of which forbid mutating caller-visible
121    /// memory. Absent for `Impure` functions and when no purity analysis has run.
122    pure_scope: Option<PureScope>,
123}
124
125/// The purity contract the interpreter enforces for the duration of one call.
126///
127/// A violation can only stem from a bug in purity analysis or in an SSA pass that
128/// invalidated its results, so the interpreter reports it as an
129/// [InterpreterError::PurityViolation] rather than treating it as program behavior.
130struct PureScope {
131    /// The purity recorded for the called function, as observed by its caller
132    /// (see [DataFlowGraph::purity_of]).
133    purity: Purity,
134
135    /// Storage identities of everything reachable from the call's arguments at
136    /// entry. An in-place write to any of these during the call is observable by
137    /// the caller, which both `Pure` and `PureWithPredicate` forbid.
138    argument_storage: HashSet<StorageIdentity>,
139
140    /// Clones of the arguments, held only to keep the collected storage alive for
141    /// the duration of the call so that a freed allocation's address cannot be
142    /// reused by unrelated storage and trigger a false violation.
143    _argument_keepalive: Vec<Value>,
144}

160            called_function: None,
161            scope: Default::default(),
162            side_effects_enabled: true,
163            pure_scope: None,
164        }
165    }

231    /// Increment the step counter, or return [`InterpreterError::OutOfBudget`].
232    ///
233    /// If there is no step limit, then it doesn't increment the counter.
234    fn inc_step_counter(&mut self) -> IResult<()> {
235        if let Some(limit) = self.options.step_limit {
236            if self.evaluation.step_counter >= limit {
237                return Err(InterpreterError::OutOfBudget { steps: self.evaluation.step_counter });
238            }
239            // With a limit we shouldn't wrap around, but just in case we wanted move this outside,
240            // use a safe wrap-around increment.
241            self.evaluation.step_counter = self.evaluation.step_counter.wrapping_add(1);
242        }
243        Ok(())
244    }

281    /// Define or redefine a value.
282    /// Redefinitions are expected in the case of loops.
283    fn define(&mut self, id: ValueId, value: Value) -> IResult<()> {
284        if self.options.trace {
285            println!("{id} = {value}");
286        }
287
288        if let Some(func) = self.try_current_function() {
289            let expected_type = func.dfg.type_of_value(id);
290            let actual_type = value.get_type();
291
292            if *expected_type != actual_type {
293                // Special case for ZST (Zero-Sized Type) arrays: Allow length mismatches.
294                // In early SSA passes, ZST arrays like [(); 3] are represented with empty element lists.
295                // Later optimization passes will fix the representation.
296                let types_compatible = match (&*expected_type, &actual_type) {
297                    (Type::Array(expected_elem, _), Type::Array(actual_elem, actual_len)) => {
298                        expected_elem == actual_elem
299                            && expected_elem.is_empty()
300                            && actual_len.to_usize() == 0
301                    }
302                    // Reference mutability doesn't affect runtime behavior —
303                    // the interpreter treats &T and &mut T identically.
304                    (Type::Reference(expected_elem, _), Type::Reference(actual_elem, _)) => {
305                        expected_elem == actual_elem
306                    }
307                    _ => false,
308                };
309
310                if !types_compatible {
311                    return Err(internal(InternalError::ValueTypeDoesNotMatchReturnType {
312                        value_id: id,
313                        expected_type: expected_type.to_string(),
314                        actual_type: actual_type.to_string(),
315                    }));
316                }
317            }
318        }
319
320        self.call_context_mut().scope.insert(id, value);
321
322        Ok(())
323    }

325    /// Interpret the global instructions.
326    ///
327    /// Once this is complete, the interpreter can be reused for multiple
328    /// function calls within the same SSA.
329    pub(crate) fn interpret_globals(&mut self) -> IResult<()> {
330        assert!(
331            self.evaluation.call_stack.is_empty(),
332            "globals should be interpreted before any call"
333        );
334        let functions: &'ssa BTreeMap<FunctionId, Function> = self.functions;
335        let Some((_, function)) = functions.first_key_value() else {
336            return Ok(());
337        };
338
339        // Global instructions run through the ordinary instruction path, which defines
340        // its results into the innermost frame, so give them a frame to land in and move
341        // its scope into `self.globals` once they have all been evaluated. Afterwards no
342        // frame holds the globals, so a call cannot reach them as if they were locals.
343        self.evaluation.call_stack.push(CallContext::global_context());
344        let result = self.interpret_global_instructions(function);
345        let context = self.evaluation.call_stack.pop().expect("the global context was just pushed");
346        self.globals.values = context.scope;
347
348        let mut storage = HashSet::default();
349        for value in self.globals.values.values() {
350            value.collect_storage_identities(&mut storage);
351        }
352        self.globals.storage = storage;
353
354        result
355    }

357    /// Evaluate every global instruction into the current frame.
358    fn interpret_global_instructions(&mut self, function: &'ssa Function) -> IResult<()> {
359        let globals = &function.dfg.globals;
360        for (global_id, global) in globals.values_iter() {
361            let value = match global {
362                super::ir::value::Value::Instruction { instruction, .. } => {
363                    let instruction = &globals[*instruction];
364                    self.interpret_instruction(instruction, &[global_id])?;
365                    continue;
366                }
367                super::ir::value::Value::NumericConstant { constant, typ } => {
368                    Value::from_constant(*constant, *typ)?
369                }
370                super::ir::value::Value::Function(id) => Value::Function(*id),
371                super::ir::value::Value::Intrinsic(intrinsic) => Value::Intrinsic(*intrinsic),
372                super::ir::value::Value::ForeignFunction { name, .. } => {
373                    Value::ForeignFunction(name.clone())
374                }
375                super::ir::value::Value::Global(_) | super::ir::value::Value::Param { .. } => {
376                    unreachable!()
377                }
378            };
379            self.define(global_id, value)?;
380        }
381
382        Ok(())
383    }

408    ///

418        arguments: Vec<Value>,

420        self.evaluation = Evaluation::default();
421        self.call_function(function_id, arguments)

424    /// Interpret a function call.
425    ///
426    /// Unlike `interpret_function` this does not reset the state;
427    /// it is meant to be used for internal calls.
428    ///
429    /// If the called function has a recorded purity, its behavior is checked against
430    /// the contract that purity promises: `Pure` and `PureWithPredicate` functions
431    /// must not mutate caller-visible memory (enforced at each in-place write via
432    /// [Self::check_purity_on_mutation]), and `Pure` functions additionally must not
433    /// fail, since passes are free to deduplicate or remove calls to them.
434    fn call_function(&mut self, function_id: FunctionId, arguments: Vec<Value>) -> IResults {
435        // The function's purity as its caller observes it; the innermost frame is
436        // still the caller here. For a top-level call there is no caller, so the
437        // callee's own runtime provides the (identity) projection.
438        let caller_id =
439            self.current_frame().and_then(|frame| frame.called_function).unwrap_or(function_id);
440        let purity = self.functions[&caller_id].dfg.purity_of(function_id);
441
442        let pure_scope = purity.filter(|purity| *purity != Purity::Impure).map(|purity| {
443            let mut argument_storage = HashSet::default();
444            for argument in &arguments {
445                argument.collect_storage_identities(&mut argument_storage);
446            }
447            PureScope { purity, argument_storage, _argument_keepalive: arguments.clone() }
448        });
449
450        let result = self.call_function_inner(function_id, arguments, pure_scope);
451
452        if purity == Some(Purity::Pure)
453            && let Err(error) = &result
454            && error.is_execution_failure()
455        {
456            return Err(self.purity_violation(
457                function_id,
458                Purity::Pure,
459                format!("it failed with: {error}"),
460            ));
461        }
462        result
463    }

465    fn call_function_inner(
466        &mut self,
467        function_id: FunctionId,
468        mut arguments: Vec<Value>,
469        pure_scope: Option<PureScope>,
470    ) -> IResults {
471        self.evaluation.call_stack.push(CallContext::new(function_id, pure_scope));
472
473        if self.evaluation.call_stack.len() >= MAX_INTERPRETER_CALL_STACK_SIZE {
474            let call_stack = self
475                .evaluation
476                .call_stack
477                .iter()
478                .map(|ctx| {
479                    let id =
480                        ctx.called_function.expect("a call frame always has a called function");
481                    let name = self.functions[&id].name().to_string();
482                    (id, name)
483                })
484                .collect();
485            return Err(InterpreterError::StackOverflow { call_stack });
486        }
487
488        let function = &self.functions[&function_id];
489        if self.options.trace {
490            println!();
491            println!("enter function {} ({})", function_id, function.name());
492        }
493
494        let mut block_id = function.entry_block();
495        let dfg = self.dfg();
496
497        // Loop over blocks & instructions inline here to avoid pushing more
498        // call frames (in rust). We only push call frames for function calls which
499        // should prevent stack overflows for all but excessively large call stacks
500        // that may overflow in the brillig vm as well.
501        let return_values = loop {
502            let block = &dfg[block_id];
503
504            if arguments.len() != block.parameters().len() {
505                return Err(internal(InternalError::BlockArgumentCountMismatch {
506                    block: block_id,
507                    arguments: arguments.len(),
508                    parameters: block.parameters().len(),
509                }));
510            }
511
512            for (parameter, argument) in block.parameters().iter().zip_eq(arguments) {
513                self.define(*parameter, argument)?;
514            }
515
516            for instruction_id in block.instructions() {
517                let results = dfg.instruction_results(*instruction_id);
518                self.interpret_instruction(&dfg[*instruction_id], results)?;
519            }
520
521            // Account for the terminator; a function might not have actual instructions, other than jumping around.
522            self.inc_step_counter()?;
523
524            match block.terminator() {
525                None => {
526                    return Err(internal(InternalError::BlockMissingTerminator {
527                        block: block_id,
528                    }));
529                }
530                Some(TerminatorInstruction::Jmp { destination, arguments: jump_args, .. }) => {
531                    block_id = *destination;
532                    if self.options.trace {
533                        println!("jump to {block_id}");
534                    }
535                    arguments = self.lookup_all(jump_args)?;
536                }
537                Some(TerminatorInstruction::JmpIf {
538                    condition,
539                    then_destination,
540                    then_arguments,
541                    else_destination,
542                    else_arguments,
543                    call_stack: _,
544                }) => {
545                    (block_id, arguments) = if self.lookup_bool(*condition, "jmpif condition")? {
546                        (*then_destination, self.lookup_all(then_arguments)?)
547                    } else {
548                        (*else_destination, self.lookup_all(else_arguments)?)
549                    };
550                    if self.options.trace {
551                        println!("jump to {block_id}");
552                    }
553                }
554                Some(TerminatorInstruction::Return { return_values, call_stack: _ }) => {
555                    let return_values = self.lookup_all(return_values)?;
556                    if self.options.trace && !return_values.is_empty() {
557                        let return_values =
558                            return_values.iter().map(ToString::to_string).collect::<Vec<_>>();
559                        println!("return {}", return_values.join(", "));
560                    }
561
562                    break return_values;
563                }
564                Some(TerminatorInstruction::Unreachable { .. }) => {
565                    return Err(InterpreterError::ReachedTheUnreachable);
566                }
567            }
568        };
569
570        if self.options.trace {
571            println!("exit function {} ({})", function_id, function.name());
572            println!();
573        }
574
575        self.evaluation.call_stack.pop();
576
577        if self.options.trace
578            && let Some(context) = self.current_frame()
579            && let Some(function_id) = context.called_function
580        {
581            let function = &self.functions[&function_id];
582            println!("back in function {} ({})", function_id, function.name());
583        }
584
585        Ok(return_values)
586    }

602    /// Whether `storage` is owned by one of the program's globals.
603    ///
604    /// An in-place write into a global is never a write the program itself can make. The
605    /// globals are established once, before any evaluation, and every evaluation shares
606    /// them; at run time each Brillig entry point re-initialises the globals region
607    /// (`brillig_ir/entry_point.rs`) and ACVM builds a fresh VM per `BrilligCall`, so no
608    /// invocation can observe another's write. The interpreter therefore treats global
609    /// storage the way it treats storage with a reference count above 1: it copies.
610    fn is_global_storage(&self, storage: StorageIdentity) -> bool {
611        self.globals.storage.contains(&storage)
612    }

614    /// Check an in-place write to `storage` against every active purity scope.
615    ///
616    /// Writing to storage reachable from a call's arguments (or from a global) is
617    /// observable by that call's caller, which both [Purity::Pure] and
618    /// [Purity::PureWithPredicate] forbid. Storage allocated during the call is not
619    /// in any scope's set, so local mutations pass freely.
620    ///
621    /// Callers only invoke this for writes that are genuinely observable under the
622    /// current runtime's semantics: constrained-context `array_set`s marked mutable
623    /// are exempt, since they merely reuse the backing store of an array value
624    /// proven dead, under ACIR's value semantics.
625    fn check_purity_on_mutation(&self, storage: StorageIdentity, what: &str) -> IResult<()> {
626        for context in self.evaluation.call_stack.iter().rev() {
627            let Some(scope) = &context.pure_scope else { continue };
628            if scope.argument_storage.contains(&storage) || self.is_global_storage(storage) {
629                let function = context
630                    .called_function
631                    .expect("purity scopes only exist on function call frames");
632                let reason = format!("it mutated caller-visible memory ({what})");
633                return Err(self.purity_violation(function, scope.purity, reason));
634            }
635        }
636        Ok(())
637    }

639    /// Check an executed foreign function call against every active purity scope.
640    ///
641    /// A foreign call is an observable side effect, so it is forbidden inside any
642    /// function whose recorded purity is not `Impure` (even a `#[pure]` oracle makes
643    /// its caller at most `PureWithPredicate`, and the only foreign function the
644    /// interpreter executes, `print`, is not a pure oracle).
645    fn check_purity_on_foreign_call(&self, name: &str) -> IResult<()> {
646        for context in self.evaluation.call_stack.iter().rev() {
647            let Some(scope) = &context.pure_scope else { continue };
648            let function =
649                context.called_function.expect("purity scopes only exist on function call frames");
650            let reason = format!("it called foreign function `{name}`");
651            return Err(self.purity_violation(function, scope.purity, reason));
652        }
653        Ok(())
654    }

656    fn lookup(&self, id: ValueId) -> IResult<Value> {
657        if let Some(frame) = self.current_frame()
658            && let Some(value) = frame.scope.get(&id)
659        {
660            return Ok(value.clone());
661        }
662
663        if let Some(value) = self.globals.values.get(&id) {
664            return Ok(value.clone());
665        }
666
667        Ok(match &self.dfg()[id] {
668            super::ir::value::Value::NumericConstant { constant, typ } => {
669                Value::from_constant(*constant, *typ)?
670            }
671            super::ir::value::Value::Function(id) => Value::Function(*id),
672            super::ir::value::Value::Intrinsic(intrinsic) => Value::Intrinsic(*intrinsic),
673            super::ir::value::Value::ForeignFunction { name, .. } => {
674                Value::ForeignFunction(name.clone())
675            }
676            super::ir::value::Value::Instruction { .. }
677            | super::ir::value::Value::Param { .. }
678            | super::ir::value::Value::Global(_) => {
679                unreachable!("`{id}` should already be in scope")
680            }
681        })
682    }

847    fn side_effects_enabled(&self, instruction: &Instruction) -> bool {
848        let Some(current_function) = self.try_current_function() else {

851        };

859        }

862    #[allow(unused)]
863    fn interpret_instruction(
864        &mut self,
865        instruction: &Instruction,
866        results: &[ValueId],
867    ) -> IResult<()> {
868        self.inc_step_counter()?;
869
870        let side_effects_enabled = self.side_effects_enabled(instruction);
871
872        match instruction {
873            Instruction::Binary(binary) => {
874                let result = self.interpret_binary(binary, side_effects_enabled)?;
875                self.define(results[0], result)?;
876                Ok(())
877            }
878            // Cast in SSA relabels the value's type, keeping its bit pattern. Like ACIR (where a
879            // cast is a no-op on the underlying field) it does not range-check: a value that
880            // overflowed its source type via unchecked field arithmetic keeps its extended bits.
881            // Narrowing casts are preceded by an explicit `truncate`, so no truncation is needed here.
882            Instruction::Cast(value, numeric_type) => {
883                let value = self.lookup_numeric(*value, "cast")?;
884                let field = value.to_field();
885                let result = Value::int_from_field(field, *numeric_type)?;
886                self.define(results[0], result)?;
887                Ok(())
888            }
889            Instruction::Not(id) => self.interpret_not(*id, results[0]),
890            Instruction::Truncate { value, bit_size, max_bit_size } => {
891                self.interpret_truncate(*value, *bit_size, *max_bit_size, results[0])
892            }
893            Instruction::Constrain(lhs_id, rhs_id, constrain_error) => {
894                let lhs = self.lookup(*lhs_id)?;
895                let rhs = self.lookup(*rhs_id)?;
896                if side_effects_enabled && lhs != rhs {
897                    let lhs = lhs.to_string();
898                    let rhs = rhs.to_string();
899                    let lhs_id = *lhs_id;
900                    let rhs_id = *rhs_id;
901                    let msg = if let Some(ConstrainError::StaticString(msg)) = constrain_error {
902                        Some(msg.clone())
903                    } else {
904                        None
905                    };
906                    return Err(InterpreterError::ConstrainEqFailed {
907                        lhs,
908                        lhs_id,
909                        rhs,
910                        rhs_id,
911                        msg,
912                    });
913                }
914                Ok(())
915            }
916            Instruction::ConstrainNotEqual(lhs_id, rhs_id, constrain_error) => {
917                let lhs = self.lookup(*lhs_id)?;
918                let rhs = self.lookup(*rhs_id)?;
919                if side_effects_enabled && lhs == rhs {
920                    let lhs = lhs.to_string();
921                    let rhs = rhs.to_string();
922                    let lhs_id = *lhs_id;
923                    let rhs_id = *rhs_id;
924                    let msg = if let Some(ConstrainError::StaticString(msg)) = constrain_error {
925                        Some(msg.clone())
926                    } else {
927                        None
928                    };
929                    return Err(InterpreterError::ConstrainNeFailed {
930                        lhs,
931                        lhs_id,
932                        rhs,
933                        rhs_id,
934                        msg,
935                    });
936                }
937                Ok(())
938            }
939            Instruction::RangeCheck { value, max_bit_size, assert_message } => self
940                .interpret_range_check(
941                    *value,
942                    *max_bit_size,
943                    assert_message.as_ref(),
944                    side_effects_enabled,
945                ),
946            Instruction::Call { func, arguments } => {
947                self.interpret_call(*func, arguments, results, side_effects_enabled)
948            }
949            Instruction::Allocate => {
950                self.interpret_allocate(results[0]);
951                Ok(())
952            }
953            Instruction::Load { address } => self.interpret_load(*address, results[0]),
954            Instruction::Store { address, value } => self.interpret_store(*address, *value),
955            Instruction::EnableSideEffectsIf { condition } => {
956                self.call_context_mut().side_effects_enabled =
957                    self.lookup_bool(*condition, "enable_side_effects")?;
958                Ok(())
959            }
960            Instruction::ArrayGet { array, index } => {
961                self.interpret_array_get(*array, *index, results[0], side_effects_enabled)
962            }
963            Instruction::ArraySet { array, index, value, mutable } => self.interpret_array_set(
964                *array,
965                *index,
966                *value,
967                *mutable,
968                results[0],
969                side_effects_enabled,
970            ),
971            Instruction::IncrementRc { value } => self.interpret_inc_rc(*value),
972            Instruction::DecrementRc { value } => self.interpret_dec_rc(*value),
973            Instruction::IfElse { then_condition, then_value, else_condition, else_value } => self
974                .interpret_if_else(
975                    *then_condition,
976                    *then_value,
977                    *else_condition,
978                    *else_value,
979                    results[0],
980                ),
981            Instruction::MakeArray { elements, typ } => {
982                self.interpret_make_array(elements, results[0], typ)
983            }
984            Instruction::Noop => Ok(()),
985        }
986    }

988    fn interpret_not(&mut self, id: ValueId, result: ValueId) -> IResult<()> {
989        let num_value = self.lookup_numeric(id, "not instruction")?;
990
991        if num_value.as_field().is_some() {
992            return Err(internal(InternalError::UnsupportedOperatorForType {
993                operator: "!",
994                typ: "Field",
995            }));
996        }
997        if let Some(value) = num_value.as_bool() {
998            return self.define(result, Value::Numeric(NumericValue::bool(!value)));
999        }
1000
1001        // Based on AcirContext::not_var: `!x == max - x` where `max == 2^bit_size - 1`, computed on
1002        // the bit pattern reduced into range. The result is always in range.
1003        let typ = num_value.get_type();
1004        let bit_size = num_value.bit_size();
1005        let reduced = truncate_field(num_value.to_field(), bit_size);
1006        let max =
1007            FieldElement::from(2u128).pow(&FieldElement::from(bit_size)) - FieldElement::one();
1008        let new_result = NumericValue::int_from_field(max - reduced, typ)?;
1009        self.define(result, Value::Numeric(new_result))
1010    }

1012    fn interpret_truncate(
1013        &mut self,
1014        value_id: ValueId,
1015        bit_size: u32,
1016        max_bit_size: u32,
1017        result: ValueId,
1018    ) -> IResult<()> {
1019        let value = self.lookup_numeric(value_id, "truncate")?;
1020        let typ = value.get_type();
1021        if bit_size == 0 {
1022            return Err(internal(InternalError::TruncateToZeroBits { value_id, max_bit_size }));
1023        }
1024
1025        if value.as_bool().is_some() {
1026            return self.define(result, Value::Numeric(value));
1027        }
1028
1029        let field = value.to_field();
1030
1031        let truncated = NumericValue::int_from_field(truncate_field(field, bit_size), typ)?;
1032        self.define(result, Value::Numeric(truncated))
1033    }

1035    fn interpret_range_check(
1036        &self,
1037        value_id: ValueId,
1038        max_bit_size: u32,
1039        error_message: Option<&String>,
1040        side_effects_enabled: bool,
1041    ) -> IResult<()> {
1042        if !side_effects_enabled {
1043            return Ok(());
1044        }
1045
1046        if max_bit_size == 0 {
1047            return Err(internal(InternalError::RangeCheckToZeroBits { value_id }));
1048        }
1049
1050        let value = self.lookup_numeric(value_id, "range check")?;
1051
1052        // max_bit_size > 0 so u1 always passes; every other variant stores its bit pattern as a
1053        // field, so the number of bits is read directly off that field (matching ACIR, which
1054        // range-checks the field representation).
1055        if value.as_bool().is_some() {
1056            return Ok(());
1057        }
1058        let bit_count = value.to_field().num_bits();
1059
1060        if bit_count > max_bit_size {
1061            let value = value.to_string();
1062            let actual_bits = bit_count;
1063            let max_bits = max_bit_size;
1064
1065            Err(InterpreterError::RangeCheckFailed {
1066                value,
1067                value_id,
1068                actual_bits,
1069                max_bits,
1070                msg: error_message.cloned(),
1071            })
1072        } else {
1073            Ok(())
1074        }
1075    }

1077    fn interpret_call(
1078        &mut self,
1079        function_id: ValueId,
1080        argument_ids: &[ValueId],
1081        results: &[ValueId],
1082        side_effects_enabled: bool,
1083    ) -> IResult<()> {
1084        let function = self.lookup(function_id)?;
1085        let mut arguments = try_vecmap(argument_ids, |argument| self.lookup(*argument))?;
1086
1087        let new_results = if side_effects_enabled {
1088            match function {
1089                Value::Function(id) => {
1090                    // If we're crossing a constrained -> unconstrained boundary we have to wipe
1091                    // any shared mutable fields in our arguments since brillig should conceptually
1092                    // receive fresh array on each invocation.
1093                    if !self.in_unconstrained_context()
1094                        && self.functions[&id].runtime().is_brillig()
1095                    {
1096                        for argument in &mut arguments {
1097                            Self::reset_array_state(argument)?;
1098                        }
1099                    }
1100                    self.call_function(id, arguments)?
1101                }
1102                Value::Intrinsic(intrinsic) => {
1103                    self.call_intrinsic(intrinsic, argument_ids, results)?
1104                }
1105                Value::ForeignFunction(name) if self.options.no_foreign_calls => {
1106                    return Err(InterpreterError::UnknownForeignFunctionCall { name });
1107                }
1108                Value::ForeignFunction(name) if name == "print" => {
1109                    self.check_purity_on_foreign_call(&name)?;
1110                    self.call_print(arguments)?
1111                }
1112                Value::ForeignFunction(name) => {
1113                    return Err(InterpreterError::UnknownForeignFunctionCall { name });
1114                }
1115                other => {
1116                    return Err(internal(InternalError::CalledNonFunction {
1117                        value: other.to_string(),
1118                        value_id: function_id,
1119                    }));
1120                }
1121            }
1122        } else {
1123            self.uninitialized_call_results(&function, argument_ids, results)?
1124        };
1125
1126        if new_results.len() != results.len() {
1127            let function_name = self.try_get_function_name(function_id);
1128            return Err(internal(InternalError::FunctionReturnedIncorrectArgCount {
1129                function: function_id,
1130                function_name,
1131                expected: results.len(),
1132                actual: new_results.len(),
1133            }));
1134        }
1135
1136        for (result, new_result) in results.iter().zip_eq(new_results) {
1137            self.define(*result, new_result)?;
1138        }
1139        Ok(())
1140    }

1142    /// Create uninitialized results for a call that was skipped due to disabled side effects.
1143    ///
1144    /// For vector intrinsics, we create properly-sized zeroed vectors rather than empty ones,
1145    /// to avoid out-of-bounds error after Remove `IfElse` that need to do `array_get` to
1146    /// merge the vector from a 'side effect disabled' branch.
1147    fn uninitialized_call_results(
1148        &self,
1149        function: &Value,
1150        argument_ids: &[ValueId],
1151        results: &[ValueId],
1152    ) -> IResult<Vec<Value>> {
1153        use crate::ssa::ir::instruction::Intrinsic;
1154        // Get the length of the vector
1155        if let Value::Intrinsic(intrinsic) = function {
1156            let input_vector_info = match intrinsic {
1157                Intrinsic::VectorPushBack
1158                | Intrinsic::VectorPushFront
1159                | Intrinsic::VectorInsert
1160                | Intrinsic::VectorPopBack
1161                | Intrinsic::VectorPopFront
1162                | Intrinsic::VectorRemove => {
1163                    let vec = self.lookup_array_or_vector(
1164                        argument_ids[1],
1165                        "uninitialized vector intrinsic",
1166                    )?;
1167                    Some((vec.elements.borrow().len(), vec.element_types.clone()))
1168                }
1169                _ => None,
1170            };
1171
1172            if let Some((input_len, element_types)) = input_vector_info {
1173                let element_count = element_types.len();
1174                let output_len = match intrinsic {
1175                    Intrinsic::VectorPushBack
1176                    | Intrinsic::VectorPushFront
1177                    | Intrinsic::VectorInsert => input_len + element_count,
1178                    Intrinsic::VectorPopBack
1179                    | Intrinsic::VectorPopFront
1180                    | Intrinsic::VectorRemove => input_len.saturating_sub(element_count),
1181                    _ => unreachable!(),
1182                };
1183
1184                return Ok(vecmap(results, |result| {
1185                    let typ = self.dfg().type_of_value(*result);
1186                    if matches!(*typ, Type::Vector(_)) {
1187                        Value::uninitialized_vector(&element_types, output_len, *result)
1188                    } else {
1189                        Value::uninitialized(&typ, *result)
1190                    }
1191                }));
1192            }
1193        }
1194
1195        Ok(vecmap(results, |result| {
1196            let typ = self.dfg().type_of_value(*result);
1197            Value::uninitialized(&typ, *result)
1198        }))
1199    }

1214    /// Reset the value's `Shared` states in each array within. This is used to mimic each
1215    /// invocation of the brillig vm receiving fresh values. No matter the history of this value
1216    /// (e.g. even if they were previously returned from another brillig function) the reference
1217    /// count should always be 1 and it shouldn't alias any other arrays.
1218    fn reset_array_state(value: &mut Value) -> IResult<()> {
1219        match value {
1220            Value::Numeric(_)
1221            | Value::Function(_)
1222            | Value::Intrinsic(_)
1223            | Value::ForeignFunction(_) => Ok(()),
1224
1225            // Immutable references are allowed to cross the constrained->unconstrained
1226            // boundary. Mutable references are rejected earlier by the frontend type check.
1227            Value::Reference(_) => Ok(()),
1228
1229            Value::ArrayOrVector(array_value) => {
1230                let mut elements = array_value.elements.borrow().to_vec();
1231                for element in &mut elements {
1232                    Self::reset_array_state(element)?;
1233                }
1234                array_value.elements = Shared::new(elements);
1235                array_value.rc = Shared::new(1);
1236                Ok(())
1237            }
1238        }
1239    }

1266    fn interpret_store(&self, address: ValueId, value: ValueId) -> IResult<()> {
1267        let reference_address = self.lookup_reference(address, "store")?;
1268
1269        let value = self.lookup(value)?;
1270
1271        if self.options.trace {
1272            println!("store {value} at {address}");
1273        }
1274
1275        let storage = reference_address.element.as_ptr() as StorageIdentity;
1276        // A global is only ever a numeric constant or a `make_array`, so no reference is owned
1277        // by one and this store cannot reach global storage. If that ever changes, the write
1278        // would land in the shared global scope: see [`Self::is_global_storage`].
1279        debug_assert!(!self.is_global_storage(storage), "store through a global reference");
1280        self.check_purity_on_mutation(storage, "a store through a reference")?;
1281
1282        *reference_address.element.borrow_mut() = Some(value);
1283
1284        Ok(())
1285    }
1286
1287    /// In the ACIR runtime a nested array must be a fresh copy rather than a shared handle:
1288    /// `array_get` returns a fresh nested array and `array_set` stores a fresh copy of an
1289    /// array-valued element. Otherwise a later mutable array set on the source array would
1290    /// also mutate the value produced here, since both would share the same `Shared` handle.
1291    /// In the Brillig runtime this aliasing is expected, so the value is cloned as-is.
1292    fn copy_nested_array_in_acir(&self, value: &Value) -> Value {
1293        if !self.in_unconstrained_context()
1294            && let Some(array) = value.as_array_or_vector()
1295        {
1296            return Value::ArrayOrVector(ArrayValue {
1297                elements: Shared::new(array.elements.borrow().to_vec()),
1298                rc: array.rc,
1299                element_types: array.element_types,
1300                length: array.length,
1301            });
1302        }
1303        value.clone()
1304    }

1306    fn interpret_array_get(
1307        &mut self,
1308        array: ValueId,
1309        index: ValueId,
1310        result: ValueId,
1311        side_effects_enabled: bool,
1312    ) -> IResult<()> {
1313        // When there is a problem indexing the array, but side effects are disabled,
1314        // define the value as uninitialized.
1315        let uninitialized = |this: &mut Self| {
1316            let typ = this.dfg().type_of_value(result);
1317            let value = Value::uninitialized(&typ, result);
1318            this.define(result, value)
1319        };
1320
1321        let offset = self.dfg().array_offset(array, index);
1322        let array = self.lookup_array_or_vector(array, "array get")?;
1323        let length = array.elements.borrow().len() as u32;
1324
1325        // Per `Instruction::requires_acir_gen_predicate`, in Brillig an
1326        // `array_get` is pure-in-isolation: the OOB check is inserted as a
1327        // separate constraint, not part of the access itself. Match that here
1328        // so the interpreter agrees with the Brillig VM on dead/unused gets.
1329        let oob_is_pure = self.current_function().runtime().is_brillig();
1330
1331        let index = match self.lookup_array_index(index, "array get index", length) {
1332            Err(InterpreterError::IndexOutOfBounds { .. })
1333                if !side_effects_enabled || oob_is_pure =>
1334            {
1335                return uninitialized(self);
1336            }
1337            other => other?,
1338        };
1339        let mut index = index - offset.to_u32();
1340
1341        if length == 0 {
1342            // Accessing an array of 0-len is replaced by asserting
1343            // the branch is not-taken during acir-gen and
1344            // a zeroed type is used in case of array get
1345            // So we can simply replace it with uninitialized value
1346            if side_effects_enabled && !oob_is_pure {
1347                return Err(InterpreterError::IndexOutOfBounds { index: index.into(), length });
1348            } else {
1349                return uninitialized(self);
1350            }
1351        }
1352
1353        if oob_is_pure && index >= length {
1354            return uninitialized(self);
1355        }
1356
1357        let element = {
1358            // An array_get with false side_effects_enabled is replaced
1359            // by a load at a valid index during acir-gen.
1360            if !side_effects_enabled {
1361                // Find a valid index
1362                let typ = self.dfg().type_of_value(result);
1363                for (i, element) in array.elements.borrow().iter().enumerate() {
1364                    if element.get_type() == *typ {
1365                        index = i as u32;
1366                        break;
1367                    }
1368                }
1369            }
1370            let elements = array.elements.borrow();
1371            let element = elements
1372                .get(index as usize)
1373                .ok_or(InterpreterError::IndexOutOfBounds { index: index.into(), length })?;
1374
1375            self.copy_nested_array_in_acir(element)
1376        };
1377        self.define(result, element)?;
1378        Ok(())
1379    }

1381    #[allow(clippy::too_many_arguments)]
1382    fn interpret_array_set(
1383        &mut self,
1384        array: ValueId,
1385        index: ValueId,
1386        value: ValueId,
1387        mutable: bool,
1388        result: ValueId,
1389        side_effects_enabled: bool,
1390    ) -> IResult<()> {
1391        let offset = self.dfg().array_offset(array, index);
1392        let array = self.lookup_array_or_vector(array, "array set")?;
1393
1394        let result_array = if side_effects_enabled {
1395            let length = array.elements.borrow().len() as u32;
1396            let index = self.lookup_array_index(index, "array set index", length)?;
1397            let index = index - offset.to_u32();
1398            let value = self.copy_nested_array_in_acir(&self.lookup(value)?);
1399
1400            let storage = array.elements.as_ptr() as StorageIdentity;
1401            let is_rc_one = *array.rc.borrow() == 1;
1402            // A global is not the sole live reference to its storage even when its reference
1403            // count says so, so `is_rc_one` alone is not licence to write through it — see
1404            // [`Self::is_global_storage`].
1405            let should_mutate = if self.in_unconstrained_context() {
1406                is_rc_one && !self.is_global_storage(storage)
1407            } else {
1408                mutable
1409            };
1410
1411            if index >= length {
1412                return Err(InterpreterError::IndexOutOfBounds { index: index.into(), length });
1413            }
1414
1415            if should_mutate {
1416                // In a constrained context arrays have value semantics: an in-place write
1417                // here only reuses the backing store of an array value that the Mutable
1418                // Array Set Optimizations pass proved dead, so it is not a caller-visible
1419                // mutation and purity analysis rightly ignores it. Only in Brillig, where
1420                // the reference count governs genuine sharing, is writing through
1421                // argument-reachable storage observable by the caller.
1422                if self.in_unconstrained_context() {
1423                    self.check_purity_on_mutation(storage, "an in-place array_set")?;
1424                }
1425                array.elements.borrow_mut()[index as usize] = value;
1426                Value::ArrayOrVector(array)
1427            } else {
1428                if !is_rc_one {
1429                    Self::decrement_rc(&array);
1430                }
1431                let mut elements = array.elements.borrow().to_vec();
1432                elements[index as usize] = value;
1433                let elements = Shared::new(elements);
1434                let rc = Shared::new(1);
1435                let element_types = array.element_types.clone();
1436                let length = array.length;
1437                Value::ArrayOrVector(ArrayValue { elements, rc, element_types, length })
1438            }
1439        } else {
1440            // Side effects are disabled, return the original array
1441            Value::ArrayOrVector(array)
1442        };
1443        self.define(result, result_array)?;
1444        Ok(())
1445    }

1480    fn interpret_if_else(
1481        &mut self,
1482        then_condition_id: ValueId,
1483        then_value: ValueId,
1484        else_condition_id: ValueId,
1485        else_value: ValueId,
1486        result: ValueId,
1487    ) -> IResult<()> {
1488        let then_condition = self.lookup_bool(then_condition_id, "then condition")?;
1489        let else_condition = self.lookup_bool(else_condition_id, "else condition")?;
1490        let then_value = self.lookup(then_value)?;
1491        let else_value = self.lookup(else_value)?;
1492
1493        // Note that `then_condition = !else_condition` doesn't always hold!
1494        // Notably if this is a nested if expression we could have something like:
1495        //   then_condition = outer_condition & a
1496        //   else_condition = outer_condition & !a
1497        // If `outer_condition` is false, both will be false.
1498        if then_condition && else_condition {
1499            return Err(InterpreterError::DoubleTrueIfElse {
1500                then_condition_id,
1501                else_condition_id,
1502            });
1503        }
1504
1505        let new_result = if !then_condition && !else_condition {
1506            // Returning uninitialized/zero if both conditions are false to match
1507            // the decomposition of `cond * then_value + !cond * else_value` for numeric values.
1508            let typ = self.dfg().type_of_value(result);
1509            Value::uninitialized(&typ, result)
1510        } else if then_condition {
1511            then_value
1512        } else {
1513            else_value
1514        };
1515
1516        self.define(result, new_result)
1517    }

1519    fn interpret_make_array(
1520        &mut self,
1521        elements: &imbl::Vector<ValueId>,
1522        result: ValueId,
1523        result_type: &Type,
1524    ) -> IResult<()> {
1525        let elements = try_vecmap(elements, |element| self.lookup(*element))?;
1526        let length = if let Type::Array(_, length) = result_type { Some(*length) } else { None };
1527
1528        // The number of elements in the array must be a multiple of the number of element types
1529        let element_types = result_type.element_types();
1530        if element_types.is_empty() {
1531            if !elements.is_empty() {
1532                return Err(internal(InternalError::MakeArrayElementCountMismatch {
1533                    result,
1534                    elements_count: elements.len(),
1535                    types_count: element_types.len(),
1536                }));
1537            }
1538        } else if elements.len() % element_types.len() != 0 {
1539            return Err(internal(InternalError::MakeArrayElementCountMismatch {
1540                result,
1541                elements_count: elements.len(),
1542                types_count: element_types.len(),
1543            }));
1544        }
1545
1546        // Make sure each element's type matches the one in element_types
1547        for (index, (element, expected_type)) in
1548            elements.iter().zip(element_types.iter().cycle()).enumerate()
1549        {
1550            let actual_type = element.get_type();
1551            if !actual_type.can_be_used_as(expected_type) {
1552                return Err(internal(InternalError::MakeArrayElementTypeMismatch {
1553                    result,
1554                    index,
1555                    actual_type: actual_type.to_string(),
1556                    expected_type: expected_type.to_string(),
1557                }));
1558            }
1559        }
1560
1561        let array = Value::ArrayOrVector(ArrayValue {
1562            elements: Shared::new(elements),
1563            rc: Shared::new(1),
1564            element_types,
1565            length,
1566        });
1567        self.define(result, array)
1568    }

1583/// Evaluate an integer (non-`Field`, non-`u1`) binary operation in the field+type model.
1584///
1585/// `lhs`/`rhs` are integer values of the same type whose stored field is the two's-complement bit
1586/// pattern (which may be out of range in ACIR mode). The runtime selects overflow behaviour:
1587/// Brillig wraps in fixed-width registers, while ACIR carries the extended field and range-checks at
1588/// checked operations. Comparisons, bitwise ops, shifts, div/mod and *signed* checked arithmetic
1589/// reduce their operands and reuse [`eval_constant_binary_op`]; only unchecked arithmetic and
1590/// *unsigned* checked arithmetic depend on the runtime.
1591fn evaluate_integer_binary(
1592    binary: &Binary,
1593    lhs: NumericValue,
1594    rhs: NumericValue,
1595    is_brillig: bool,
1596    display_binary: impl Fn(&Binary) -> String,
1597) -> IResult<NumericValue> {
1598    use BinaryOp::{Add, Mul, Sub};
1599
1600    let operator = binary.operator;
1601    let typ = lhs.get_type();
1602    let bit_size = lhs.bit_size();
1603    let lhs_field = lhs.to_field();
1604    let rhs_field = rhs.to_field();
1605
1606    let overflow = || overflow_error(binary, &display_binary, lhs, rhs);
1607
1608    // Field arithmetic for add/sub/mul (no wrapping, no reduction).
1609    let field_arith = || match operator {
1610        Add { .. } => lhs_field + rhs_field,
1611        Sub { .. } => lhs_field - rhs_field,
1612        Mul { .. } => lhs_field * rhs_field,
1613        _ => unreachable!("field_arith called on a non-arithmetic operator"),
1614    };
1615
1616    match operator {
1617        // Unchecked arithmetic: Brillig wraps to the bit size; ACIR extends in the field.
1618        Add { unchecked: true } | Sub { unchecked: true } | Mul { unchecked: true } => {
1619            let result = if is_brillig {
1620                let a = bits_u128(lhs_field, bit_size);
1621                let b = bits_u128(rhs_field, bit_size);
1622                let wrapped = match operator {
1623                    Add { .. } => a.wrapping_add(b),
1624                    Sub { .. } => a.wrapping_sub(b),
1625                    Mul { .. } => a.wrapping_mul(b),
1626                    _ => unreachable!(),
1627                };
1628                FieldElement::from(truncate(wrapped, bit_size))
1629            } else {
1630                field_arith()
1631            };
1632            NumericValue::int_from_field(result, typ)
1633        }
1634
1635        // Unsigned checked arithmetic. ACIR computes in the field and range-checks the result, so an
1636        // out-of-range operand or an overflowing result is rejected; Brillig does true fixed-width
1637        // checked arithmetic. (They only diverge once values can exceed the field modulus, i.e.
1638        // `u128`, but keeping them separate is faithful to both.)
1639        Add { unchecked: false } | Sub { unchecked: false } | Mul { unchecked: false }
1640            if !lhs.is_signed() =>
1641        {
1642            if is_brillig {
1643                eval_via_constant_binary_op(lhs_field, rhs_field, operator, typ, binary, &overflow)
1644            } else {
1645                let value = NumericValue::int_from_field(field_arith(), typ)?;
1646                if value.is_in_range() { Ok(value) } else { Err(overflow()) }
1647            }
1648        }
1649
1650        // Shifts wrap the value to the bit size (a shift amount >= the bit width is an overflow,
1651        // matching `checked_shl`/`checked_shr`); they are not folded as checked overflows.
1652        BinaryOp::Shl | BinaryOp::Shr => {
1653            let value_bits = bits_u128(lhs_field, bit_size);
1654            let shift = bits_u128(rhs_field, bit_size);
1655            if shift >= u128::from(bit_size) {
1656                return Err(overflow());
1657            }
1658            let shift = shift as u32;
1659            let result = match operator {
1660                BinaryOp::Shl => {
1661                    FieldElement::from(truncate(value_bits.wrapping_shl(shift), bit_size))
1662                }
1663                // Arithmetic shift for signed values, logical shift for unsigned.
1664                BinaryOp::Shr if lhs.is_signed() => {
1665                    let signed = try_convert_field_element_to_signed_integer(
1666                        FieldElement::from(value_bits),
1667                        bit_size,
1668                    )
1669                    .expect("a reduced signed value converts");
1670                    convert_signed_integer_to_field_element(signed >> shift, bit_size)
1671                }
1672                BinaryOp::Shr => FieldElement::from(value_bits >> shift),
1673                _ => unreachable!(),
1674            };
1675            NumericValue::int_from_field(result, typ)
1676        }
1677
1678        // Signed checked arithmetic, div/mod, comparisons and bitwise ops all reduce their
1679        // operands; reuse the constant-folder's semantics.
1680        _ => eval_via_constant_binary_op(lhs_field, rhs_field, operator, typ, binary, &overflow),
1681    }
1682}

1684/// Apply [`eval_constant_binary_op`] to two operand fields and map the result into the interpreter's
1685/// value/error types. `eval_constant_binary_op` truncates its operands to the type's bit size, so
1686/// this is the "reduce then compute" path shared by both runtimes.
1687fn eval_via_constant_binary_op(
1688    lhs_field: FieldElement,
1689    rhs_field: FieldElement,
1690    operator: BinaryOp,
1691    typ: NumericType,
1692    binary: &Binary,
1693    overflow: &impl Fn() -> InterpreterError,
1694) -> IResult<NumericValue> {
1695    // Reduce the operands to their bit-pattern representatives first. `eval_constant_binary_op`
1696    // also truncates internally, but its signed conversion requires the field to fit in `u128`,
1697    // which an extended (out-of-range) ACIR operand such as `p - delta` does not.
1698    let bit_size = typ.bit_size::<FieldElement>();
1699    let lhs_field = truncate_field(lhs_field, bit_size);
1700    let rhs_field = truncate_field(rhs_field, bit_size);
1701
1702    match eval_constant_binary_op(lhs_field, rhs_field, operator, typ) {
1703        BinaryEvaluationResult::Success(field, result_type) => {
1704            NumericValue::int_from_field(field, result_type)
1705        }
1706        BinaryEvaluationResult::Failure(_) => {
1707            let divisor_is_zero = matches!(operator, BinaryOp::Div | BinaryOp::Mod)
1708                && truncate_field(rhs_field, typ.bit_size::<FieldElement>()).is_zero();
1709            if divisor_is_zero {
1710                Err(division_by_zero(binary.lhs, lhs_field, binary.rhs, rhs_field))
1711            } else {
1712                Err(overflow())
1713            }
1714        }
1715        // Only an overflowing `shl` reports this; treat it as an overflow.
1716        BinaryEvaluationResult::CouldNotEvaluate => Err(overflow()),
1717    }
1718}

1720fn evaluate_binary(
1721    binary: &Binary,
1722    lhs: NumericValue,
1723    rhs: NumericValue,
1724    side_effects_enabled: bool,
1725    is_brillig: bool,
1726    display_binary: impl Fn(&Binary) -> String,
1727) -> IResult<NumericValue> {
1728    let lhs_id = binary.lhs;
1729    let rhs_id = binary.rhs;
1730
1731    if lhs.get_type() != rhs.get_type() {
1732        return Err(internal(InternalError::MismatchedTypesInBinaryOperator {
1733            lhs_id,
1734            lhs: lhs.to_string(),
1735            operator: binary.operator,
1736            rhs_id,
1737            rhs: rhs.to_string(),
1738        }));
1739    }
1740
1741    // Disable this instruction if it is side-effectual and side effects are disabled.
1742    if !side_effects_enabled {
1743        return Ok(NumericValue::zero(lhs.get_type()));
1744    }
1745
1746    if let (Some(lhs), Some(rhs)) = (lhs.as_field(), rhs.as_field()) {
1747        return interpret_field_binary_op(lhs, binary.operator, rhs, lhs_id, rhs_id);
1748    }
1749
1750    if let (Some(lhs), Some(rhs)) = (lhs.as_bool(), rhs.as_bool()) {
1751        return interpret_u1_binary_op(lhs, rhs, binary, &display_binary);
1752    }
1753
1754    evaluate_integer_binary(binary, lhs, rhs, is_brillig, display_binary)
1755}

1757fn interpret_field_binary_op(
1758    lhs: FieldElement,
1759    operator: BinaryOp,
1760    rhs: FieldElement,
1761    lhs_id: ValueId,
1762    rhs_id: ValueId,
1763) -> IResult<NumericValue> {
1764    let unsupported_operator = |operator| -> IResult<NumericValue> {
1765        let typ = "Field";
1766        Err(internal(InternalError::UnsupportedOperatorForType { operator, typ }))
1767    };
1768
1769    let result = match operator {
1770        BinaryOp::Add { unchecked: _ } => NumericValue::field(lhs + rhs),
1771        BinaryOp::Sub { unchecked: _ } => NumericValue::field(lhs - rhs),
1772        BinaryOp::Mul { unchecked: _ } => NumericValue::field(lhs * rhs),
1773        BinaryOp::Div => {
1774            if rhs.is_zero() {
1775                return Err(division_by_zero(lhs_id, lhs, rhs_id, rhs));
1776            }
1777            NumericValue::field(lhs / rhs)
1778        }
1779        BinaryOp::Mod => return unsupported_operator("%"),
1780        BinaryOp::Eq => NumericValue::bool(lhs == rhs),
1781        BinaryOp::Lt => NumericValue::bool(lhs < rhs),
1782        BinaryOp::And => return unsupported_operator("&"),
1783        BinaryOp::Or => return unsupported_operator("|"),
1784        BinaryOp::Xor => return unsupported_operator("^"),
1785        BinaryOp::Shl => return unsupported_operator("<<"),
1786        BinaryOp::Shr => return unsupported_operator(">>"),
1787    };
1788    Ok(result)
1789}

1791fn interpret_u1_binary_op(
1792    lhs: bool,
1793    rhs: bool,
1794    binary: &Binary,
1795    display_binary: &impl Fn(&Binary) -> String,
1796) -> IResult<NumericValue> {
1797    let overflow = || overflow_error(binary, display_binary, lhs, rhs);
1798
1799    let lhs_id = binary.lhs;
1800    let rhs_id = binary.rhs;
1801
1802    let result = match binary.operator {
1803        BinaryOp::Add { unchecked: true } => lhs ^ rhs,
1804        BinaryOp::Add { unchecked: false } => {
1805            if lhs && rhs {
1806                return Err(overflow());
1807            } else {
1808                lhs ^ rhs
1809            }
1810        }
1811        BinaryOp::Sub { unchecked: true } => {
1812            if !lhs && rhs {
1813                return Err(overflow());
1814            } else {
1815                lhs ^ rhs
1816            }
1817        }
1818        BinaryOp::Sub { unchecked: false } => {
1819            if !lhs && rhs {
1820                return Err(overflow());
1821            } else {
1822                lhs ^ rhs
1823            }
1824        }
1825        BinaryOp::Mul { unchecked: _ } => lhs & rhs, // (*) = (&) for u1
1826        BinaryOp::Div => {
1827            // (0, 0) -> (division by 0)
1828            // (0, 1) -> 0
1829            // (1, 0) -> (division by 0)
1830            // (1, 1) -> 1
1831            if !rhs {
1832                return Err(division_by_zero(lhs_id, u8::from(lhs), rhs_id, u8::from(rhs)));
1833            }
1834            lhs
1835        }
1836        BinaryOp::Mod => {
1837            // (0, 0) -> (division by 0)
1838            // (0, 1) -> 0
1839            // (1, 0) -> (division by 0)
1840            // (1, 1) -> 0
1841            if !rhs {
1842                let lhs = format!("u1 {}", u8::from(lhs));
1843                let rhs = format!("u1 {}", u8::from(rhs));
1844                return Err(division_by_zero(lhs_id, lhs, rhs_id, rhs));
1845            }
1846            false
1847        }
1848        BinaryOp::Eq => lhs == rhs,
1849        // clippy complains when you do `lhs < rhs` and recommends this instead
1850        BinaryOp::Lt => !lhs & rhs,
1851        BinaryOp::And => lhs & rhs,
1852        BinaryOp::Or => lhs | rhs,
1853        BinaryOp::Xor => lhs ^ rhs,
1854        BinaryOp::Shl => {
1855            if rhs {
1856                return Err(overflow());
1857            } else {
1858                lhs
1859            }
1860        }
1861        BinaryOp::Shr => {
1862            if rhs {
1863                return Err(overflow());
1864            } else {
1865                lhs
1866            }
1867        }
1868    };
1869    Ok(NumericValue::bool(result))
1870}

1872#[cfg_attr(not(test), expect(dead_code))]
1873fn truncate_unsigned<T>(value: T, bit_size: u32) -> IResult<T>
1874where
1875    u128: From<T>,
1876    T: TryFrom<u128>,
1877    <T as TryFrom<u128>>::Error: std::fmt::Debug,
1878{
1879    let value_u128 = u128::from(value);
1880    let bit_mask = match bit_size.cmp(&MAX_UNSIGNED_BIT_SIZE) {
1881        Ordering::Less => (1u128 << bit_size) - 1,
1882        Ordering::Equal => u128::MAX,
1883        Ordering::Greater => {
1884            return Err(internal(InternalError::InvalidUnsignedTruncateBitSize { bit_size }));
1885        }
1886    };
1887
1888    let result = value_u128 & bit_mask;
1889    Ok(T::try_from(result).expect(
1890        "The truncated result should always be smaller than or equal to the original `value`",
1891    ))
1892}

1894/// Reduce `field` to its `bit_size`-bit pattern as a `u128`.
1895///
1896/// ACIR operands may carry an extended (out-of-range) field; truncating to the bit size yields the
1897/// canonical bit pattern, which always fits in `u128` for the integer widths the interpreter models.
1898fn bits_u128(field: FieldElement, bit_size: u32) -> u128 {
1899    truncate_field(field, bit_size).try_into_u128().expect("a reduced value fits in u128")
1900}
