1//! The dominator tree of a function, represented as a hash map of each reachable block id to its
2//! immediate dominator.
3//!
4//! Dominator trees are useful for tasks such as identifying back-edges in loop analysis or
5//! calculating dominance frontiers.
6

14/// Dominator tree node. We keep one of these per reachable block.
15#[derive(Clone, Default)]
16struct DominatorTreeNode {
17    /// The block's idx in the control flow graph's reverse post-order
18    reverse_post_order_idx: u32,
19
20    /// The block that immediately dominated that of the node in question.
21    ///
22    /// This will be None for the entry block, which has no immediate dominator.
23    immediate_dominator: Option<BasicBlockId>,
24}
25
26/// The entry and exit time of a block in a depth-first walk of the dominator tree.
27///
28/// Together they bracket the block's whole dominator subtree, which is what makes
29/// [`DominatorTree::dominates`] a pair of integer comparisons: `a` dominates `b` exactly when
30/// `b`'s interval nests inside `a`'s.
31#[derive(Clone, Copy, PartialEq, Eq)]
32struct DfsInterval {
33    entry: u32,
34    exit: u32,
35}
36
37impl DfsInterval {
38    /// The interval of a block that has no place in the dominator tree, either because it is
39    /// unreachable or because the tree was built without dominance queries.
40    ///
41    /// It is empty rather than merely out of range, so no pair of real intervals can be confused
42    /// with it, and [`DominatorTree::dfs_interval`] rejects it before any comparison is made.
43    const ABSENT: Self = Self { entry: u32::MAX, exit: 0 };
44
45    /// Does this interval nest `other` inside itself?
46    fn contains(self, other: Self) -> bool {
47        self.entry <= other.entry && other.exit <= self.exit
48    }
49}
50
51impl DominatorTreeNode {
52    /// Updates the immediate dominator estimate, returning true if it has changed.
53    ///
54    /// This is used internally as a shorthand during `compute_dominator_tree`.
55    pub(self) fn update_estimate(&mut self, immediate_dominator: BasicBlockId) -> bool {
56        let immediate_dominator = Some(immediate_dominator);
57        if self.immediate_dominator == immediate_dominator {
58            false
59        } else {
60            self.immediate_dominator = immediate_dominator;
61            true
62        }
63    }
64}

66/// The dominator tree for a single function.
67#[derive(Default)]
68pub(crate) struct DominatorTree {
69    /// The nodes of the dominator tree
70    ///
71    /// After dominator tree computation has complete, this will contain a node for every
72    /// reachable block, and no nodes for unreachable blocks.
73    nodes: HashMap<BasicBlockId, DominatorTreeNode>,
74
75    /// The depth-first interval of each block, indexed by `BasicBlockId::to_u32`.
76    ///
77    /// Block ids are dense, so a flat array keeps `dominates` down to two loads and two integer
78    /// comparisons; reaching the intervals through `nodes` would put a hash lookup on either side
79    /// of that. Blocks with no place in the tree hold [`DfsInterval::ABSENT`], and the array is
80    /// empty when the tree was built with [`DominanceQueries::Disabled`].
81    dfs_intervals: Vec<DfsInterval>,
82}
83
84/// Whether a [`DominatorTree`] should be able to answer [`DominatorTree::dominates`].
85///
86/// The intervals that make that query cheap are a linear pass over the tree at construction, and
87/// several callers only ever ask for [`DominatorTree::immediate_dominator`] or
88/// [`DominatorTree::common_dominator`], which the tree answers without them.
89#[derive(Clone, Copy, PartialEq, Eq)]
90pub(crate) enum DominanceQueries {
91    Enabled,
92    Disabled,
93}

95/// Methods for querying the dominator tree.
96impl DominatorTree {
97    /// Is `block_id` reachable from the entry block?
98    pub(crate) fn is_reachable(&self, block_id: BasicBlockId) -> bool {
99        self.nodes.contains_key(&block_id)
100    }

102    /// Returns the immediate dominator of `block_id`.
103    ///
104    /// A block is said to *dominate* `block_id` if all control flow paths from the function
105    /// entry to `block_id` must go through the block.
106    ///
107    /// The *immediate dominator* is the dominator that is closest to `block_id`. All other
108    /// dominators also dominate the immediate dominator.
109    ///
110    /// This returns `None` if `block_id` is not reachable from the entry block, or if it is the
111    /// entry block which has no dominators.
112    pub(crate) fn immediate_dominator(&self, block_id: BasicBlockId) -> Option<BasicBlockId> {
113        let node = self.nodes.get(&block_id)?;
114        node.immediate_dominator
115    }

117    /// Compare two blocks relative to the reverse post-order.
118    pub(crate) fn reverse_post_order_cmp(&self, a: BasicBlockId, b: BasicBlockId) -> Ordering {
119        match (self.reverse_post_order_idx(a), self.reverse_post_order_idx(b)) {
120            (Some(a), Some(b)) => a.cmp(&b),
121            _ => unreachable!("Post order for unreachable block is undefined"),
122        }
123    }

125    /// Position in the Reverse Post-Order.
126    pub(crate) fn reverse_post_order_idx(&self, block_id: BasicBlockId) -> Option<u32> {
127        self.nodes.get(&block_id).map(|n| n.reverse_post_order_idx)
128    }

130    /// Returns `true` if `block_a_id` dominates `block_b_id`.
131    ///
132    /// This means that every control-flow path from the function entry to `block_b_id` must go
133    /// through `block_a_id`.
134    ///
135    /// This function panics if either of the blocks are unreachable.
136    ///
137    /// A block is considered to dominate itself.
138    pub(crate) fn dominates(&self, block_a_id: BasicBlockId, block_b_id: BasicBlockId) -> bool {
139        // `a` dominates `b` exactly when `b` sits in `a`'s subtree of the dominator tree, and a
140        // depth-first walk brackets each subtree in a contiguous interval, so subtree membership
141        // is interval containment. Reflexive because a block's own interval contains itself.
142        self.dfs_interval(block_a_id).contains(self.dfs_interval(block_b_id))
143    }
144
145    fn dfs_interval(&self, block_id: BasicBlockId) -> DfsInterval {
146        match self.dfs_intervals.get(block_id.to_u32() as usize) {
147            Some(&interval) if interval != DfsInterval::ABSENT => interval,
148            _ => self.no_dfs_interval(block_id),
149        }
150    }

152    /// Panics with whichever of the two reasons a block can have no interval applies here.
153    #[cold]
154    #[inline(never)]
155    fn no_dfs_interval(&self, block_id: BasicBlockId) -> ! {
156        assert!(
157            !self.dfs_intervals.is_empty() || self.nodes.is_empty(),
158            "`dominates` needs a dominator tree built with `DominanceQueries::Enabled`"
159        );
160        panic!("Dominance for unreachable block {block_id} is undefined");
161    }

163    /// Walk up the dominator tree until we find a block for which `f` returns `Some` value.
164    /// Otherwise return `None` when we reach the top.
165    ///
166    /// Similar to `Iterator::filter_map` but only returns the first hit.
167    pub(crate) fn find_map_dominator<T>(
168        &self,
169        mut block_id: BasicBlockId,
170        f: impl Fn(BasicBlockId) -> Option<T>,
171    ) -> Option<T> {
172        if !self.is_reachable(block_id) {
173            return None;
174        }
175        loop {
176            if let Some(value) = f(block_id) {
177                return Some(value);
178            }
179            block_id = self.immediate_dominator(block_id)?;
180        }
181    }

222    /// Build a dominator tree from a control flow graph using Keith D. Cooper's
223    /// "Simple, Fast Dominator Algorithm."
224    fn compute_dominator_tree(
225        &mut self,
226        cfg: &ControlFlowGraph,
227        post_order: &PostOrder,
228        queries: DominanceQueries,
229    ) {
230        // We'll be iterating over a reverse post-order of the CFG, skipping the entry block.
231        let Some((entry_block_id, entry_free_post_order)) = post_order.as_slice().split_last()
232        else {
233            return;
234        };
235
236        // Do a first pass where we assign reverse post-order indices to all reachable nodes. The
237        // entry block will be the only node with no immediate dominator.
238        self.nodes.insert(
239            *entry_block_id,
240            DominatorTreeNode { reverse_post_order_idx: 0, immediate_dominator: None },
241        );
242        for (i, &block_id) in entry_free_post_order.iter().rev().enumerate() {
243            // Indices have been displaced by 1 by the removal of the entry node
244            let reverse_post_order_idx = i as u32 + 1;
245
246            // Due to the nature of the post-order traversal, every node we visit will have at
247            // least one predecessor that has previously been assigned during this loop.
248            let immediate_dominator = self.compute_immediate_dominator(block_id, cfg);
249            self.nodes.insert(
250                block_id,
251                DominatorTreeNode {
252                    immediate_dominator: Some(immediate_dominator),
253                    reverse_post_order_idx,
254                },
255            );
256        }
257
258        // Now that we have reverse post-order indices for everything and initial immediate
259        // dominator estimates, iterate until convergence.
260        //
261        // If the function is free of irreducible control flow, this will exit after one iteration.
262        let mut changed = true;
263        while changed {
264            changed = false;
265            for &block_id in entry_free_post_order.iter().rev() {
266                let immediate_dominator = self.compute_immediate_dominator(block_id, cfg);
267                changed |= self
268                    .nodes
269                    .get_mut(&block_id)
270                    .expect("Assigned in first pass")
271                    .update_estimate(immediate_dominator);
272            }
273        }
274
275        if queries == DominanceQueries::Enabled {
276            self.compute_dfs_intervals(*entry_block_id);
277        }
278    }
279
280    /// Number every node with the entry and exit time of a depth-first walk of the dominator tree,
281    /// so that dominance is interval containment rather than a walk up the tree.
282    ///
283    /// The walk is never actually performed. A block's immediate dominator always precedes it in
284    /// the reverse post-order, so one descending pass over that order accumulates subtree sizes
285    /// into parents, and one ascending pass hands each node the next free slot in its parent's
286    /// interval — which is the depth-first numbering, in two linear scans of a flat array.
287    fn compute_dfs_intervals(&mut self, entry_block: BasicBlockId) {
288        let num_nodes = self.nodes.len();
289        debug_assert_eq!(self.reverse_post_order_idx(entry_block), Some(0));
290
291        // Most functions are a single block, which is its own whole subtree. Saying so here keeps
292        // the scratch buffers below off the common path entirely.
293        if num_nodes == 1 {
294            self.dfs_intervals = vec![DfsInterval::ABSENT; entry_block.to_u32() as usize + 1];
295            self.dfs_intervals[entry_block.to_u32() as usize] = DfsInterval { entry: 0, exit: 0 };
296            return;
297        }
298
299        // The blocks in reverse post-order, and the reverse post-order index of each one's
300        // immediate dominator. The entry block's parent slot is unused.
301        let mut blocks = vec![entry_block; num_nodes];
302        let mut parents = vec![0u32; num_nodes];
303        for (&block_id, node) in &self.nodes {
304            let idx = node.reverse_post_order_idx as usize;
305            blocks[idx] = block_id;
306            if let Some(immediate_dominator) = node.immediate_dominator {
307                parents[idx] = self
308                    .reverse_post_order_idx(immediate_dominator)
309                    .expect("Immediate dominator is a reachable block");
310            }
311        }
312        debug_assert_eq!(
313            blocks.iter().collect::<HashSet<_>>().len(),
314            num_nodes,
315            "reverse post-order indices must be a permutation of 0..n"
316        );
317
318        // Children precede their parents here, so every subtree is complete before it is counted
319        // into the subtree above it.
320        let mut subtree_sizes = vec![1u32; num_nodes];
321        for idx in (1..num_nodes).rev() {
322            let parent = parents[idx] as usize;
323            debug_assert!(
324                parent < idx,
325                "a block's immediate dominator must precede it in the reverse post-order: both \
326                 scans below read their parents' slots before writing their own, so a parent that \
327                 sorted after its child silently corrupts the numbering for the whole function"
328            );
329            subtree_sizes[parent] += subtree_sizes[idx];
330        }
331
332        // And parents precede their children here, so each node's interval is already open when
333        // its children come to claim slots inside it.
334        let capacity = blocks.iter().map(|block| block.to_u32() as usize + 1).max().unwrap_or(0);
335        self.dfs_intervals = vec![DfsInterval::ABSENT; capacity];
336        let mut next_free_slot = vec![0u32; num_nodes];
337        next_free_slot[0] = 1;
338        self.dfs_intervals[entry_block.to_u32() as usize] =
339            DfsInterval { entry: 0, exit: num_nodes as u32 - 1 };
340        for idx in 1..num_nodes {
341            let parent = parents[idx] as usize;
342            let entry = next_free_slot[parent];
343            next_free_slot[parent] += subtree_sizes[idx];
344            next_free_slot[idx] = entry + 1;
345            self.dfs_intervals[blocks[idx].to_u32() as usize] =
346                DfsInterval { entry, exit: entry + subtree_sizes[idx] - 1 };
347        }
348
349        // Every node handed out exactly as many slots as its subtree has members, so the
350        // intervals nest and the root's spans them all. This catches any slip in the arithmetic
351        // above, where a wrong subtree size stays a perfectly plausible-looking interval.
352        debug_assert!((0..num_nodes).all(|idx| {
353            let interval = self.dfs_intervals[blocks[idx].to_u32() as usize];
354            next_free_slot[idx] == interval.entry + subtree_sizes[idx]
355                && interval.exit == interval.entry + subtree_sizes[idx] - 1
356        }));
357    }

359    // Compute the immediate dominator for `block_id` using the pre-calculate immediate dominators
360    // of reachable nodes.
361    fn compute_immediate_dominator(
362        &self,
363        block_id: BasicBlockId,
364        cfg: &ControlFlowGraph,
365    ) -> BasicBlockId {
366        // Get an iterator with just the reachable, already visited predecessors to `block_id`.
367        // Note that during the first pass `node` was pre-populated with all reachable blocks.
368        let mut reachable_predecessors =
369            cfg.predecessors(block_id).filter(|pred_id| self.nodes.contains_key(pred_id));
370
371        // This function isn't called on unreachable blocks or the entry block, so the reverse
372        // post-order will contain at least one predecessor to this block.
373        let mut immediate_dominator =
374            reachable_predecessors.next().expect("block node must have one reachable predecessor");
375
376        for predecessor in reachable_predecessors {
377            immediate_dominator = self.common_dominator(immediate_dominator, predecessor);
378        }
379
380        immediate_dominator
381    }

383    /// Compute the common dominator of two basic blocks.
384    ///
385    /// Both basic blocks are assumed to be reachable.
386    pub(crate) fn common_dominator(
387        &self,
388        mut block_a_id: BasicBlockId,
389        mut block_b_id: BasicBlockId,
390    ) -> BasicBlockId {
391        loop {
392            match self.reverse_post_order_cmp(block_a_id, block_b_id) {
393                Ordering::Less => {
394                    // "a" comes before "b" in the reverse post-order. Move "b" up.
395                    block_b_id = self.nodes[&block_b_id]
396                        .immediate_dominator
397                        .expect("Unreachable basic block?");
398                }
399                Ordering::Greater => {
400                    // "b" comes before "a" in the reverse post-order. Move "a" up.
401                    block_a_id = self.nodes[&block_a_id]
402                        .immediate_dominator
403                        .expect("Unreachable basic block?");
404                }
405                Ordering::Equal => break,
406            }
407        }
408
409        assert_eq!(block_a_id, block_b_id, "Unreachable block passed to common_dominator?");
410        block_a_id
411    }

413    /// Computes the dominance frontier for all blocks in the dominator tree.
414    ///
415    /// The Dominance Frontier of a basic block X is the set of all blocks that are immediate
416    /// successors to blocks dominated by X, but which aren’t themselves strictly dominated by X.
417    /// It is the set of blocks that are not dominated X, and which are “first reached” on paths from X.
418    ///
419    /// For example in the following CFG the DF of B is {E}, because B dominates {C},
420    /// but it's just one edge away from dominating E, as there is another path to E through D.
421    /// ```text
422    ///    A
423    ///   / \
424    ///  B   D
425    ///  |   |
426    ///  C   |
427    ///   \ /
428    ///    E
429    /// ```
430    ///
431    /// This method uses the algorithm specified in Cooper, Keith D. et al. “A Simple, Fast Dominance Algorithm.” (1999).
432    /// As referenced in the paper a dominance frontier is the set of all CFG nodes, y, such that
433    /// b dominates a predecessor of y but does not strictly dominate y.
434    ///
435    /// This method expects the appropriate CFG depending on whether we are operating over
436    /// a dominator tree (standard CFG) or a post-dominator tree (reversed CFG).
437    /// Calling this method on a dominator tree will return a function's dominance frontiers,
438    /// while on a post-dominator tree the method will return the function's reverse (or post) dominance frontiers.
439    ///
440    /// Note: this variant filters out back-edges, so loop headers are NOT included in the
441    /// frontier of loop body blocks. Use `compute_dominance_frontiers_with_back_edges` for
442    /// the standard definition needed by SSA construction (block parameter placement).
443    pub(crate) fn compute_dominance_frontiers(
444        &self,
445        cfg: &ControlFlowGraph,
446    ) -> HashMap<BasicBlockId, HashSet<BasicBlockId>> {
447        self.compute_dominance_frontiers_inner(cfg, false)
448    }
449
450    /// Compute dominance frontiers using the standard definition (Cytron et al. 1991).
451    ///
452    /// Unlike `compute_dominance_frontiers`, this includes loop headers in the frontier of
453    /// loop body blocks. This matches the standard definition: DF(X) = { Y | ∃ pred Z of Y:
454    /// X dom Z ∧ X !sdom Y }. The standard definition is required for correct block parameter
455    /// placement during SSA construction (e.g., in mem2reg).
456    pub(crate) fn compute_dominance_frontiers_with_back_edges(
457        &self,
458        cfg: &ControlFlowGraph,
459    ) -> HashMap<BasicBlockId, HashSet<BasicBlockId>> {
460        self.compute_dominance_frontiers_inner(cfg, true)
461    }
462
463    fn compute_dominance_frontiers_inner(
464        &self,
465        cfg: &ControlFlowGraph,
466        include_back_edges: bool,
467    ) -> HashMap<BasicBlockId, HashSet<BasicBlockId>> {
468        let mut dominance_frontiers: HashMap<BasicBlockId, HashSet<BasicBlockId>> =
469            HashMap::default();
470
471        let nodes = self.nodes.keys().copied().collect::<Vec<_>>();
472        // Find out about each block which dominance frontiers they belong to, if any.
473        for block_id in nodes {
474            let predecessors = cfg.predecessors(block_id);
475            // Dominance frontier nodes must have more than one predecessor. They are join points in the CFG.
476            if predecessors.len() <= 1 {
477                continue;
478            }
479            let Some(immediate_dominator) = self.immediate_dominator(block_id) else {
480                continue;
481            };
482            // Iterate over the predecessors of the current block and walk backwards from them in the dominator tree.
483            for pred_id in predecessors {
484                let mut runner = pred_id;
485                loop {
486                    // Once we reach the immediate dominator of the current block, we know the current block
487                    // won't be in the frontier of any further blocks (frontier blocks are *not* dominated by them).
488                    if immediate_dominator == runner {
489                        break;
490                    }
491                    // Checking if the current block dominates the predecessor;
492                    // for example a loop header has the loop body as one of its predecessors, which it dominates,
493                    // but we don't consider following back-edges as alternative paths on which we reach the header first.
494                    //
495                    // When `include_back_edges` is true (standard SSA definition), we skip this check
496                    // so loop headers ARE included in the frontier of loop body blocks. This is needed
497                    // for correct block parameter placement in mem2reg.
498                    if !include_back_edges && self.dominates(block_id, runner) {
499                        break;
500                    }
501                    dominance_frontiers.entry(runner).or_default().insert(block_id);
502                    // Continue walking backwards to the dominators of the runner, which also have the
503                    // current block in their frontier, unless they dominate it.
504                    let Some(runner_immediate_dom) = self.immediate_dominator(runner) else {
505                        break;
506                    };
507                    runner = runner_immediate_dom;
508                }
509            }
510        }
511
512        dominance_frontiers
513    }

1065    /// Deterministic xorshift so a failure reproduces from its seed alone.
1066    struct Rng(u64);

1068    impl Rng {
1069        fn next_u64(&mut self) -> u64 {
1070            let mut x = self.0;
1071            x ^= x << 13;
1072            x ^= x >> 7;
1073            x ^= x << 17;
1074            self.0 = x;
1075            x
1076        }
1077
1078        fn below(&mut self, n: usize) -> usize {
1079            (self.next_u64() % n as u64) as usize
1080        }

1083    /// Build a function whose control flow graph is arbitrary: any block may jump to any block,
1084    /// which produces self loops, irreducible loops, multiple exits and unreachable blocks.
1085    fn random_function(seed: u64, num_blocks: usize) -> Ssa {
1086        let mut rng = Rng(seed);
1087        let mut builder = FunctionBuilder::new("func".into(), Id::test_new(0));
1088
1089        let mut blocks = vec![builder.current_block()];
1090        for _ in 1..num_blocks {
1091            blocks.push(builder.insert_block());
1092        }
1093
1094        // Targets are drawn from the non-entry blocks: a jump back to the entry leaves the
1095        // post-order empty and the dominator tree unpopulated, which tests nothing.
1096        let targets = &blocks[1..];
1097        let condition = builder.numeric_constant(1u128, NumericType::bool());
1098        for &block in &blocks {
1099            builder.switch_to_block(block);
1100            match rng.below(10) {
1101                0..=1 => builder.terminate_with_return(vec![]),
1102                2..=4 => {
1103                    let target = targets[rng.below(targets.len())];
1104                    builder.terminate_with_jmp(target, vec![]);
1105                }
1106                _ => {
1107                    let then_target = targets[rng.below(targets.len())];
1108                    let else_target = targets[rng.below(targets.len())];
1109                    builder.terminate_with_jmpif_no_args(condition, then_target, else_target);
1110                }
1111            }
1112        }

1117    /// Dominator sets by iterative dataflow: `dom(entry) = {entry}` and
1118    /// `dom(b) = {b} + intersection of dom(p) over p in preds(b)`, iterated to a fixpoint.
1119    ///
1120    /// Restricted to the node set the dominator tree itself covers (the post-order), so that both
1121    /// sides are answering the same question on the same graph.
1122    fn reference_dominator_sets(
1123        cfg: &ControlFlowGraph,
1124        post_order: &[BasicBlockId],
1125    ) -> HashMap<BasicBlockId, Vec<bool>> {
1126        let index_of: HashMap<BasicBlockId, usize> =
1127            post_order.iter().enumerate().map(|(i, &b)| (b, i)).collect();
1128        let num_nodes = post_order.len();
1129        let entry = *post_order.last().expect("non-empty post-order");
1130
1131        let mut dominators: HashMap<BasicBlockId, Vec<bool>> =
1132            post_order.iter().map(|&b| (b, vec![true; num_nodes])).collect();
1133        let mut entry_set = vec![false; num_nodes];
1134        entry_set[index_of[&entry]] = true;
1135        dominators.insert(entry, entry_set);
1136
1137        let mut changed = true;
1138        while changed {
1139            changed = false;
1140            // Reverse post-order, entry first, so information propagates fast.
1141            for &block in post_order.iter().rev() {
1142                if block == entry {
1143                    continue;
1144                }
1145                let mut new_set: Option<Vec<bool>> = None;
1146                for predecessor in cfg.predecessors(block) {
1147                    let Some(predecessor_set) = dominators.get(&predecessor) else {
1148                        continue; // Not covered by the post-order.
1149                    };
1150                    match &mut new_set {
1151                        None => new_set = Some(predecessor_set.clone()),
1152                        Some(set) => {
1153                            for (slot, dominated) in set.iter_mut().zip(predecessor_set) {
1154                                *slot &= *dominated;
1155                            }
1156                        }
1157                    }
1158                }
1159                let mut new_set = new_set.unwrap_or_else(|| vec![false; num_nodes]);
1160                new_set[index_of[&block]] = true;
1161                if dominators[&block] != new_set {
1162                    dominators.insert(block, new_set);
1163                    changed = true;
1164                }

1171    /// The dominance query as it was before the depth-first interval numbering: walk up the
1172    /// dominator tree from `b` until we meet or pass `a` in the reverse post-order.
1173    fn dominates_by_walking(
1174        tree: &DominatorTree,
1175        block_a_id: BasicBlockId,
1176        mut block_b_id: BasicBlockId,
1177    ) -> bool {
1178        loop {
1179            match tree.reverse_post_order_cmp(block_a_id, block_b_id) {
1180                Ordering::Less => {
1181                    block_b_id = match tree.immediate_dominator(block_b_id) {
1182                        Some(immediate_dominator) => immediate_dominator,
1183                        None => return false,
1184                    }
1185                }
1186                Ordering::Greater => return false,
1187                Ordering::Equal => return true,
1188            }
1189        }
1190    }

1192    fn check_tree_against_reference(
1193        tree: &DominatorTree,
1194        cfg: &ControlFlowGraph,
1195        post_order: &[BasicBlockId],
1196        context: &str,
1197    ) {
1198        let index_of: HashMap<BasicBlockId, usize> =
1199            post_order.iter().enumerate().map(|(i, &b)| (b, i)).collect();
1200        let dominators = reference_dominator_sets(cfg, post_order);
1201
1202        for &b in post_order {
1203            for &a in post_order {
1204                let expected = dominators[&b][index_of[&a]];
1205                assert_eq!(
1206                    tree.dominates(a, b),
1207                    expected,
1208                    "{context}: dominates({a}, {b}) disagrees with the reference dominator sets"
1209                );
1210                assert_eq!(
1211                    dominates_by_walking(tree, a, b),
1212                    expected,
1213                    "{context}: the reference dominator sets disagree with the tree walk for \
1214                     ({a}, {b}), so the reference itself is suspect"
1215                );
1216            }
1217
1218            // The immediate dominator is the strict dominator dominated by every other strict
1219            // dominator of the block.
1220            let strict_dominators: Vec<_> = post_order
1221                .iter()
1222                .copied()
1223                .filter(|&a| a != b && dominators[&b][index_of[&a]])
1224                .collect();
1225            let expected_immediate = strict_dominators
1226                .iter()
1227                .copied()
1228                .find(|&a| strict_dominators.iter().all(|&other| dominators[&a][index_of[&other]]));
1229            assert_eq!(
1230                tree.immediate_dominator(b),
1231                expected_immediate,
1232                "{context}: immediate_dominator({b}) is wrong"
1233            );
1234        }
1235    }

1276    /// A CFG is irreducible when some loop has more than one entry, which shows up as a
1277    /// back edge whose target does not dominate its source.
1278    fn is_irreducible(cfg: &ControlFlowGraph, post_order: &[BasicBlockId]) -> bool {
1279        let index_of: HashMap<BasicBlockId, usize> =
1280            post_order.iter().rev().enumerate().map(|(i, &b)| (b, i)).collect();
1281        let dominators = reference_dominator_sets(cfg, post_order);
1282        let forward_index: HashMap<BasicBlockId, usize> =
1283            post_order.iter().enumerate().map(|(i, &b)| (b, i)).collect();
1284        for (&block, &block_index) in &index_of {
1285            for predecessor in cfg.predecessors(block) {
1286                let Some(&predecessor_index) = index_of.get(&predecessor) else { continue };
1287                // A back edge in the reverse post-order whose target does not dominate its source.
1288                if predecessor_index >= block_index
1289                    && !dominators[&predecessor][forward_index[&block]]
1290                {
1291                    return true;
1292                }
1293            }
1294        }
1295        false
1296    }

1338    /// `ControlFlowGraph::extended_reverse` wires every block that cannot reach an exit into a
1339    /// synthesised exit node, and asserts on the way that no block gains a third successor. A
1340    /// block sitting in an infinite loop behind a conditional branch already has two, so these
1341    /// graphs are excluded here rather than tripping an assertion unrelated to dominance.
1342    fn extended_reverse_shape(func: &Function) -> (bool, bool) {
1343        let cfg = ControlFlowGraph::with_function(func);
1344        let exits: Vec<_> = func
1345            .reachable_blocks()
1346            .into_iter()
1347            .filter(|&block| cfg.successors(block).len() == 0)
1348            .collect();
1349        let reverse = cfg.reverse();
1350        let reaches_an_exit: std::collections::HashSet<_> =
1351            PostOrder::with_cfg(&reverse).into_vec().into_iter().collect();
1352        let dead: Vec<_> = func
1353            .reachable_blocks()
1354            .into_iter()
1355            .filter(|block| !reaches_an_exit.contains(block))
1356            .collect();
1357        let needs_exit = exits.len() > 1 || !dead.is_empty();
1358        let would_ice = needs_exit && dead.iter().any(|&block| cfg.successors(block).len() >= 2);
1359        (would_ice, needs_exit)
1360    }
