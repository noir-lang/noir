1//! The registry of lints understood by the compiler.
2//!
3//! A "lint" is an opinionated warning: a diagnostic that flags legitimate but
4//! usually-undesirable code (an unused item, an unnecessary `mut`, a constant
5//! return value, …). Each lint has a stable, human-readable slug (`dead_code`,
6//! `unused_variables`, …) that a user can name in an `#[allow(...)]` attribute
7//! to silence it.
8//!
9//! [`Lint`] is the closed set of valid slugs. Making the set closed is what lets the
10//! parser reject an unrecognised slug: parsing `#[allow(<slug>)]`
11//! (`parser/parser/attributes.rs`) looks the slug up with [`Lint::from_slug`], and an
12//! unknown slug raises the `UnknownLint` warning rather than being accepted as an inert
13//! no-op the author mistakes for a working suppression. That warning is a parser warning,
14//! which never blocks elaboration, so the lint the author *meant* to silence still fires.
15//!
16//! Consumers that honour `#[allow(...)]` compare against a [`Lint`] (via
17//! [`crate::token::Attributes::has_allow`] / `SecondaryAttributeKind::is_allow`) rather
18//! than a bare string, so the slug spellings live only in [`Lint::slug`].
19//!
20//! [`Lint`] lists only the lints `#[allow(...)]` actually silences: a slug is "known" only
21//! when naming it has a real effect. See `design/lints.md` for the rationale and the
22//! intended direction.
23
24use std::fmt;
25
26use strum::IntoEnumIterator;
27use strum_macros::EnumIter;
28
29/// A lint the compiler knows about and that can be named in `#[allow(...)]`.
30#[derive(Debug, Copy, Clone, PartialEq, Eq, EnumIter)]
31pub enum Lint {
32    /// Item is never used (`dead_code`).
33    DeadCode,
34    /// Local variable is never used (`unused_variables`).
35    UnusedVariables,
36    /// Binding is marked `mut` but never mutated (`unused_mut`).
37    UnusedMut,
38    /// Function always returns a constant (`constant_return`).
39    ConstantReturn,
40}
41
42impl Lint {
43    /// The stable, human-readable slug used to refer to this lint in source
44    /// (e.g. `#[allow(dead_code)]`).
45    pub fn slug(self) -> &'static str {
46        match self {
47            Lint::DeadCode => "dead_code",
48            Lint::UnusedVariables => "unused_variables",
49            Lint::UnusedMut => "unused_mut",
50            Lint::ConstantReturn => "constant_return",
51        }
52    }
53
54    /// Look up a lint by its slug, returning `None` if the slug is not recognised.
55    pub fn from_slug(slug: &str) -> Option<Lint> {
56        Lint::iter().find(|lint| lint.slug() == slug)
57    }
58}
59
60impl fmt::Display for Lint {
61    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
62        f.write_str(self.slug())
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::Lint;
69    use strum::IntoEnumIterator;
70
71    #[test]
72    fn slug_round_trips() {
73        for lint in Lint::iter() {
74            assert_eq!(Lint::from_slug(lint.slug()), Some(lint));
75        }
76    }
77
78    #[test]
79    fn unknown_slug_is_none() {
80        assert_eq!(Lint::from_slug("dead_cod"), None);
81        assert_eq!(Lint::from_slug(""), None);
82    }
