1//! Defines [ComptimeScopes], the local variables of the comptime code being interpreted.
2use rustc_hash::FxHashMap as HashMap;
3
4use crate::node_interner::DefinitionId;
5
6use super::Value;
7
8/// The comptime local variables in scope, one map per scope with the innermost scope last.
9///
10/// The stack spans every comptime function currently being interpreted, not just the innermost
11/// one. Rather than physically removing a caller's scopes when a function is entered, the
12/// `floor` is raised to the top of the stack so that only the callee's own scopes are visible.
13/// This keeps entering a function O(1) regardless of how deep the call stack is.
14///
15/// The runtime elaborator pushes a scope here for every block it enters as well, so that a
16/// `comptime let` in a runtime block is dropped when the block ends.
17///
18/// Globals are not stored here: their values live in the
19/// `GlobalInfo` of each global in the [`NodeInterner`](crate::node_interner::NodeInterner).
20#[derive(Debug)]
21pub(crate) struct ComptimeScopes {
22    /// Never empty: the bottom scope holds variables defined outside of any block and is
23    /// never popped.
24    scopes: Vec<HashMap<DefinitionId, Value>>,
25
26    /// Index into `scopes` of the first scope visible to the function currently being
27    /// interpreted. Scopes below it belong to enclosing callers.
28    floor: usize,
29}
30
31impl Default for ComptimeScopes {
32    fn default() -> Self {
33        Self { scopes: vec![HashMap::default()], floor: 0 }
34    }
35}
36
37impl ComptimeScopes {
38    pub(crate) fn push(&mut self) {
39        self.scopes.push(HashMap::default());
40    }
41
42    /// Pops the innermost scope.
43    ///
44    /// Panics if that would pop the bottom scope.
45    pub(crate) fn pop(&mut self) {
46        assert!(self.scopes.len() > 1, "Cannot pop the bottom comptime scope");
47        self.scopes.pop();
48    }
49
50    /// Hides every scope currently on the stack and pushes a scope for the callee's
51    /// parameters. Returns the previous floor, to be given back to [`Self::exit_function`].
52    pub(crate) fn enter_function(&mut self) -> usize {
53        let previous_floor = std::mem::replace(&mut self.floor, self.scopes.len());
54        self.push();
55        previous_floor
56    }
57
58    /// Drops every scope pushed since the matching [`Self::enter_function`] and makes the
59    /// caller's scopes visible again.
60    pub(crate) fn exit_function(&mut self, previous_floor: usize) {
61        self.scopes.truncate(self.floor);
62        self.floor = previous_floor;
63    }
64
65    /// Defines `id` in the innermost scope.
66    pub(crate) fn define(&mut self, id: DefinitionId, value: Value) {
67        self.scopes.last_mut().expect("There is always a bottom scope").insert(id, value);
68    }
69
70    /// The value of `id` in the innermost visible scope that defines it.
71    pub(crate) fn get(&self, id: DefinitionId) -> Option<&Value> {
72        self.scopes[self.floor..].iter().rev().find_map(|scope| scope.get(&id))
73    }
74
75    /// The value of `id` in the innermost visible scope that defines it.
76    pub(crate) fn get_mut(&mut self, id: DefinitionId) -> Option<&mut Value> {
77        self.scopes[self.floor..].iter_mut().rev().find_map(|scope| scope.get_mut(&id))
78    }
79
80    /// The variables visible to the function currently being interpreted, outermost scope
81    /// first.
82    pub(crate) fn visible_scopes(&self) -> &[HashMap<DefinitionId, Value>] {
83        &self.scopes[self.floor..]
84    }
85}
