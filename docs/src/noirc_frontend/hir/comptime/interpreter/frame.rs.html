5/// The types the function being interpreted sees: the bindings of its generics (a call's
6/// instantiation and impl bindings, or the ones a closure was created under) plus the types
7/// solved while it runs, such as a macro call's result type. Every type the interpreter reads
8/// from the HIR of that function goes through [`Frame::substitute`].
9///
10/// A value keeps the types it was built with, so a value built before one of its types was solved
11/// only resolves against a frame that holds the solution. A frame therefore starts from its
12/// caller's bindings, and on return hands back what it solved that its caller can see. See
13/// `design/comptime.md`.
14#[derive(Debug, Clone, Default, PartialEq, Eq)]
15pub(crate) struct Frame {
16    bindings: TypeBindings,
17    /// The type variables in `bindings` that were solved while interpreting, rather than bound as
18    /// generics. While there are none, every value's types are already as resolved as this frame
19    /// can make them.
20    runtime_solves: HashSet<TypeVariableId>,
21}
22
23impl Frame {
24    pub(crate) fn bindings(&self) -> &TypeBindings {
25        &self.bindings
26    }
27
28    pub(crate) fn has_runtime_solves(&self) -> bool {
29        !self.runtime_solves.is_empty()
30    }
31
32    pub(crate) fn substitute(&self, typ: &Type) -> Type {
33        typ.substitute(&self.bindings)
34    }
35
36    /// This frame (the caller's) extended into the frame of a call made from it. The callee's own
37    /// instantiation and impl bindings take precedence, which keeps each recursive call's
38    /// generics its own.
39    pub(crate) fn for_call(mut self, own_bindings: &TypeBindings) -> Frame {
40        for (var_id, binding) in own_bindings {
41            self.bindings.insert(*var_id, binding.clone());
42            self.runtime_solves.remove(var_id);
43        }
44        self
45    }
46
47    /// This frame (the caller's) extended into the frame of a call to a closure created in
48    /// `created_in`, whose bindings take precedence.
49    pub(crate) fn for_closure(mut self, created_in: Frame) -> Frame {
50        for var_id in created_in.bindings.keys() {
51            self.runtime_solves.remove(var_id);
52        }
53        self.bindings.extend(created_in.bindings);
54        self.runtime_solves.extend(created_in.runtime_solves);
55        self
56    }
57
58    /// Records types solved while interpreting.
59    pub(crate) fn solve(&mut self, bindings: TypeBindings) {
60        self.runtime_solves.extend(bindings.keys().copied());
61        self.bindings.extend(bindings);
62    }
63
64    /// Takes back solved types so that they can be solved again, possibly differently.
65    pub(crate) fn forget(&mut self, var_ids: &[TypeVariableId]) {
66        for var_id in var_ids {
67            if self.runtime_solves.remove(var_id) {
68                self.bindings.remove(var_id);
69            }
70        }
71    }
72
73    /// Copies into this frame each type variable `callee` solved that occurs in one of `visible`,
74    /// and returns them. Solves `callee` inherited from this frame and still holds unchanged are
75    /// skipped.
76    ///
77    /// A callee can only solve its own body's variables or ones that reached it through the types
78    /// its caller can see, so `visible` is the types its generics were instantiated with, or a
79    /// closure's own type (which includes its captures). A variable only the callee's body
80    /// mentions stays behind, so a recursive call that solves it differently cannot overwrite its
81    /// caller's solution.
82    pub(crate) fn take_solves(&mut self, callee: &Frame, visible: &[&Type]) -> Vec<TypeVariableId> {
83        let mut taken = Vec::new();
84        for var_id in &callee.runtime_solves {
85            let (var, kind, typ) = &callee.bindings[var_id];
86            if self.bindings.get(var_id).is_some_and(|(_, _, own)| own == typ)
87                || !visible.iter().any(|visible| visible.occurs(*var_id))
88            {
89                continue;
90            }
91            let typ = callee.substitute(typ);
92            self.bindings.insert(*var_id, (var.clone(), kind.clone(), typ));
93            self.runtime_solves.insert(*var_id);
94            taken.push(*var_id);
95        }
96        taken
97    }
98}

100#[cfg(test)]
101mod tests {
102    use crate::{Kind, Type, TypeBindings, TypeVariable, TypeVariableId};
103
104    use super::Frame;
105
106    fn binding(id: usize, typ: Type) -> TypeBindings {
107        let id = TypeVariableId(id);
108        let mut bindings = TypeBindings::default();
109        bindings.insert(id, (TypeVariable::unbound(id, Kind::Normal), Kind::Normal, typ));
110        bindings
111    }
112
113    #[test]
114    fn forgetting_every_runtime_solve_leaves_none() {
115        let mut frame = Frame::default();
116        frame.solve(binding(0, Type::Bool));
117        assert!(frame.has_runtime_solves());
118
119        frame.forget(&[TypeVariableId(0)]);
120        assert!(!frame.has_runtime_solves());
121        assert!(frame.bindings().is_empty());
122    }
123
124    #[test]
125    fn forgetting_a_generic_keeps_its_binding() {
126        let mut frame = Frame::default().for_call(&binding(0, Type::Bool));
127        frame.forget(&[TypeVariableId(0)]);
128        assert!(!frame.has_runtime_solves());
129        assert_eq!(frame.bindings().len(), 1);
130    }
131
132    #[test]
133    fn a_callees_generic_shadowing_a_solve_is_not_a_solve() {
134        let mut caller = Frame::default();
135        caller.solve(binding(0, Type::Bool));
136        let callee = caller.for_call(&binding(0, Type::FieldElement));
137        assert!(!callee.has_runtime_solves());
138    }
