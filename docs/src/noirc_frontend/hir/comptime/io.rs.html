1use std::cell::RefCell;
2use std::io::Write;
3use std::rc::Rc;
4
5use super::EvaluationTracker;
6
7/// What running comptime code may write to, besides the program being elaborated.
8///
9/// It is handed to each operation that runs comptime code, so the caller decides per operation
10/// where the output goes and whether evaluations are tracked.
11#[derive(Default)]
12pub struct ComptimeIo {
13    /// Receives whatever comptime code prints. `None` discards it.
14    pub output: Option<Rc<RefCell<dyn Write>>>,
15
16    /// Records the locations of the comptime expressions which are evaluated, to facilitate code
17    /// coverage. `None` records nothing.
18    pub evaluation_tracker: Option<EvaluationTracker>,
19}
20
21impl ComptimeIo {
22    /// Prints to stdout and tracks nothing.
23    pub fn stdout() -> Self {
24        Self::printing_to(Rc::new(RefCell::new(std::io::stdout())))
25    }
26
27    /// Discards prints and tracks nothing.
28    pub fn silent() -> Self {
29        Self::default()
30    }
31
32    /// Prints to `output` and tracks nothing.
33    pub fn printing_to(output: Rc<RefCell<dyn Write>>) -> Self {
34        Self { output: Some(output), evaluation_tracker: None }
35    }
36}
