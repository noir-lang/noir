1//! This module contains the liveness analysis which is run on each function before the
2//! ownership pass.
3//!
4//! A use of a variable can be *moved* — handed over with no reference-count bump — exactly
5//! when the variable is **dead immediately after that use**: no path onwards from that point
6//! reaches another use of the variable before the variable is reassigned. Everywhere else the
7//! use must be copied, so that a write through the new owner cannot be observed through the old
8//! name. That is backward liveness, and this module computes it directly.
9//!
10//! `visit` walks the expression tree in reverse, threading the set of variables live *after*
11//! an expression and returning those live *before* it. At each identifier the decision is one
12//! membership test. Branches join by union; a loop takes two passes so that the back edge is
13//! accounted for (see `visit_loop`), and a loop's exit and jump edges are modelled explicitly:
14//!
15//! - a `while` and a `for` are both tested at their header, so the test has two successors,
16//!   the body and the loop exit, and whatever is live after the loop is live at the test.
17//!   This is what makes a read that the body reassigns afterwards a copy rather than a move:
18//!   the exit edge reaches the post-loop reader without crossing that reassignment. A bare
19//!   `loop` has no header test and leaves only via `break`.
20//! - a `break`/`continue` does not fall through: the set live before it is the set live at its
21//!   jump target, not the set live after it in the tree. A `break`/`continue` written in a
22//!   `while` condition targets the *enclosing* loop (consistent with SSA lowering and the
23//!   comptime interpreter), so the condition is visited with this loop's targets popped.
24//!
25//! Aliasing is a separate question that liveness does not answer: a variable that has had a
26//! reference taken to it can be observed through that reference even when it is dead by name,
27//! so `referenced_variables` blocks moves of such variables outright.
28//!
29//! This pass is not sophisticated with regard to struct and tuple fields. It currently
30//! ignores these entirely and counts each use as a use of the entire variable. This is an
31//! area for future optimization. E.g. the program `a.b.c; a.e.f` will result in `a` being
32//! cloned in its entirety in the first statement. Note that this is lessened in the overall
33//! ownership pass such that only `.c` is cloned but it is still an area for improvement.
34
35use crate::ast::UnaryOp;
36use crate::monomorphization::ast::{self, Definition, IdentId, LocalId};
37use crate::monomorphization::ast::{Expression, Function, Literal};
38use rustc_hash::{FxHashMap as HashMap, FxHashSet as HashSet};
39
40/// The set of variables live at a program point.
41type Live = HashSet<LocalId>;
42
43struct LivenessContext {
44    /// Each instance of a variable that can be moved rather than cloned.
45    moves: HashMap<LocalId, Vec<IdentId>>,
46
47    /// Variables that have been aliased via a reference expression (`&var` or `&mut var`).
48    ///
49    /// A reference creates an invisible alias: if the variable were moved (sharing the same
50    /// array pointer with refcount=1), a later write through the reference would mutate the
51    /// "moved" copy in place, bypassing copy-on-write. Preventing moves of aliased variables
52    /// keeps the refcount above 1 so those writes correctly trigger COW.
53    referenced_variables: HashSet<LocalId>,
54
55    /// The set live at the exit of each enclosing loop, innermost last. A `break` jumps here.
56    break_live: Vec<Live>,
57
58    /// The set live at the header of each enclosing loop, innermost last. A `continue` jumps here.
59    continue_live: Vec<Live>,
60
61    /// Off during the first pass over a loop, whose header set is provisional, so that only
62    /// the second pass records moves. See `visit_loop`.
63    recording: bool,
64}
65
66/// Traverse the given function and return each use of a local variable that can be moved.
67/// A variable may have several such uses: one per branch of an `if`/`match`, for instance.
68pub(super) fn find_variables_to_move(function: &Function) -> HashMap<LocalId, Vec<IdentId>> {
69    let mut context = LivenessContext {
70        moves: HashMap::default(),
71        referenced_variables: HashSet::default(),
72        break_live: Vec::new(),
73        continue_live: Vec::new(),
74        recording: true,
75    };
76
77    // Nothing is live on return: the function's result has already been consumed by the
78    // tail expression itself, and parameters are dead once the body finishes.
79    context.visit(&function.body, Live::default());
80
81    let mut moves = context.moves;
82    moves.retain(|id, _| !context.referenced_variables.contains(id));
83    moves
84}

86impl LivenessContext {
87    /// Returns the set of variables live immediately *before* `expr`, given those live
88    /// immediately after it.
89    ///
90    /// Sub-expressions are visited in reverse of their forward evaluation order, so that the
91    /// set threaded into each one is the set live after it actually runs.
92    fn visit(&mut self, expr: &Expression, live: Live) -> Live {
93        match expr {
94            Expression::Ident(ident) => self.visit_ident(ident, live),
95            Expression::Literal(literal) => self.visit_literal(literal, live),
96            Expression::Block(exprs) => {
97                exprs.iter().rev().fold(live, |live, expr| self.visit(expr, live))
98            }
99            Expression::Unary(unary) => self.visit_unary(unary, live),
100            Expression::Binary(binary) => {
101                let live = self.visit(&binary.rhs, live);
102                self.visit(&binary.lhs, live)
103            }
104            Expression::Index(index) => {
105                // SSA codegen evaluates the index before the collection, so in forward order
106                // the collection is used *after* the index.
107                let live = self.visit(&index.collection, live);
108                self.visit(&index.index, live)
109            }
110            Expression::Cast(cast) => self.visit(&cast.lhs, live),
111            Expression::For(for_expr) => self.visit_for(for_expr, live),
112            // A bare `loop` has no header test: it leaves only via `break`, which reads
113            // `break_live` directly.
114            Expression::Loop(body) => self.visit_loop(body, None, false, live),
115            Expression::While(while_expr) => {
116                self.visit_loop(&while_expr.body, Some(&while_expr.condition), true, live)
117            }
118            Expression::If(if_expr) => self.visit_if(if_expr, live),
119            Expression::Match(match_expr) => self.visit_match(match_expr, live),
120            Expression::Tuple(elements) => {
121                elements.iter().rev().fold(live, |live, elem| self.visit(elem, live))
122            }
123            Expression::ExtractTupleField(tuple, _) => self.visit(tuple, live),
124            Expression::Call(call) => self.visit_call(call, live),
125            Expression::Let(let_expr) => {
126                // The binding is a definition: the variable is not live before it.
127                let mut live = live;
128                live.remove(&let_expr.id);
129                self.visit(&let_expr.expression, live)
130            }
131            Expression::Constrain(boolean, _, msg) => {
132                let live = match msg {
133                    Some(msg) => self.visit(&msg.0, live),
134                    None => live,
135                };
136                self.visit(boolean, live)
137            }
138            Expression::Assign(assign) => self.visit_assign(assign, live),
139            Expression::Semi(expr) => self.visit(expr, live),
140            Expression::Clone(_) => unreachable!("liveness is called before clones are inserted"),
141            Expression::Drop(_) => unreachable!("liveness is called before drops are inserted"),
142            // Control does not fall through a jump, so the incoming set is discarded in
143            // favour of the set live at the jump's target.
144            Expression::Break => self.break_live.last().cloned().unwrap_or_default(),
145            Expression::Continue => self.continue_live.last().cloned().unwrap_or_default(),
146        }
147    }
148
149    fn visit_ident(&mut self, ident: &ast::Ident, mut live: Live) -> Live {
150        if let Definition::Local(local_id) = &ident.definition {
151            if self.recording && !live.contains(local_id) {
152                self.moves.entry(*local_id).or_default().push(ident.id);
153            }
154            live.insert(*local_id);
155        }
156        live
157    }
158
159    fn visit_literal(&mut self, literal: &Literal, live: Live) -> Live {
160        match literal {
161            Literal::Integer(..) | Literal::Bool(_) | Literal::Unit | Literal::Str(_) => live,
162            Literal::FmtStr(_, _, captures) => self.visit(captures, live),
163            Literal::Array(array) | Literal::Vector(array) => {
164                array.contents.iter().rev().fold(live, |live, element| self.visit(element, live))
165            }
166            Literal::Repeated { element, .. } => self.visit(element, live),
167        }
168    }
169
170    fn visit_unary(&mut self, unary: &ast::Unary, live: Live) -> Live {
171        if matches!(unary.operator, UnaryOp::Reference { .. })
172            && let Some(local_id) = base_ident_of_field_access(&unary.rhs)
173        {
174            // Taking a reference to a local variable or one of its fields (e.g. `&mut x` or
175            // `&mut x.field`) aliases `x`; no copy of `x` may be a move from here on.
176            self.referenced_variables.insert(local_id);
177        }
178        self.visit(&unary.rhs, live)
179    }
180
181    fn visit_if(&mut self, if_expr: &ast::If, live: Live) -> Live {
182        // A variable is live before the branch if it is live entering either arm.
183        let mut merged = self.visit(&if_expr.consequence, live.clone());
184        match &if_expr.alternative {
185            Some(alt) => merged.extend(self.visit(alt, live)),
186            // The implicit "do nothing" path falls straight through.
187            None => merged.extend(live),
188        }
189        self.visit(&if_expr.condition, merged)
190    }
191
192    fn visit_match(&mut self, match_expr: &ast::Match, live: Live) -> Live {
193        // Note: `variable_to_match` is not a use here. It is a `LocalId` referring to a
194        // variable bound earlier, and the analysis for it happens at its actual use sites.
195        let mut merged = Live::default();
196        for case in &match_expr.cases {
197            let mut case_live = self.visit(&case.branch, live.clone());
198            // The case's bindings are defined by the match, so they are not live before it.
199            for (argument, _) in &case.arguments {
200                case_live.remove(argument);
201            }
202            merged.extend(case_live);
203        }
204        match &match_expr.default_case {
205            Some(default_case) => merged.extend(self.visit(default_case, live)),
206            // With no default case some path skips every arm.
207            None => merged.extend(live),
208        }
209        merged
210    }
211
212    fn visit_call(&mut self, call: &ast::Call, live: Live) -> Live {
213        // A reference passed directly as a call argument (e.g. `foo(&mut x)`) is temporary:
214        // it only lives for the duration of the call. After the call returns, `x` is no longer
215        // aliased, so future copies of `x` don't need to clone.
216        //
217        // We must fall back to conservative (mark `x` as aliased) if the reference could escape:
218        // 1. The call returns a reference type — the passed reference might be returned.
219        // 2. Another argument has type `&mut T` where `T` contains a reference — the function
220        //    could write the passed reference into `*that_arg`, making it escape without returning.
221        let conservative = type_contains_reference(&call.return_type)
222            || call.arguments.iter().any(arg_can_store_reference);
223
224        let mut live = live;
225        for (index, arg) in call.arguments.iter().enumerate().rev() {
226            if !conservative
227                && let Expression::Unary(unary) = arg
228                && matches!(unary.operator, UnaryOp::Reference { .. })
229                && let Some(base) = base_ident_of_field_access(&unary.rhs)
230                // Even though the reference cannot escape the call, the *other* arguments of
231                // this same call are evaluated and handed to the callee while the reference is
232                // live. If one of them mentions `x` (e.g. `foo(&mut x, x)`), moving that use
233                // would let the callee's writes through the reference be observed through a
234                // by-value argument, so `x` must be treated as aliased.
235                && !call.arguments.iter().enumerate().any(|(other_index, other_arg)| {
236                    other_index != index && local_occurs_in(base, other_arg)
237                })
238            {
239                // Count the use of the variable inside the reference, but skip the unary
240                // handler, which would mark the variable as aliased.
241                live = self.visit(&unary.rhs, live);
242            } else {
243                live = self.visit(arg, live);
244            }
245        }
246        self.visit(&call.func, live)
247    }
248
249    fn visit_assign(&mut self, assign: &ast::Assign, live: Live) -> Live {
250        if let ast::LValue::Ident(ast::Ident { definition: Definition::Local(local_id), .. }) =
251            &assign.lvalue
252        {
253            // Assigning the whole variable ends its old value, so a use of it inside the
254            // right-hand side can be the last one.
255            let mut live = live;
256            live.remove(local_id);
257            return self.visit(&assign.expression, live);
258        }
259
260        // A compound lvalue (e.g. `a[i] = expr`) reads `a` as well as writing it, so it is
261        // not a definition. SSA codegen accesses the lvalue after evaluating the RHS.
262        let live = self.visit_lvalue(&assign.lvalue, false, live);
263        self.visit(&assign.expression, live)
264    }
265
266    /// A variable in an lvalue position is never moved (otherwise you wouldn't be able to
267    /// access the variable you assigned to afterward). It is still a use, which keeps earlier
268    /// reads from being moved. However, the index in an array expression `a[i] = ...` is an
269    /// arbitrary expression that is actually in an rvalue position and can thus be moved.
270    ///
271    /// The `nested` parameter indicates whether this lvalue is nested inside another lvalue.
272    /// A top-level identifier is overwritten rather than read, so it is not a use; an
273    /// identifier reached through an index or field (`ident[index] = ...`) is read.
274    fn visit_lvalue(&mut self, lvalue: &ast::LValue, nested: bool, live: Live) -> Live {
275        match lvalue {
276            ast::LValue::Ident(ident) => match (nested, &ident.definition) {
277                (true, Definition::Local(local_id)) => {
278                    let mut live = live;
279                    live.insert(*local_id);
280                    live
281                }
282                _ => live,
283            },
284            ast::LValue::Index { array, index, .. } => {
285                // As in the rvalue Index case, SSA codegen evaluates the index before
286                // touching the array in an lvalue position.
287                let live = self.visit_lvalue(array, true, live);
288                self.visit(index, live)
289            }
290            ast::LValue::MemberAccess { object, .. } => self.visit_lvalue(object, true, live),
291            ast::LValue::Dereference { reference, .. } => self.visit_lvalue(reference, true, live),
292            ast::LValue::Clone(_) => {
293                unreachable!("LValue::Clone should only be inserted by the ownership pass")
294            }
295        }
296    }
297
298    fn visit_for(&mut self, for_expr: &ast::For, live: Live) -> Live {
299        // The ranges are evaluated once, before the loop, so they sit outside it.
300        // `for` carries no condition *expression*, but `index < end_range` is still tested at
301        // the header, so the loop exits from there just as a `while` does.
302        let mut live = self.visit_loop(&for_expr.block, None, true, live);
303        // The loop header defines the index variable on every iteration.
304        live.remove(&for_expr.index_variable);
305        let live = self.visit(&for_expr.end_range, live);
306        self.visit(&for_expr.start_range, live)
307    }
308
309    /// Returns the set live at the loop's header, given the set live after the loop.
310    ///
311    /// The back edge and every `continue` make the header its own successor, so its live set
312    /// `H` is a fixpoint. Two passes find it without iterating. Every transfer function in this
313    /// analysis has the form `X ↦ A ∪ (X ∩ T)`: a use adds to `A`, a definition removes from `T`,
314    /// a jump replaces `X` with its target's set, and unions, sequencing and nested loops all
315    /// preserve the form. So `H = A ∪ (H ∩ T)`, whose least solution is `A` itself, which is
316    /// what one pass computes with the header taken to be empty. Put in terms of paths: a path
317    /// that goes round the back edge before reaching a use adds nothing, because the part after
318    /// its last visit to the header already reaches that use within a single iteration.
319    ///
320    /// The first pass runs with recording off, because its header set is not yet the true one.
321    /// The second pass, with `H` in hand, records the moves.
322    fn visit_loop(
323        &mut self,
324        body: &Expression,
325        condition: Option<&Expression>,
326        exits_from_header: bool,
327        live_after: Live,
328    ) -> Live {
329        let recording = std::mem::replace(&mut self.recording, false);
330        let header =
331            self.visit_loop_once(body, condition, exits_from_header, &Live::default(), &live_after);
332        self.recording = recording;
333        self.visit_loop_once(body, condition, exits_from_header, &header, &live_after)
334    }
335
336    fn visit_loop_once(
337        &mut self,
338        body: &Expression,
339        condition: Option<&Expression>,
340        exits_from_header: bool,
341        header: &Live,
342        live_after: &Live,
343    ) -> Live {
344        self.break_live.push(live_after.clone());
345        self.continue_live.push(header.clone());
346
347        // The body's successor is the back edge to the header.
348        let body_in = self.visit(body, header.clone());
349
350        self.continue_live.pop();
351        self.break_live.pop();
352
353        match condition {
354            // The condition's successors are the body and the loop exit. A `break`/`continue`
355            // written in the condition targets the *enclosing* loop, so the condition is
356            // visited with this loop's jump targets already popped.
357            Some(condition) => {
358                let condition_out = body_in.union(live_after).copied().collect::<Live>();
359                self.visit(condition, condition_out)
360            }
361            // A `for` has no condition expression, but its header test still exits the loop.
362            None if exits_from_header => body_in.union(live_after).copied().collect(),
363            None => body_in,
364        }
365    }
366}

368/// Returns `true` if `local_id` occurs anywhere within `expr`, in value position (`x`, `x[i]`)
369/// or as the base of an assignment lvalue (`x[i] = ..`). Monomorphized `LocalId`s are unique,
370/// so any occurrence is a free occurrence.
371fn local_occurs_in(local_id: LocalId, expr: &Expression) -> bool {
372    let mut occurs = false;
373    crate::monomorphization::visitor::visit_expr_be(
374        expr,
375        &mut |_| (true, ()),
376        &mut |_, ()| {},
377        &mut |ident: &ast::Ident| {
378            if let Definition::Local(id) = ident.definition
379                && id == local_id
380            {
381                occurs = true;
382            }
383        },
384    );
385    occurs
386}

388/// Given an expression that is the operand of a reference (`&expr` or `&mut expr`),
389/// walk through any chain of struct-field accesses (`expr.field` = `ExtractTupleField`)
390/// and return the `LocalId` of the base variable, if it is a local variable.
391///
392/// For example:
393/// - `&mut x`         → `Some(x_id)`
394/// - `&mut x.field`   → `Some(x_id)`  (field is `ExtractTupleField(x, _)`)
395/// - `&mut x.a.b`     → `Some(x_id)`
396/// - `&mut some_call()` → `None`
397///
398/// Block expressions intentionally return `None`: `&mut { ...; expr }` always allocates
399/// fresh storage and copies the tail's value into it, so the operand is no longer a
400/// "direct reference" to a local. The clones needed to keep refcounts honest are
401/// inserted by the forward pass when it processes the block's tail in normal context
402/// (see `handle_reference_expression`).
403fn base_ident_of_field_access(expr: &Expression) -> Option<LocalId> {
404    match expr {
405        Expression::Ident(ident) => {
406            if let Definition::Local(local_id) = ident.definition {
407                Some(local_id)
408            } else {
409                None
410            }
411        }
412        Expression::ExtractTupleField(inner, _) => base_ident_of_field_access(inner),
413        _ => None,
414    }
415}

417/// Returns `true` if `arg`'s type can be used to store a reference, i.e. the type
418/// contains a `&mut T` (at any depth) where `T` itself contains a reference.
419///
420/// When this is true for any argument, all `&mut x` arguments in the call must conservatively
421/// be treated as aliasing `x`, because the callee might write the reference into the location
422/// reachable through that argument.
423fn arg_can_store_reference(arg: &Expression) -> bool {
424    match arg.return_type() {
425        Some(typ) => type_can_store_reference(&typ),
426        None => true,
427    }
428}
429
430/// Returns `true` if `typ` contains — at any depth — a `&mut T` where `T` itself contains
431/// a reference. Such a type allows a reference to be written somewhere persistent.
432fn type_can_store_reference(typ: &ast::Type) -> bool {
433    use ast::Type;
434    match typ {
435        Type::Reference(inner, true /* mutable */) => type_contains_reference(inner),
436        Type::Reference(inner, false) => type_can_store_reference(inner),
437        Type::Tuple(elements) => elements.iter().any(type_can_store_reference),
438        Type::Array(_, elem) | Type::Vector(elem) | Type::FmtString(_, elem) => {
439            type_can_store_reference(elem)
440        }
441        Type::Function(args, ret, env, _) => {
442            args.iter().any(type_can_store_reference)
443                || type_can_store_reference(ret)
444                || type_can_store_reference(env)
445        }
446        Type::Field | Type::Integer(..) | Type::Bool | Type::String(..) | Type::Unit => false,
447    }
448}

450/// Returns `true` if the type contains a `Reference` anywhere (directly or nested within
451/// tuples, arrays, or function types). Used to decide whether a call might return a
452/// reference that aliases a variable passed by `&mut` to that call.
453fn type_contains_reference(typ: &ast::Type) -> bool {
454    use ast::Type;
455    match typ {
456        Type::Reference(..) => true,
457        Type::Tuple(elements) => elements.iter().any(type_contains_reference),
458        Type::Array(_, elem) | Type::Vector(elem) | Type::FmtString(_, elem) => {
459            type_contains_reference(elem)
460        }
461        Type::Function(args, ret, env, _) => {
462            args.iter().any(type_contains_reference)
463                || type_contains_reference(ret)
464                || type_contains_reference(env)
465        }
466        Type::Field | Type::Integer(..) | Type::Bool | Type::String(..) | Type::Unit => false,
467    }
