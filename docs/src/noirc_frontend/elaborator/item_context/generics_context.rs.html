1//! The part of an item's context holding the generics and trait bounds that are in scope.
2
3use std::collections::BTreeSet;
4
5use iter_extended::vecmap;
6
7use crate::{
8    Type, TypeVariable,
9    hir_def::{traits::TraitConstraint, types::ResolvedGeneric},
10    node_interner::TraitId,
11};
12
13/// The generic parameters an item introduces, together with the trait bounds they carry.
14///
15/// The bounds live here rather than in a context of their own because a bound is what gives a
16/// generic parameter its members: resolving `T::AssociatedType` needs both the parameter and the
17/// `where` clause naming the trait that declares the associated type.
18///
19/// Generics nest - a method's generics are added on top of its impl's - so this is entered and
20/// left as a stack via [`Self::enter_scope`] and [`Self::exit_scope`] rather than replaced.
21#[derive(Default)]
22pub(crate) struct GenericsContext {
23    /// Contains a mapping of the current struct or functions's generics to
24    /// unique type variables if we're resolving a struct. Empty otherwise.
25    /// This is a Vec rather than a map to preserve the order a functions generics
26    /// were declared in.
27    params: Vec<ResolvedGeneric>,
28
29    /// Each constraint in the `where` clause of the function currently being resolved.
30    trait_bounds: Vec<TraitConstraint>,
31
32    /// Every `(object type, trait)` pair brought into scope by implication rather than by a
33    /// `where` clause naming it: a bound declared on an associated type, or a parent trait.
34    ///
35    /// A written bound which duplicates one of these is not reported as unnecessary. The two can
36    /// be registered in either order - the implication may come from a later clause in the same
37    /// `where` list - so remembering the implied pairs is what makes the diagnostic independent of
38    /// that order. Emptied together with the assumed impls by
39    /// [`Elaborator::remove_trait_constraints_from_scope`](crate::elaborator::Elaborator::remove_trait_constraints_from_scope).
40    implied_trait_bounds: BTreeSet<(Type, TraitId)>,
41}
42
43/// Saved generics state for restoration after a scope exits.
44pub(crate) struct GenericsState {
45    params_count: usize,
46}
47
48impl GenericsContext {
49    /// A context with `params` in scope, holding the caller's `trait_bounds`.
50    pub(crate) fn new(params: Vec<ResolvedGeneric>, trait_bounds: Vec<TraitConstraint>) -> Self {
51        Self { params, trait_bounds, implied_trait_bounds: BTreeSet::new() }
52    }
53
54    /// Saves the current generics state to be restored later with [`Self::exit_scope`].
55    /// Note that all of the generics in scope will still be in scope after this call. This will
56    /// only save the position of the current generics so that any generics added afterward can
57    /// later be discarded via a call to [`Self::exit_scope`].
58    pub(crate) fn enter_scope(&self) -> GenericsState {
59        GenericsState { params_count: self.params.len() }
60    }
61
62    /// Restores the generics state saved by [`Self::enter_scope`].
63    #[tracing::instrument(level = "trace", skip_all)]
64    pub(crate) fn exit_scope(&mut self, state: GenericsState) {
65        self.params.truncate(state.params_count);
66    }
67
68    /// The generics in scope, in the order they were declared in.
69    pub(crate) fn params(&self) -> &[ResolvedGeneric] {
70        &self.params
71    }
72
73    /// Brings one more generic into scope, after the ones already there.
74    pub(crate) fn add_param(&mut self, param: ResolvedGeneric) {
75        self.params.push(param);
76    }
77
78    /// Brings several more generics into scope, after the ones already there.
79    pub(crate) fn add_params(&mut self, params: impl IntoIterator<Item = ResolvedGeneric>) {
80        self.params.extend(params);
81    }
82
83    /// Replaces the generics in scope wholesale, for an item whose generics are known up front
84    /// rather than accumulated as its declaration is resolved.
85    pub(crate) fn set_params(&mut self, params: Vec<ResolvedGeneric>) {
86        self.params = params;
87    }
88
89    /// Takes the generics out of scope and hands them to the caller.
90    pub(crate) fn take_params(&mut self) -> Vec<ResolvedGeneric> {
91        std::mem::take(&mut self.params)
92    }
93
94    /// Takes every generic out of scope.
95    pub(crate) fn clear_params(&mut self) {
96        self.params.clear();
97    }
98
99    /// Find a generic variable among the generics of the current struct or function.
100    pub(crate) fn find(&self, target_name: &str) -> Option<&ResolvedGeneric> {
101        self.params.iter().find(|generic| generic.name.as_ref() == target_name)
102    }
103
104    /// The type variables of the generics in scope, in declaration order.
105    pub(crate) fn type_vars(&self) -> Vec<TypeVariable> {
106        vecmap(&self.params, |generic| generic.type_var.clone())
107    }
108
109    /// Whether any bound at all is in scope, written or implied.
110    pub(crate) fn has_bounds(&self) -> bool {
111        !self.trait_bounds.is_empty()
112    }
113
114    /// Assumes `constraint` for the rest of the item.
115    pub(crate) fn add_bound(&mut self, constraint: TraitConstraint) {
116        self.trait_bounds.push(constraint);
117    }
118
119    /// Whether `object_type: trait` is one of the bounds in scope.
120    pub(crate) fn has_bound(&self, object_type: &Type, trait_id: TraitId) -> bool {
121        self.trait_bounds
122            .iter()
123            .any(|bound| bound.trait_bound.trait_id == trait_id && bound.typ == *object_type)
124    }
125
126    /// The bounds written on the generic parameter named `name`, e.g. both `Foo` and `Bar` for
127    /// `T` given `where T: Foo, T: Bar`.
128    pub(crate) fn bounds_on_generic<'a>(
129        &'a self,
130        name: &'a str,
131    ) -> impl Iterator<Item = &'a TraitConstraint> {
132        self.trait_bounds.iter().filter(move |constraint| {
133            matches!(&constraint.typ, Type::NamedGeneric(generic) if generic.name.as_str() == name)
134        })
135    }
136
137    /// Drops every bound on `trait_id`, for when the constraints that brought them into scope
138    /// leave it.
139    pub(crate) fn remove_bounds_on_trait(&mut self, trait_id: TraitId) {
140        self.trait_bounds.retain(|bound| bound.trait_bound.trait_id != trait_id);
141    }
142
143    /// Records `object_type: trait` as a bound that was implied rather than written, so that a
144    /// `where` clause repeating it is not reported as unnecessary.
145    pub(crate) fn record_implied_bound(&mut self, object_type: &Type, trait_id: TraitId) {
146        self.implied_trait_bounds.insert((object_type.clone(), trait_id));
147    }
148
149    /// Whether `object_type: trait` was brought into scope by implication.
150    pub(crate) fn is_implied_bound(&self, object_type: &Type, trait_id: TraitId) -> bool {
151        self.implied_trait_bounds.contains(&(object_type.clone(), trait_id))
152    }
153
154    /// Forgets the implied bounds, for when the bounds they were implied by leave scope.
155    pub(crate) fn clear_implied_bounds(&mut self) {
156        self.implied_trait_bounds.clear();
157    }
158}
