1//! The part of an item's context that says which item is being elaborated, and where it lives.
2
3use crate::{
4    hir::def_map::{LocalModuleId, ModuleId},
5    node_interner::DependencyId,
6};
7
8/// Which item the elaborator is working on and which module it was written in.
9///
10/// The module is what unqualified paths in the item resolve against, so it moves with the item:
11/// an item elaborated on demand from the middle of another one is resolved in its own module, not
12/// in the module of whatever mentioned it.
13pub(crate) struct ModuleContext {
14    /// The module the item was written in, which every item that resolves a path has.
15    local_module: LocalModuleId,
16
17    /// The current dependency item we're resolving.
18    /// Used to link items to their dependencies in the dependency graph
19    current_item: Option<DependencyId>,
20
21    /// When set, visibility checks during path resolution use this module
22    /// instead of the default importing module.
23    ///
24    /// Set when resolving an expression on behalf of comptime code from another module (see
25    /// `Expr::resolve`), so that the item resolved that way is held to the caller's visibility
26    /// rather than to that of the scope it is resolved in.
27    caller_module: Option<ModuleId>,
28}
29
30impl ModuleContext {
31    /// Elaborating `current_item`, which was written in `local_module`.
32    pub(crate) fn of_item(local_module: LocalModuleId, current_item: DependencyId) -> Self {
33        Self { local_module, current_item: Some(current_item), caller_module: None }
34    }
35
36    /// Elaborating in `local_module` without naming an item, for work that belongs to no single
37    /// item and so should register no dependencies, such as resolving an impl's header.
38    pub(crate) fn in_module(local_module: LocalModuleId) -> Self {
39        Self { local_module, current_item: None, caller_module: None }
40    }
41
42    /// Elaborating `current_item`, which is written inside the item this context describes: a
43    /// trait's method, say. The module is the enclosing item's.
44    pub(crate) fn nested_item(&self, current_item: DependencyId) -> Self {
45        Self {
46            local_module: self.local_module,
47            current_item: Some(current_item),
48            caller_module: None,
49        }
50    }
51
52    /// The module the item was written in.
53    pub(crate) fn local_module(&self) -> LocalModuleId {
54        self.local_module
55    }
56
57    /// Whether the item being elaborated was written in `module`.
58    pub(crate) fn is_in_module(&self, module: LocalModuleId) -> bool {
59        self.local_module == module
60    }
61
62    /// Resolves the rest of the item in `module`, returning the module it replaces so the caller
63    /// can put it back with [`Self::set_local_module`].
64    #[must_use]
65    pub(crate) fn replace_local_module(&mut self, module: LocalModuleId) -> LocalModuleId {
66        std::mem::replace(&mut self.local_module, module)
67    }
68
69    pub(crate) fn set_local_module(&mut self, module: LocalModuleId) {
70        self.local_module = module;
71    }
72
73    pub(crate) fn current_item(&self) -> Option<DependencyId> {
74        self.current_item
75    }
76
77    pub(crate) fn set_current_item(&mut self, current_item: Option<DependencyId>) {
78        self.current_item = current_item;
79    }
80
81    pub(crate) fn caller_module(&self) -> Option<ModuleId> {
82        self.caller_module
83    }
84
85    pub(crate) fn set_caller_module(&mut self, caller_module: Option<ModuleId>) {
86        self.caller_module = caller_module;
87    }
88}
