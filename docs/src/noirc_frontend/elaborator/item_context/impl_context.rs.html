1//! The part of an item's context that says which impl or trait the item belongs to.
2
3use crate::{
4    Type,
5    hir_def::function::FuncMeta,
6    node_interner::{ImplId, TraitId, TraitImplId, TypeId},
7};
8
9/// Where the item being elaborated sits with respect to impls and traits: what `Self` names, and
10/// which trait, trait impl or inherent impl block encloses it.
11///
12/// These fields describe one fact between them - `Self` means nothing without knowing whether it
13/// came from a trait, a trait impl or an inherent impl - so the constructors below fill them in as
14/// a unit and nothing else can change them afterwards, save for the `Self` type of a comptime
15/// function set ([`Self::set_self_type`]). A scope inside no impl or trait leaves this
16/// [`Default`], which is what [`Self::is_outside_any_impl_or_trait`] reports.
17#[derive(Default, Clone)]
18pub(crate) struct ImplContext {
19    /// Set to the current type if we're resolving an impl
20    self_type: Option<Type>,
21
22    /// The trait we're currently resolving or implementing, if any.
23    /// Set during both trait definitions (`trait Foo { ... }`) and
24    /// trait impl elaboration (`impl Foo for Bar { ... }`).
25    current_trait: Option<TraitId>,
26
27    /// If we're currently resolving methods within a trait impl, this will be set
28    /// to the corresponding trait impl ID.
29    current_trait_impl: Option<TraitImplId>,
30
31    /// If we're currently resolving methods within an inherent (non-trait) impl,
32    /// this will be set to the corresponding impl ID.
33    current_impl: Option<ImplId>,
34}
35
36impl ImplContext {
37    /// Inside `trait Foo { ... }`, where `Self` is the trait's own type variable.
38    pub(crate) fn in_trait(current_trait: TraitId, self_type: Type) -> Self {
39        Self {
40            self_type: Some(self_type),
41            current_trait: Some(current_trait),
42            current_trait_impl: None,
43            current_impl: None,
44        }
45    }
46
47    /// Inside `impl Foo for Bar { ... }`.
48    ///
49    /// Each argument is optional because the impl is entered before it is fully resolved: the
50    /// trait path may not have named a trait, and the self type is resolved after the header.
51    pub(crate) fn in_trait_impl(
52        self_type: Option<Type>,
53        current_trait: Option<TraitId>,
54        current_trait_impl: Option<TraitImplId>,
55    ) -> Self {
56        Self { self_type, current_trait, current_trait_impl, current_impl: None }
57    }
58
59    /// Inside an inherent `impl Bar { ... }`.
60    pub(crate) fn in_inherent_impl(current_impl: ImplId, self_type: Type) -> Self {
61        Self {
62            self_type: Some(self_type),
63            current_trait: None,
64            current_trait_impl: None,
65            current_impl: Some(current_impl),
66        }
67    }
68
69    /// The impl or trait a function was declared in, as recorded on its [`FuncMeta`].
70    pub(crate) fn of_function(meta: &FuncMeta) -> Self {
71        Self {
72            self_type: meta.self_type.clone(),
73            current_trait: meta.trait_id,
74            current_trait_impl: meta.trait_impl,
75            current_impl: meta.impl_id,
76        }
77    }
78
79    pub(crate) fn self_type(&self) -> Option<&Type> {
80        self.self_type.as_ref()
81    }
82
83    /// The type `Self` refers to. Only call this where an impl, trait or trait impl context is
84    /// installed; items that have no `Self` leave this unset.
85    pub(crate) fn expect_self_type(&self) -> &Type {
86        self.self_type.as_ref().expect("self_type is unset")
87    }
88
89    pub(crate) fn current_trait(&self) -> Option<TraitId> {
90        self.current_trait
91    }
92
93    pub(crate) fn current_trait_impl(&self) -> Option<TraitImplId> {
94        self.current_trait_impl
95    }
96
97    pub(crate) fn current_impl(&self) -> Option<ImplId> {
98        self.current_impl
99    }
100
101    /// Takes the `Self` type out of scope and hands it to the caller, to be put back with
102    /// [`Self::set_self_type`].
103    ///
104    /// The `Self` type is the one part of this context that a scope narrower than an item can
105    /// change: a comptime function set is interpreted with its own `Self` in an enclosing item's
106    /// context.
107    #[must_use]
108    pub(crate) fn take_self_type(&mut self) -> Option<Type> {
109        self.self_type.take()
110    }
111
112    /// Installs `self_type` as what `Self` names for the rest of the item.
113    pub(crate) fn set_self_type(&mut self, self_type: Option<Type>) {
114        self.self_type = self_type;
115    }
116
117    /// True when no impl or trait encloses the item being elaborated.
118    pub(crate) fn is_outside_any_impl_or_trait(&self) -> bool {
119        // Destructured rather than accessed field by field so that a field added to this struct
120        // has to be considered here.
121        let Self { self_type, current_trait, current_trait_impl, current_impl } = self;
122        self_type.is_none()
123            && current_trait.is_none()
124            && current_trait_impl.is_none()
125            && current_impl.is_none()
126    }
127
128    /// The struct or enum `Self` names, when `Self` is one.
129    ///
130    /// `None` when there is no `Self` in scope and when `Self` is something other than a data
131    /// type, such as the trait's own type variable inside a trait definition.
132    pub(crate) fn self_data_type_id(&self) -> Option<TypeId> {
133        match self.self_type.as_ref()? {
134            Type::DataType(data_type, _) => Some(data_type.borrow().id),
135            _ => None,
136        }
137    }
138
139    /// The type `Self` refers to, when the item is inside a trait or a trait impl.
140    ///
141    /// `None` for an inherent impl: a constraint is only resolved against `Self` when `Self` is
142    /// the trait's own type variable or the type a trait impl is written for.
143    pub(crate) fn trait_self_type(&self) -> Option<Type> {
144        self.current_trait?;
145        self.self_type.clone()
146    }
147}
