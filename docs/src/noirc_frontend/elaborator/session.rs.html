1//! The elaborator state shared by every [`Elaborator`] taking part in one elaboration.
2//!
3//! Comptime code can ask for more elaboration in the middle of being interpreted: unquoting a
4//! type or module, expanding a macro call, or lazily elaborating the body of a function it is
5//! about to call. Each of these runs in a fresh [`Elaborator`] so that the item being elaborated
6//! by the parent is not disturbed (see [`Elaborator::elaborate_item_from_comptime`]).
7//!
8//! A fresh elaborator is only fresh with respect to its item. The limits, flags and pending work
9//! of the surrounding comptime evaluation must carry through it, or a nested elaboration could
10//! reset a recursion limit, clear a halt, or lose deferred items. That state lives in
11//! [`ElaborationSession`], which the fresh elaborator takes over for its lifetime and hands back
12//! when it is done. Because the struct moves as a unit, a field added here is shared by
13//! construction.
14//!
15//! Per-elaborator state that is deliberately *not* shared - the scope forest, the function
16//! context stack, the in-resolution type ids and the collected errors - stays on the
17//! [`Elaborator`] itself, and the item being elaborated lives in its
18//! [`ItemContext`](super::item_context::ItemContext).
19
20use noirc_errors::Location;
21use rustc_hash::FxHashSet as HashSet;
22
23use super::{ElaborateReason, Elaborator, deferred::DeferredItems};
24
25/// State that spans every [`Elaborator`] taking part in one elaboration, including the fresh
26/// elaborators created to elaborate items on behalf of comptime code.
27#[derive(Default)]
28pub(crate) struct ElaborationSession {
29    /// Locations of the comptime calls currently being interpreted, outermost first.
30    pub(super) interpreter_call_stack: imbl::Vector<Location>,
31
32    /// Sometimes items are elaborated because a function attribute ran and generated items.
33    /// The Elaborator keeps track of these reasons so that when an error is produced it will
34    /// be wrapped in another error that will include this reason.
35    pub(super) elaborate_reasons: imbl::Vector<ElaborateReason>,
36
37    /// Set to true when the interpreter encounters an errored expression/statement,
38    /// causing all subsequent comptime evaluation to be skipped.
39    pub(super) comptime_evaluation_halted: bool,
40
41    /// Tracks the current macro expansion depth to prevent infinite recursion
42    /// when an attribute generates code that triggers further attribute expansion.
43    /// This catches both single-function and mutual recursion, including recursion
44    /// that passes through a nested elaborator.
45    pub(super) macro_expansion_depth: usize,
46
47    /// Current expression nesting depth, counted across nested elaborators since they
48    /// all share the same native stack.
49    pub(super) recursion_depth: usize,
50
51    /// Variable names from enclosing runtime scopes, used for error reporting only.
52    /// While a nested elaborator runs, this holds the names of the variables visible in every
53    /// elaborator that encloses it. If a variable lookup fails and the name is in this set, we
54    /// can report a more specific error about runtime variables not being available in comptime
55    /// code.
56    pub(super) parent_runtime_variables: HashSet<String>,
57
58    /// Items registered for resolution later, and the trait bookkeeping that waits on them.
59    /// See the [`deferred`](super::deferred) module for why each kind is deferred and what the
60    /// drains guarantee.
61    pub(super) deferred: DeferredItems,
62}
63
64impl Elaborator<'_> {
65    /// Whether comptime evaluation has been halted by an earlier error.
66    pub(crate) fn comptime_evaluation_halted(&self) -> bool {
67        self.session.comptime_evaluation_halted
68    }
69
70    /// Skip all further comptime evaluation in this elaboration.
71    pub(crate) fn halt_comptime_evaluation(&mut self) {
72        self.session.comptime_evaluation_halted = true;
73    }
74}
