1//! The elaborator state that belongs to the item currently being elaborated, and the swap that
2//! keeps one item's elaboration from observing or changing another's.
3//!
4//! Elaborating an item can require elaborating a different item first. For example, a reference to
5//! a function returning `impl Trait` needs that function's body to learn the concrete type, so the
6//! callee is elaborated in the middle of the caller's body. The callee must start from its own
7//! context, and the caller must get its context back unchanged: otherwise the caller goes on
8//! resolving `Self`, generics, trait bounds, loops and `unsafe` blocks as if it were the callee.
9//!
10//! [`Elaborator::with_item_context`] installs a whole [`ItemContext`] for the duration of a closure
11//! and reinstates the previous one afterwards. Since the state lives in one struct that is swapped
12//! as a unit, a field added here is saved and restored by construction.
13//!
14//! Within that struct, fields which are only meaningful together live in a sub-context of their
15//! own - [`ModuleContext`] for which item is being elaborated and where it lives, [`ImplContext`]
16//! for the enclosing impl or trait, [`GenericsContext`] for the generics and bounds in scope,
17//! [`BodyContext`] for where in the item's body the elaborator is. Each sub-context owns the
18//! operations over its own fields, so a caller that needs one group does not get a handle on the
19//! rest.
20

33/// The elaborator state describing one item's elaboration.
34///
35/// This is the state that belongs to an item rather than to the crate being elaborated: anything
36/// here is expected to start fresh for an item and to be irrelevant once that item is done. State
37/// shared by the whole elaboration - the interner, the collected errors, the recursion depth - stays
38/// on the [`Elaborator`] itself.
39pub(super) struct ItemContext {
40    /// The item being elaborated, and the module it was written in.
41    pub(super) module: ModuleContext,
42
43    /// The impl or trait the item belongs to, if any.
44    pub(super) impl_context: ImplContext,
45
46    /// The generics in scope, and the trait bounds they carry.
47    pub(super) generics: GenericsContext,
48
49    /// Where in the item's body the elaborator is.
50    pub(super) body: BodyContext,
51
52    /// True if we're elaborating a comptime item such as a comptime function,
53    /// block, global, or attribute.
54    pub(super) in_comptime_context: bool,
55
56    /// Set when resolving types in positions where `impl Trait` is not allowed
57    /// (e.g., struct fields, globals, type aliases, enum variants).
58    /// `impl Trait` is only valid in function parameter and return type positions.
59    ///
60    /// This is stored as a field rather than checked at the call site so that it
61    /// propagates through recursive `resolve_type` calls.
62    pub(super) impl_trait_is_disallowed: Option<ImplTraitDisallowedContext>,
63}
64
65impl ItemContext {
66    /// The context of an item described by `module` that is outside any impl or trait, has no
67    /// generics in scope and is not comptime. An entry point whose item has any of those layers
68    /// them on with the chained setters below.
69    pub(super) fn new(module: ModuleContext) -> Self {
70        Self {
71            module,
72            impl_context: ImplContext::default(),
73            generics: GenericsContext::default(),
74            body: BodyContext::default(),
75            in_comptime_context: false,
76            impl_trait_is_disallowed: None,
77        }
78    }
79
80    /// The item belongs to the impl or trait described by `impl_context`.
81    pub(super) fn with_impl(mut self, impl_context: ImplContext) -> Self {
82        self.impl_context = impl_context;
83        self
84    }
85
86    /// The item has `generics` in scope.
87    pub(super) fn with_generics(mut self, generics: GenericsContext) -> Self {
88        self.generics = generics;
89        self
90    }
91
92    /// Whether the item is a comptime one.
93    pub(super) fn in_comptime(mut self, in_comptime_context: bool) -> Self {
94        self.in_comptime_context = in_comptime_context;
95        self
96    }
97
98    /// Types resolved for the item may not mention `impl Trait`, because they sit in `context`.
99    pub(super) fn disallowing_impl_trait(mut self, context: ImplTraitDisallowedContext) -> Self {
100        self.impl_trait_is_disallowed = Some(context);
101        self
102    }
103}
104
105impl Elaborator<'_> {
106    /// Runs `f` with `context` installed, then reinstates the context that was active before.
107    ///
108    /// Whatever `f` leaves in the context is discarded, so an entry point that elaborates another
109    /// item through this helper cannot leak that item's state back to its caller.
110    pub(super) fn with_item_context<T>(
111        &mut self,
112        context: ItemContext,
113        f: impl FnOnce(&mut Self) -> T,
114    ) -> T {
115        let outer = std::mem::replace(&mut self.item, context);
116        let result = f(self);
117        self.item = outer;
118        result
119    }
120}
