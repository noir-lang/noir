1//! The elaborator state that belongs to the item currently being elaborated, and the swap that
2//! keeps one item's elaboration from observing or changing another's.
3//!
4//! Elaborating an item can require elaborating a different item first. For example, a reference to
5//! a function returning `impl Trait` needs that function's body to learn the concrete type, so the
6//! callee is elaborated in the middle of the caller's body. The callee must start from its own
7//! context, and the caller must get its context back unchanged: otherwise the caller goes on
8//! resolving `Self`, generics, trait bounds, loops and `unsafe` blocks as if it were the callee.
9//!
10//! [`Elaborator::with_item_context`] installs a whole [`ItemContext`] for the duration of a closure
11//! and reinstates the previous one afterwards. Since the state lives in one struct that is swapped
12//! as a unit, a field added here is saved and restored by construction.
13
14use crate::{
15    Type,
16    hir::def_map::{LocalModuleId, ModuleId},
17    hir_def::{traits::TraitConstraint, types::ResolvedGeneric},
18    node_interner::{DependencyId, ImplId, TraitId, TraitImplId},
19};
20
21use super::{
22    Elaborator, LambdaContext, Loop, UnsafeBlockStatus, types::ImplTraitDisallowedContext,
23};

25/// The elaborator state describing one item's elaboration.
26///
27/// This is the state that belongs to an item rather than to the crate being elaborated: anything
28/// here is expected to start fresh for an item and to be irrelevant once that item is done. State
29/// shared by the whole elaboration - the interner, the collected errors, the recursion depth - stays
30/// on the [`Elaborator`] itself.
31#[derive(Default)]
32pub(super) struct ItemContext {
33    /// The current module this elaborator is in.
34    /// Initially None, it is set whenever a new top-level item is resolved.
35    pub(super) local_module: Option<LocalModuleId>,
36
37    /// The current dependency item we're resolving.
38    /// Used to link items to their dependencies in the dependency graph
39    pub(super) current_item: Option<DependencyId>,
40
41    /// When set, visibility checks during path resolution use this module
42    /// instead of the default importing module.
43    ///
44    /// Set when resolving an expression on behalf of comptime code from another module (see
45    /// `Expr::resolve`), so that the item resolved that way is held to the caller's visibility
46    /// rather than to that of the scope it is resolved in.
47    pub(super) caller_module: Option<ModuleId>,
48
49    /// Set to the current type if we're resolving an impl
50    pub(super) self_type: Option<Type>,
51
52    /// The trait we're currently resolving or implementing, if any.
53    /// Set during both trait definitions (`trait Foo { ... }`) and
54    /// trait impl elaboration (`impl Foo for Bar { ... }`).
55    pub(super) current_trait: Option<TraitId>,
56
57    /// If we're currently resolving methods within a trait impl, this will be set
58    /// to the corresponding trait impl ID.
59    pub(super) current_trait_impl: Option<TraitImplId>,
60
61    /// If we're currently resolving methods within an inherent (non-trait) impl,
62    /// this will be set to the corresponding impl ID.
63    pub(super) current_impl: Option<ImplId>,
64
65    /// Contains a mapping of the current struct or functions's generics to
66    /// unique type variables if we're resolving a struct. Empty otherwise.
67    /// This is a Vec rather than a map to preserve the order a functions generics
68    /// were declared in.
69    pub(super) generics: Vec<ResolvedGeneric>,
70
71    /// Each constraint in the `where` clause of the function currently being resolved.
72    pub(super) trait_bounds: Vec<TraitConstraint>,
73
74    /// When resolving lambda expressions, we need to keep track of the variables
75    /// that are captured. We do this in order to create the hidden environment
76    /// parameter for the lambda function.
77    pub(super) lambda_stack: Vec<LambdaContext>,
78
79    pub(super) current_loop: Option<Loop>,
80
81    pub(super) unsafe_block_status: UnsafeBlockStatus,
82
83    /// True if we're elaborating a comptime item such as a comptime function,
84    /// block, global, or attribute.
85    pub(super) in_comptime_context: bool,
86
87    /// True if we are elaborating arguments of a function call to an unconstrained function.
88    pub(super) in_unconstrained_args: bool,
89
90    /// Set when resolving types in positions where `impl Trait` is not allowed
91    /// (e.g., struct fields, globals, type aliases, enum variants).
92    /// `impl Trait` is only valid in function parameter and return type positions.
93    ///
94    /// This is stored as a field rather than checked at the call site so that it
95    /// propagates through recursive `resolve_type` calls.
96    pub(super) impl_trait_is_disallowed: Option<ImplTraitDisallowedContext>,
97
98    /// If greater than 0, field visibility errors won't be reported.
99    /// This is used when elaborating a comptime expression that is a struct constructor
100    /// like `Foo { inner: 5 }`: in that case we already elaborated the code that led to
101    /// that comptime value and any visibility errors were already reported.
102    pub(super) silence_field_visibility_errors: usize,
103
104    /// Counter used to define temporary variables for non-simple indexes in l-values.
105    ///
106    /// For example, this expression:
107    ///
108    /// ```noir
109    /// array[x + y] = 10;
110    /// ```
111    ///
112    /// is transformed into:
113    ///
114    /// ```noir
115    /// let i_0 = x + y;
116    /// array[i_0] = 10;
117    /// ```
118    pub(super) lvalue_index_counter: usize,
119}
120
121impl ItemContext {
122    #[tracing::instrument(level = "trace", skip_all)]
123    pub(super) fn next_lvalue_index_counter(&mut self) -> usize {
124        let lvalue_index_counter = self.lvalue_index_counter;
125        self.lvalue_index_counter += 1;
126        lvalue_index_counter
127    }
128}
129
130impl Elaborator<'_> {
131    /// Runs `f` with `context` installed, then reinstates the context that was active before.
132    ///
133    /// Whatever `f` leaves in the context is discarded, so an entry point that elaborates another
134    /// item through this helper cannot leak that item's state back to its caller.
135    pub(super) fn with_item_context<T>(
136        &mut self,
137        context: ItemContext,
138        f: impl FnOnce(&mut Self) -> T,
139    ) -> T {
140        let outer = std::mem::replace(&mut self.item, context);
141        let result = f(self);
142        self.item = outer;
143        result
144    }
145}
