1//! Items whose resolution is deferred until something asks for them.
2//!
3//! Function signatures, struct fields and enum variants may all mention items that a comptime
4//! attribute has not generated yet, so the elaborator cannot resolve them in source order. Each
5//! kind is instead *registered* up front and *resolved* lazily on first read, with whatever is
6//! left drained once the attributes have run.
7//!
8//! All of them follow the same three rules, which is why they share one type:
9//!
10//! 1. Resolving a key removes it, so a re-entrant read of a key that is mid-resolution finds
11//!    nothing and stops rather than looping.
12//! 2. A drain takes a snapshot of what to skip. When `elaborate_items` runs recursively - from a
13//!    comptime attribute that generated new items - the parent's still-pending keys live in the
14//!    same map, and must stay pending for the parent's own drain.
15//! 3. Order is by key, so a drain visits entries in `Id` order and inter-item dependencies that
16//!    follow declaration order are respected.
17
18use std::collections::{BTreeMap, HashSet};

30/// Items of one kind registered for deferred resolution, keyed so that a lazy read can find one
31/// entry and a drain can walk them all in key order.
32pub(crate) struct Deferred<K, V>(BTreeMap<K, V>);
33
34impl<K: Ord + Copy, V> Deferred<K, V> {
35    /// Records `value` as needing resolution later.
36    pub(crate) fn register(&mut self, key: K, value: V) {
37        self.0.insert(key, value);
38    }
39
40    /// Removes the entry for `key`, returning it if this is the first time it has been claimed.
41    ///
42    /// Returning `None` is the ordinary case for a key that was already resolved, and is also what
43    /// breaks cycles: a key currently being resolved is no longer in the map, so a read of it from
44    /// within its own resolution finds nothing instead of recurring.
45    pub(crate) fn take(&mut self, key: &K) -> Option<V> {
46        self.0.remove(key)
47    }
48
49    /// Removes and returns the lowest-keyed entry, for draining in declaration order.
50    pub(crate) fn take_first(&mut self) -> Option<V> {
51        self.0.pop_first().map(|(_, value)| value)
52    }
53
54    /// The keys registered right now, to be passed back to [`Self::keys_except`] later so that a
55    /// nested drain leaves the enclosing one's entries alone.
56    pub(crate) fn pending(&self) -> HashSet<K>
57    where
58        K: std::hash::Hash,
59    {
60        self.0.keys().copied().collect()
61    }
62
63    /// The keys to drain: everything registered except `skip`, in key order.
64    ///
65    /// Collected rather than borrowed because resolving one entry needs `&mut Elaborator`, and may
66    /// itself register or resolve entries here.
67    pub(crate) fn keys_except(&self, skip: &HashSet<K>) -> Vec<K>
68    where
69        K: std::hash::Hash,
70    {
71        self.0.keys().copied().filter(|key| !skip.contains(key)).collect()
72    }
73}
74
75impl<K, V> Default for Deferred<K, V> {
76    fn default() -> Self {
77        Self(BTreeMap::new())
78    }
79}

81#[derive(Default)]
82pub(super) struct PendingTraitWork {
83    /// Trait method declarations registered with deferred meta resolution. These need
84    /// their `TraitFunction` records (in `the_trait.methods`) populated after the
85    /// post-attribute drain, since the records are filled with stub types up-front so
86    /// `collect_trait_impl` can do name-based matching while the real signatures are
87    /// still pending. Each entry is `(trait_id, func_id, name)`.
88    pub(super) records: Vec<(TraitId, FuncId, Ident)>,
89
90    /// Trait method declarations without a body whose signature still needs the
91    /// `elaborate_function` step run after their meta is defined. We can't run it at
92    /// registration time because the meta is deferred.
93    pub(super) no_body_func_ids: Vec<FuncId>,
94
95    /// Pending where-clause-against-trait checks deferred from `collect_trait_impl_methods`
96    /// so they run after the post-attribute drain (when both the trait method's and the
97    /// impl method's metas are fully resolved).
98    pub(super) where_clause_checks: Vec<PendingWhereClauseCheck>,
99}
100
101#[derive(Clone)]
102pub(super) struct PendingWhereClauseCheck {
103    pub(super) impl_method_func_id: FuncId,
104    pub(super) trait_id: TraitId,
105    pub(super) impl_id: TraitImplId,
106    pub(super) module_id: LocalModuleId,
107    pub(super) trait_method_name: String,
108    pub(super) trait_impl_where_clause: Vec<TraitConstraint>,
109    pub(super) ordered_generics: Vec<Type>,
110}

112/// Every kind of deferred work the elaborator owns, in one place so that handing it to a child
113/// elaborator, or scoping it to one `elaborate_items` call, moves all of it or none of it.
114///
115/// Globals are deferred the same way but are not here: they live on the [`Context`] and are shared
116/// by reference, because a global resolved by one elaborator must stay resolved for the next.
117///
118/// [`Context`]: crate::hir::Context
119#[derive(Default)]
120pub(super) struct DeferredItems {
121    /// Function signatures registered before their bodies are elaborated. Registering up front and
122    /// resolving on first read is what lets functions, globals and trait associated constants
123    /// refer to each other regardless of source order.
124    pub(super) function_metas: Deferred<FuncId, UnresolvedFunctionMeta>,
125
126    /// Struct fields, whose types may name items produced by comptime attribute expansion.
127    pub(super) struct_fields: Deferred<TypeId, UnresolvedStructFields>,
128
129    /// Enum variants, deferred for the same reason as [`Self::struct_fields`].
130    pub(super) enum_variants: Deferred<TypeId, UnresolvedEnumVariants>,
131
132    /// Trait bookkeeping that cannot run until the drains above have resolved the metas it reads.
133    pub(super) trait_work: PendingTraitWork,
134}
