1//! The monomorphizer state that belongs to the function currently being monomorphized.
2//!
3//! The [`Monomorphizer`] runs over a whole program, but most of what it tracks while lowering an
4//! expression only makes sense within one function: which HIR definitions map to which locals,
5//! the closure environment, whether the code is unconstrained, and the bindings for the
6//! function's generics. That state lives in a [`FunctionContext`], and
7//! [`Monomorphizer::with_function_context`] installs a fresh one for each function taken off the
8//! queue, so nothing one function leaves behind is visible to the next. State that spans the
9//! whole program - the queue, the finished functions and globals, the ID counters - stays on the
10//! [`Monomorphizer`] itself.
11//!
12//! A lambda body is a function too, and [`Monomorphizer::with_lambda_context`] gives it a context
13//! of its own. That context starts with only the lambda's parameters as locals: everything else
14//! the body reads from the enclosing function is a capture, reached through the lambda's
15//! environment. The one thing it shares with the enclosing function is the generic bindings,
16//! since the lambda's types mention the enclosing function's generics.
17

26/// The closure environment of a lambda whose body is being monomorphized.
27#[derive(Clone)]
28pub(super) struct LambdaContext {
29    pub(super) env_ident: ast::Ident,
30    pub(super) captures: Vec<HirCapturedVar>,
31}

33/// The monomorphizer state describing one function's monomorphization.
34pub(super) struct FunctionContext {
35    /// The monomorphized local created for each HIR definition in the function.
36    ///
37    /// Locals are only keyed by their unique ID because they are never duplicated during
38    /// monomorphization. Doing so would allow them to be used polymorphically but would also
39    /// cause them to be re-evaluated, which is a performance trap that would confuse users.
40    pub(super) locals: HashMap<node_interner::DefinitionId, LocalId>,
41
42    /// The environment holding the captures of the closure being monomorphized, if this is a
43    /// closure body.
44    pub(super) lambda_env: Option<LambdaContext>,
45
46    /// Whether the code being monomorphized is unconstrained. A constrained function called from
47    /// unconstrained code is monomorphized as unconstrained too.
48    pub(super) in_unconstrained_function: bool,
49
50    /// Set while monomorphizing an expression whose target slot is typed `unconstrained fn(..)`.
51    /// Note that this also changes the first-class function representation from a pair of
52    /// `(constrained, unconstrained)` to `(unconstrained, unconstrained)`, so that a constrained
53    /// caller dispatching through slot `.0` still runs the unconstrained version.
54    ///
55    /// This is a property of the position being monomorphized: it holds for the value stored into
56    /// that slot and for nothing else. A binding nested inside that value's expression carries its
57    /// own type and its own slot, so it is monomorphized under its own value of this field.
58    pub(super) force_unconstrained: bool,
59
60    /// Bindings for the generics of the function being monomorphized: the union of every set
61    /// passed to a live [`Monomorphizer::with_bindings`] call. Every type the pass reads from the
62    /// HIR goes through [`Monomorphizer::ty`], which applies them.
63    pub(super) substitution: TypeBindings,
64}
65
66impl FunctionContext {
67    /// The context of a function with no locals, closures or generic bindings yet, whose code is
68    /// unconstrained if `in_unconstrained_function` is set.
69    ///
70    /// `force_brillig` is the program-wide `--force-brillig` setting, which forces every position
71    /// in every function to be unconstrained.
72    pub(super) fn new(in_unconstrained_function: bool, force_brillig: bool) -> Self {
73        Self {
74            locals: HashMap::default(),
75            lambda_env: None,
76            in_unconstrained_function,
77            force_unconstrained: force_brillig,
78            substitution: TypeBindings::default(),
79        }
80    }
81}
82
83impl Monomorphizer<'_> {
84    /// Runs `f` with `context` installed, then reinstates the context that was active before.
85    ///
86    /// Whatever `f` leaves in the context is discarded, so the function it monomorphizes cannot
87    /// leak locals, closure environments or generic bindings into the next one. The only thing
88    /// kept is the function's locals, for [`Monomorphizer::locals`] to report.
89    pub(super) fn with_function_context<T>(
90        &mut self,
91        context: FunctionContext,
92        f: impl FnOnce(&mut Self) -> T,
93    ) -> T {
94        let outer = std::mem::replace(&mut self.function, context);
95        let result = f(self);
96        let inner = std::mem::replace(&mut self.function, outer);
97        self.last_function_locals = inner.locals;
98        result
99    }
100
101    /// Runs `f` with [`FunctionContext::in_unconstrained_function`] set to `unconstrained`, then
102    /// reinstates the previous value.
103    pub(super) fn with_in_unconstrained_function<T>(
104        &mut self,
105        unconstrained: bool,
106        f: impl FnOnce(&mut Self) -> T,
107    ) -> T {
108        let outer = std::mem::replace(&mut self.function.in_unconstrained_function, unconstrained);
109        let result = f(self);
110        self.function.in_unconstrained_function = outer;
111        result
112    }
113
114    /// Runs `f` with [`FunctionContext::force_unconstrained`] set to `forced`, then reinstates the
115    /// previous value.
116    pub(super) fn with_force_unconstrained<T>(
117        &mut self,
118        forced: bool,
119        f: impl FnOnce(&mut Self) -> T,
120    ) -> T {
121        let outer = std::mem::replace(&mut self.function.force_unconstrained, forced);
122        let result = f(self);
123        self.function.force_unconstrained = outer;
124        result
125    }
126
127    /// Runs `f` with the locals that `f` defines collected into a map of their own, which is
128    /// returned alongside `f`'s result. The current context's locals are untouched.
129    ///
130    /// This is how a lambda's parameters are defined: they belong to the lambda's context, which
131    /// [`Self::with_lambda_context`] then installs for the lambda's body.
132    pub(super) fn collecting_locals<T>(
133        &mut self,
134        f: impl FnOnce(&mut Self) -> T,
135    ) -> (T, HashMap<node_interner::DefinitionId, LocalId>) {
136        let outer = std::mem::take(&mut self.function.locals);
137        let result = f(self);
138        let collected = std::mem::replace(&mut self.function.locals, outer);
139        (result, collected)
140    }
141
142    /// Runs `f` to monomorphize the body of a lambda, in a context of the lambda's own.
143    ///
144    /// The context has `locals` (the lambda's parameters) as its only locals, `lambda_env` as its
145    /// closure environment if the lambda has captures, and is unconstrained if `unconstrained` is
146    /// set. It borrows the enclosing context's generic bindings for the duration, then hands them
147    /// back.
148    pub(super) fn with_lambda_context<T>(
149        &mut self,
150        locals: HashMap<node_interner::DefinitionId, LocalId>,
151        lambda_env: Option<LambdaContext>,
152        unconstrained: bool,
153        f: impl FnOnce(&mut Self) -> T,
154    ) -> T {
155        let context = FunctionContext {
156            locals,
157            lambda_env,
158            in_unconstrained_function: unconstrained,
159            force_unconstrained: self.force_brillig,
160            substitution: std::mem::take(&mut self.function.substitution),
161        };
162        let outer = std::mem::replace(&mut self.function, context);
163        let result = f(self);
164        let inner = std::mem::replace(&mut self.function, outer);
165        self.function.substitution = inner.substitution;
166        result
167    }
168}
