50/// Represents a trait in the type system. Each instance of this struct
51/// will be shared across all `Type::Trait` variants that represent
52/// the same trait.
53#[derive(Debug, Eq)]
54pub struct Trait {
55    /// A unique id representing this trait type. Used to check if two
56    /// struct traits are equal.
57    pub id: TraitId,
58
59    pub crate_id: CrateId,
60
61    pub methods: Vec<TraitFunction>,
62
63    /// Maps `method_name` -> method id.
64    /// This map is separate from methods since `TraitFunction` ids
65    /// are created during collection where we don't yet have all
66    /// the information needed to create the full `TraitFunction`.
67    pub method_ids: HashMap<String, FuncId>,
68
69    /// Named generics of the trait.
70    pub associated_types: ResolvedGenerics,
71    pub associated_type_bounds: HashMap<String, Vec<ResolvedTraitBound>>,
72
73    pub name: Ident,
74    /// Ordered generics of the trait.
75    pub generics: ResolvedGenerics,
76    pub location: Location,
77    pub visibility: ItemVisibility,
78
79    /// When resolving the types of Trait elements, all references to `Self` resolve
80    /// to this `TypeVariable`. Then when we check if the types of trait impl elements
81    /// match the definition in the trait, we bind this `TypeVariable` to whatever
82    /// the correct Self type is for that particular impl block.
83    pub self_type_typevar: TypeVariable,
84
85    /// The trait's where clause. Super-trait bounds (`trait Foo: Bar`) are lowered into
86    /// this list as `TraitConstraint { typ: Self, trait_bound: Bar }` so that parent
87    /// bounds and where-clause constraints share a single representation. Use
88    /// [`Trait::parent_bounds`] to extract just the parent-trait bounds.
89    pub where_clause: Vec<TraitConstraint>,
90
91    /// Bounds implied on associated types reached through this trait's own where clause.
92    /// E.g. for `trait Baz<T> where T: Foo` with `trait Foo { type E: Bar; }`, this holds
93    /// the constraint `<T as Foo>::E: Bar`. These are assumed when elaborating the trait's
94    /// default method bodies, but are deliberately kept out of `where_clause` so they do not
95    /// participate in trait-impl where-clause matching or trait-method signature matching.
96    pub implicit_associated_type_constraints: Vec<(TraitConstraint, Location)>,
97
98    pub all_generics: ResolvedGenerics,
99
100    /// Map from each associated constant's name to a unique `DefinitionId` for that constant.
101    pub associated_constant_ids: HashMap<String, DefinitionId>,
102}

154    pub trait_bound: ResolvedTraitBound,
155}

157impl TraitConstraint {
158    /// Update the type in the constraint by substituting the bindings onto it,
159    /// then apply the bindings onto the trait bounds as well.
160    pub fn apply_bindings(&mut self, type_bindings: &TypeBindings) {
161        self.typ = self.typ.substitute(type_bindings);
162        self.trait_bound.apply_bindings(type_bindings);
163    }
164
165    pub fn to_string(&self, interner: &NodeInterner) -> String {
166        interner.trait_constraint_string(
167            &self.typ,
168            self.trait_bound.trait_id,
169            &self.trait_bound.trait_generics.ordered,
170            &self.trait_bound.trait_generics.named,
171        )
172    }
173
174    /// Whether `self` and `other` denote the same bound, treating an unspecified associated
175    /// type as matching any other unspecified one.
176    ///
177    /// This is weaker than equality, for comparing a constraint against a copy of it that was
178    /// resolved independently — e.g. an inherent impl's where clause copied onto a method, or a
179    /// trait's supertrait bound propagated onto a method. Each resolution fills in any
180    /// associated type the bound leaves unspecified with a *fresh* type variable, so the copies
181    /// aren't `==` even though they're the same bound. Associated types the user bound to a
182    /// concrete type (`Foo<Bar = u32>`) are still compared, so those bounds aren't conflated.
183    pub fn matches_ignoring_unspecified_associated_types(&self, other: &TraitConstraint) -> bool {
184        if self.typ != other.typ
185            || self.trait_bound.trait_id != other.trait_bound.trait_id
186            || self.trait_bound.trait_generics.ordered != other.trait_bound.trait_generics.ordered
187        {
188            return false;
189        }
190
191        let self_named = &self.trait_bound.trait_generics.named;
192        let other_named = &other.trait_bound.trait_generics.named;
193        if self_named.len() != other_named.len() {
194            return false;
195        }
196
197        // An associated type the bound leaves unspecified is filled in with a fresh type
198        // variable each time the bound is resolved, so two copies of the same bound carry
199        // different ones there. Depending on the resolution path that filler is a bare type
200        // variable or a named generic, so accept either (when unbound) as matching, while still
201        // comparing associated types the user bound to a concrete type (e.g. `Foo<Bar = u32>`).
202        let is_unbound = |typ: &Type| match typ {
203            Type::TypeVariable(var) | Type::NamedGeneric(NamedGeneric { type_var: var, .. }) => {
204                var.binding().is_unbound()
205            }
206            _ => false,
207        };
208        self_named.iter().zip(other_named).all(|(a, b)| {
209            a.name == b.name && (a.typ == b.typ || (is_unbound(&a.typ) && is_unbound(&b.typ)))
210        })
211    }
212
213    /// Looks up a trait implementation which satisfies this constraint and returns it.
214    ///
215    /// Note that if successful, any type bindings from the impl search will be automatically
216    /// applied, unless `current_trait_self` is `Some` type variable matching the constraint,
217    /// representing a trait definition, in which case we don't bind it, but also don't reject
218    /// it as a missing type.
219    pub fn find_impl(
220        &self,
221        interner: &NodeInterner,
222        current_trait_self: Option<&Type>,
223    ) -> Result<(TraitImplKind, TypeBindings), ImplSearchErrorKind> {
224        match current_trait_self {
225            Some(typ) if *typ == self.typ && typ.is_bindable() => {
226                let (impl_kind, _bindings, instantiation_bindings) = interner
227                    .try_lookup_trait_implementation(
228                        &self.typ,
229                        self.trait_bound.trait_id,
230                        &self.trait_bound.trait_generics.ordered,
231                        &self.trait_bound.trait_generics.named,
232                        TraitLookupMode::SelfAssumedOnly,
233                    )?;
234                // Do not bind it the self type.
235                Ok((impl_kind, instantiation_bindings))
236            }
237            _ => interner.lookup_trait_implementation(
238                &self.typ,
239                self.trait_bound.trait_id,
240                &self.trait_bound.trait_generics.ordered,
241                &self.trait_bound.trait_generics.named,
242            ),
243        }
244    }
245}

248pub struct ResolvedTraitBound {
249    pub trait_id: TraitId,
250    pub trait_generics: TraitGenerics,
251    pub location: Location,
252}

254/// A parent-trait bound (the `Bar<Self>` in `trait Foo: Bar<Self>`) as declared on a trait.
255///
256/// It is written in terms of the declaring trait's own `Self`, generics and associated types,
257/// which are shared by every use of that trait. It only describes a real bound once those are
258/// replaced by the ones of a particular `T: Foo<..>`, so the declared form is not exposed except
259/// for display: read it with [`ParentBound::instantiate`].
260#[derive(Debug, Clone)]
261pub struct ParentBound(ResolvedTraitBound);
262
263impl ParentBound {
264    pub fn trait_id(&self) -> TraitId {
265        self.0.trait_id
266    }
267
268    /// The bound for one use of the declaring trait. `bindings` maps the declaring trait's
269    /// `Self`, generics and associated types to that use's (see [`Trait::bound_bindings`]).
270    ///
271    /// An associated item the bound leaves out (`trait Foo: Bar` where `Bar` has `type Out`) is
272    /// a placeholder declared once for the whole trait; `fresh_placeholder` replaces it with one
273    /// for this instantiation, so that resolving one use cannot bind it for every other.
274    pub fn instantiate(
275        &self,
276        bindings: &TypeBindings,
277        mut fresh_placeholder: impl FnMut(Kind) -> Type,
278    ) -> ResolvedTraitBound {
279        let mut trait_generics = self.0.trait_generics.map(|typ| typ.substitute(bindings));
280        for (named, declared) in trait_generics.named.iter_mut().zip(&self.0.trait_generics.named) {
281            if let Type::TypeVariable(placeholder) = &declared.typ {
282                named.typ = fresh_placeholder(placeholder.kind().into_owned());
283            }
284        }
285        ResolvedTraitBound { trait_generics, ..self.0 }
286    }
287
288    /// The bound exactly as written in the trait declaration, mentioning the declaring trait's
289    /// own `Self` and generics. Only for displaying the declaration.
290    pub fn as_written(&self) -> &ResolvedTraitBound {
291        &self.0
292    }
293}
294
295impl ResolvedTraitBound {
296    /// Update all [Type]s in the bound generics by substituting some [`TypeBindings`] onto them.
297    pub fn apply_bindings(&mut self, type_bindings: &TypeBindings) {
298        for typ in &mut self.trait_generics.ordered {
299            *typ = typ.substitute(type_bindings);
300        }
301
302        for named in &mut self.trait_generics.named {
303            named.typ = named.typ.substitute(type_bindings);
304        }
305    }
306}

312    }
313}

336    /// The parent-trait bounds of this trait (the `Bar` in `trait Foo: Bar`).
337    ///
338    /// Parent bounds are stored in `where_clause` as constraints whose `typ` is this
339    /// trait's `Self` (see [`Self::is_self_type`]); this accessor filters them back out.
340    pub fn parent_bounds(&self) -> impl Iterator<Item = ParentBound> + '_ {
341        self.where_clause
342            .iter()
343            .filter(|c| self.is_self_type(&c.typ))
344            .map(|c| ParentBound(c.trait_bound.clone()))
345    }
346
347    /// Bindings from this trait's own ordered generics and associated types to the arguments
348    /// of `trait_generics`, a bound on this trait.
349    pub fn bind_generics(&self, trait_generics: &TraitGenerics, bindings: &mut TypeBindings) {
350        bind_ordered_generics(&self.generics, &trait_generics.ordered, bindings);
351        bind_named_generics(self.associated_types.clone(), &trait_generics.named, bindings);
352    }
353
354    /// Bindings from this trait's own `Self`, ordered generics and associated types to those of
355    /// the bound `self_type: ThisTrait<trait_generics>`. Substituting them into anything declared
356    /// on this trait gives its meaning for that bound.
357    pub fn bound_bindings(&self, self_type: &Type, trait_generics: &TraitGenerics) -> TypeBindings {
358        let mut bindings = TypeBindings::default();
359        self.bind_generics(trait_generics, &mut bindings);
360        self.bind_self(self_type, &mut bindings);
361        bindings
362    }
363
364    /// Binds this trait's `Self` to `self_type`, and each of its ordered generics and associated
365    /// types to the argument `generics` gives for it, adding to `bindings`. An associated type
366    /// that `generics` leaves out gets no entry.
367    ///
368    /// Panics if `generics` doesn't give exactly one argument per ordered generic, or names an
369    /// associated type this trait doesn't have.
370    pub fn bind_given_arguments(
371        &self,
372        self_type: &Type,
373        generics: &TraitGenerics,
374        bindings: &mut TypeBindings,
375    ) {
376        self.bind_self(self_type, bindings);
377        bind_ordered_generics(&self.generics, &generics.ordered, bindings);
378        for arg in &generics.named {
379            let param = self.get_associated_type(arg.name.as_str()).unwrap_or_else(|| {
380                unreachable!("Expected to find associated type named {}", arg.name)
381            });
382            bind_generic(param, &arg.typ, bindings);
383        }
384    }
385
386    fn bind_self(&self, self_type: &Type, bindings: &mut TypeBindings) {
387        let self_var = self.self_type_typevar.clone();
388        let self_kind = self_var.kind().into_owned();
389        bindings.insert(self_var.id(), (self_var, self_kind, self_type.clone()));
390    }
391
392    /// Whether `typ` is this trait's own `Self`, in either its rigid form (a named generic
393    /// over `self_type_typevar`) or its bindable form (the bare type variable).
394    pub fn is_self_type(&self, typ: &Type) -> bool {
395        match typ {
396            Type::TypeVariable(v) | Type::NamedGeneric(NamedGeneric { type_var: v, .. }) => {
397                v.id() == self.self_type_typevar.id()
398            }
399            _ => false,
400        }
401    }

439    /// Find an associated type by the last segments in its name.
440    ///
441    /// For example if a method returns `Self::Foo`, here we will be looking for it by the name `"Foo"`.
442    pub fn get_associated_type(&self, last_name: &str) -> Option<&ResolvedGeneric> {
443        self.associated_types.iter().find(|typ| typ.name.as_ref() == last_name)
444    }
445
446    /// Returns both the ordered generics of this type, and its named, associated types.
447    /// These types are all as-is and are not instantiated.
448    pub fn get_generics(&self) -> (Vec<Type>, Vec<Type>) {
449        let ordered = vecmap(&self.generics, |generic| generic.clone().into_named_generic(None));
450        let named = vecmap(&self.associated_types, |generic| {
451            generic.clone().into_named_generic(Some((SELF_TYPE_NAME, self.name.as_str())))
452        });
453        (ordered, named)
454    }
455
456    pub fn get_trait_generics(&self, location: Location) -> TraitGenerics {
457        let ordered = vecmap(&self.generics, |generic| generic.clone().into_named_generic(None));
458        let named = vecmap(&self.associated_types, |generic| {
459            let name = Ident::new(generic.name.to_string(), location);
460            NamedType {
461                name,
462                typ: generic.clone().into_named_generic(Some((SELF_TYPE_NAME, self.name.as_str()))),
463            }
464        });
465        TraitGenerics { ordered, named }
466    }
467
468    /// Returns a `TraitConstraint` for this trait using Self as the object
469    /// type and the uninstantiated generics for any trait generics.
470    ///
471    /// `Self` is the rigid named generic over `self_type_typevar`, the same form trait method
472    /// signatures use. A bindable `Type::TypeVariable` here would let a default body unify
473    /// `Self` with a concrete type, and that binding would be seen by every other default
474    /// method of the trait type-checked afterwards.
475    pub fn as_constraint(&self, location: Location) -> TraitConstraint {
476        let trait_generics = self.get_trait_generics(location);
477        TraitConstraint {
478            typ: self.self_type(),
479            trait_bound: ResolvedTraitBound { trait_generics, trait_id: self.id, location },
480        }
481    }
482
483    /// The rigid `Self` type for this trait: a named generic over `self_type_typevar`, which
484    /// cannot be unified with a concrete type. Every use of `Self` outside of the trait's own
485    /// declaration bookkeeping (method signatures, assumed `Self: CurrentTrait` bounds, the
486    /// elaboration context installed while checking a default method body) must go through
487    /// this accessor rather than wrapping `self_type_typevar` in `Type::TypeVariable` directly.
488    /// That bindable form lets any unification (including an impl search) bind the variable,
489    /// and the binding is then visible to every other use of the trait's `Self` for the rest of
490    /// compilation, since there is exactly one `self_type_typevar` per trait.
491    pub fn self_type(&self) -> Type {
492        let self_type_name = Rc::new(SELF_TYPE_NAME.to_string());
493        self.self_type_typevar.clone().into_named_generic(&self_type_name, None)
494    }
495}

503impl TraitFunction {
504    pub fn arguments(&self) -> &[Type] {
505        match &self.typ {
506            Type::Function(args, _, _, _) => args,
507            Type::Forall(_, typ) => match typ.as_ref() {
508                Type::Function(args, _, _, _) => args,
509                _ => unreachable!("Trait function does not have a function type"),
510            },
511            _ => unreachable!("Trait function does not have a function type"),
512        }
513    }
514
515    pub fn generics(&self) -> &[TypeVariable] {
516        match &self.typ {
517            Type::Function(..) => &[],
518            Type::Forall(generics, _) => generics,
519            _ => unreachable!("Trait function does not have a function type"),
520        }
521    }
522
523    pub fn return_type(&self) -> &Type {
524        match &self.typ {
525            Type::Function(_, return_type, _, _) => return_type,
526            Type::Forall(_, typ) => match typ.as_ref() {
527                Type::Function(_, return_type, _, _) => return_type,
528                _ => unreachable!("Trait function does not have a function type"),
529            },
530            _ => unreachable!("Trait function does not have a function type"),
531        }
532    }
533}

535/// Binds the ordered [`ResolvedGeneric`]s of a trait to the ordered generics in a [`ResolvedTraitBound`].
536///
537/// Panics if the number of types do not match the ordered generics in the trait.
538fn bind_ordered_generics(params: &[ResolvedGeneric], args: &[Type], bindings: &mut TypeBindings) {
539    assert_eq!(params.len(), args.len(), "unexpected number of ordered generics");
540
541    for (param, arg) in params.iter().zip(args) {
542        bind_generic(param, arg, bindings);
543    }
544}
545
546/// Binds the associated [`ResolvedGeneric`]s of a trait to the named generics in a [`ResolvedTraitBound`].
547///
548/// Panics if the number of types exceeds the named generics in the trait.
549/// Any named parameter that does not appear in the arguments is bound to [`Type::Error`].
550fn bind_named_generics(
551    mut params: Vec<ResolvedGeneric>,
552    args: &[NamedType],
553    bindings: &mut TypeBindings,
554) {
555    assert!(
556        args.len() <= params.len(),
557        "bind_named_generics: trait bound has more named generics than associated types"
558    );
559
560    if params.is_empty() {
561        return;
562    }
563
564    for arg in args {
565        let i = params
566            .iter()
567            .position(|typ| *typ.name == arg.name.as_str())
568            .unwrap_or_else(|| unreachable!("Expected to find associated type named {}", arg.name));
569
570        let param = params.swap_remove(i);
571
572        bind_generic(&param, &arg.typ, bindings);
573    }
574
575    for unbound_param in params {
576        bind_generic(&unbound_param, &Type::Error, bindings);
577    }
578}

580/// Binds the type variable in a [`ResolvedGeneric`], e.g. a generic parameter of a trait,
581/// to a [Type], which itself can be an unbound type variable.
582///
583/// If the type variable itself appears in the type, then it does nothing.
584fn bind_generic(param: &ResolvedGeneric, arg: &Type, bindings: &mut TypeBindings) {
585    // Avoid binding t = t
586    if !arg.occurs(param.type_var.id()) {
587        bindings.insert(
588            param.type_var.id(),
589            (param.type_var.clone(), param.kind().into_owned(), arg.clone()),
590        );
591    }
