50/// Represents a trait in the type system. Each instance of this struct
51/// will be shared across all `Type::Trait` variants that represent
52/// the same trait.
53#[derive(Debug, Eq)]
54pub struct Trait {
55    /// A unique id representing this trait type. Used to check if two
56    /// struct traits are equal.
57    pub id: TraitId,
58
59    pub crate_id: CrateId,
60
61    pub methods: Vec<TraitFunction>,
62
63    /// Maps `method_name` -> method id.
64    /// This map is separate from methods since `TraitFunction` ids
65    /// are created during collection where we don't yet have all
66    /// the information needed to create the full `TraitFunction`.
67    pub method_ids: HashMap<String, FuncId>,
68
69    /// Named generics of the trait.
70    pub associated_types: ResolvedGenerics,
71    pub associated_type_bounds: HashMap<String, Vec<ResolvedTraitBound>>,
72
73    pub name: Ident,
74    /// Ordered generics of the trait.
75    pub generics: ResolvedGenerics,
76    pub location: Location,
77    pub visibility: ItemVisibility,
78
79    /// When resolving the types of Trait elements, all references to `Self` resolve
80    /// to this `TypeVariable`. Then when we check if the types of trait impl elements
81    /// match the definition in the trait, we bind this `TypeVariable` to whatever
82    /// the correct Self type is for that particular impl block.
83    pub self_type_typevar: TypeVariable,
84
85    /// The trait's where clause. Super-trait bounds (`trait Foo: Bar`) are lowered into
86    /// this list as `TraitConstraint { typ: Self, trait_bound: Bar }` so that parent
87    /// bounds and where-clause constraints share a single representation. Use
88    /// [`Trait::parent_bounds`] to extract just the parent-trait bounds.
89    pub where_clause: Vec<TraitConstraint>,
90
91    /// Bounds implied on associated types reached through this trait's own where clause.
92    /// E.g. for `trait Baz<T> where T: Foo` with `trait Foo { type E: Bar; }`, this holds
93    /// the constraint `<T as Foo>::E: Bar`. These are assumed when elaborating the trait's
94    /// default method bodies, but are deliberately kept out of `where_clause` so they do not
95    /// participate in trait-impl where-clause matching or trait-method signature matching.
96    pub implicit_associated_type_constraints: Vec<(TraitConstraint, Location)>,
97
98    pub all_generics: ResolvedGenerics,
99
100    /// Map from each associated constant's name to a unique `DefinitionId` for that constant.
101    pub associated_constant_ids: HashMap<String, DefinitionId>,
102}

151#[derive(Debug, Clone, PartialEq, Eq)]
152pub struct TraitConstraint {
153    pub typ: Type,
154    pub trait_bound: ResolvedTraitBound,
155}
156
157impl TraitConstraint {
158    /// Update the type in the constraint by substituting the bindings onto it,
159    /// then apply the bindings onto the trait bounds as well.
160    pub fn apply_bindings(&mut self, type_bindings: &TypeBindings) {
161        self.typ = self.typ.substitute(type_bindings);
162        self.trait_bound.apply_bindings(type_bindings);
163    }
164
165    pub fn to_string(&self, interner: &NodeInterner) -> String {
166        interner.trait_constraint_string(
167            &self.typ,
168            self.trait_bound.trait_id,
169            &self.trait_bound.trait_generics.ordered,
170            &self.trait_bound.trait_generics.named,
171        )
172    }
173
174    /// Whether `self` and `other` denote the same bound, treating an unspecified associated
175    /// type as matching any other unspecified one.
176    ///
177    /// This is weaker than equality, for comparing a constraint against a copy of it that was
178    /// resolved independently — e.g. an inherent impl's where clause copied onto a method, or a
179    /// trait's supertrait bound propagated onto a method. Each resolution fills in any
180    /// associated type the bound leaves unspecified with a *fresh* type variable, so the copies
181    /// aren't `==` even though they're the same bound. Associated types the user bound to a
182    /// concrete type (`Foo<Bar = u32>`) are still compared, so those bounds aren't conflated.
183    pub fn matches_ignoring_unspecified_associated_types(&self, other: &TraitConstraint) -> bool {
184        if self.typ != other.typ
185            || self.trait_bound.trait_id != other.trait_bound.trait_id
186            || self.trait_bound.trait_generics.ordered != other.trait_bound.trait_generics.ordered
187        {
188            return false;
189        }
190
191        let self_named = &self.trait_bound.trait_generics.named;
192        let other_named = &other.trait_bound.trait_generics.named;
193        if self_named.len() != other_named.len() {
194            return false;
195        }
196
197        // An associated type the bound leaves unspecified is filled in with a fresh type
198        // variable each time the bound is resolved, so two copies of the same bound carry
199        // different ones there. Depending on the resolution path that filler is a bare type
200        // variable or a named generic, so accept either (when unbound) as matching, while still
201        // comparing associated types the user bound to a concrete type (e.g. `Foo<Bar = u32>`).
202        let is_unbound = |typ: &Type| match typ {
203            Type::TypeVariable(var) | Type::NamedGeneric(NamedGeneric { type_var: var, .. }) => {
204                var.binding().is_unbound()
205            }
206            _ => false,
207        };
208        self_named.iter().zip(other_named).all(|(a, b)| {
209            a.name == b.name && (a.typ == b.typ || (is_unbound(&a.typ) && is_unbound(&b.typ)))
210        })
211    }
212
213    /// Looks up a trait implementation which satisfies this constraint and returns it.
214    ///
215    /// Note that if successful, any type bindings from the impl search will be automatically
216    /// applied, unless `current_trait_self` is `Some` type variable matching the constraint,
217    /// representing a trait definition, in which case we don't bind it, but also don't reject
218    /// it as a missing type.
219    pub fn find_impl(
220        &self,
221        interner: &NodeInterner,
222        current_trait_self: Option<&Type>,
223    ) -> Result<(TraitImplKind, TypeBindings), ImplSearchErrorKind> {
224        match current_trait_self {
225            Some(typ) if *typ == self.typ && typ.is_bindable() => {
226                let (impl_kind, _bindings, instantiation_bindings) = interner
227                    .try_lookup_trait_implementation(
228                        &self.typ,
229                        self.trait_bound.trait_id,
230                        &self.trait_bound.trait_generics.ordered,
231                        &self.trait_bound.trait_generics.named,
232                        TraitLookupMode::SelfAssumedOnly,
233                    )?;
234                // Do not bind it the self type.
235                Ok((impl_kind, instantiation_bindings))
236            }
237            _ => interner.lookup_trait_implementation(
238                &self.typ,
239                self.trait_bound.trait_id,
240                &self.trait_bound.trait_generics.ordered,
241                &self.trait_bound.trait_generics.named,
242            ),
243        }
244    }
245}

247#[derive(Debug, Clone, Eq)]
248pub struct ResolvedTraitBound {
249    pub trait_id: TraitId,
250    pub trait_generics: TraitGenerics,
251    pub location: Location,
252}
253
254impl ResolvedTraitBound {
255    /// Update all [Type]s in the bound generics by substituting some [`TypeBindings`] onto them.
256    pub fn apply_bindings(&mut self, type_bindings: &TypeBindings) {
257        for typ in &mut self.trait_generics.ordered {
258            *typ = typ.substitute(type_bindings);
259        }
260
261        for named in &mut self.trait_generics.named {
262            named.typ = named.typ.substitute(type_bindings);
263        }
264    }
265}
266
267impl PartialEq for ResolvedTraitBound {
268    fn eq(&self, other: &Self) -> bool {
269        // Location doesn't matter for equality
270        self.trait_id == other.trait_id && self.trait_generics == other.trait_generics
271    }
272}

294
295    /// The parent-trait bounds of this trait (the `Bar` in `trait Foo: Bar`).
296    ///
297    /// Parent bounds are stored in `where_clause` as constraints whose `typ` is this
298    /// trait's `Self` (see [`Self::is_self_type`]); this accessor filters them back out.
299    pub fn parent_bounds(&self) -> impl Iterator<Item = &ResolvedTraitBound> {
300        self.where_clause.iter().filter(|c| self.is_self_type(&c.typ)).map(|c| &c.trait_bound)
301    }
302
303    /// Whether `typ` is this trait's own `Self`, in either its rigid form (a named generic
304    /// over `self_type_typevar`) or its bindable form (the bare type variable).
305    pub fn is_self_type(&self, typ: &Type) -> bool {
306        match typ {
307            Type::TypeVariable(v) | Type::NamedGeneric(NamedGeneric { type_var: v, .. }) => {
308                v.id() == self.self_type_typevar.id()
309            }
310            _ => false,
311        }
312    }

320    }

322    pub fn set_associated_type_bounds(

324        associated_type_bounds: HashMap<String, Vec<ResolvedTraitBound>>,

326        self.associated_type_bounds = associated_type_bounds;
327    }

329    pub fn find_method(&self, name: &str, interner: &NodeInterner) -> Option<DefinitionId> {
330        for method in &self.methods {
331            if &method.name == name {
332                let id = *self.method_ids.get(name).unwrap();
333                return Some(interner.function_definition_id(id));
334            }
335        }
336        None
337    }
338
339    pub fn find_method_or_constant(
340        &self,
341        name: &str,
342        interner: &NodeInterner,
343    ) -> Option<DefinitionId> {
344        if let Some(method) = self.find_method(name, interner) {
345            return Some(method);
346        }
347        self.associated_constant_ids.get(name).copied()
348    }
349
350    /// Find an associated type by the last segments in its name.
351    ///
352    /// For example if a method returns `Self::Foo`, here we will be looking for it by the name `"Foo"`.
353    pub fn get_associated_type(&self, last_name: &str) -> Option<&ResolvedGeneric> {
354        self.associated_types.iter().find(|typ| typ.name.as_ref() == last_name)
355    }
356
357    /// Returns both the ordered generics of this type, and its named, associated types.
358    /// These types are all as-is and are not instantiated.
359    pub fn get_generics(&self) -> (Vec<Type>, Vec<Type>) {
360        let ordered = vecmap(&self.generics, |generic| generic.clone().into_named_generic(None));
361        let named = vecmap(&self.associated_types, |generic| {
362            generic.clone().into_named_generic(Some((SELF_TYPE_NAME, self.name.as_str())))
363        });
364        (ordered, named)
365    }
366
367    pub fn get_trait_generics(&self, location: Location) -> TraitGenerics {
368        let ordered = vecmap(&self.generics, |generic| generic.clone().into_named_generic(None));
369        let named = vecmap(&self.associated_types, |generic| {
370            let name = Ident::new(generic.name.to_string(), location);
371            NamedType {
372                name,
373                typ: generic.clone().into_named_generic(Some((SELF_TYPE_NAME, self.name.as_str()))),
374            }
375        });
376        TraitGenerics { ordered, named }
377    }
378
379    /// Returns a `TraitConstraint` for this trait using Self as the object
380    /// type and the uninstantiated generics for any trait generics.
381    ///
382    /// `Self` is the rigid named generic over `self_type_typevar`, the same form trait method
383    /// signatures use. A bindable `Type::TypeVariable` here would let a default body unify
384    /// `Self` with a concrete type, and that binding would be seen by every other default
385    /// method of the trait type-checked afterwards.
386    pub fn as_constraint(&self, location: Location) -> TraitConstraint {
387        let trait_generics = self.get_trait_generics(location);
388        TraitConstraint {
389            typ: self.self_type(),
390            trait_bound: ResolvedTraitBound { trait_generics, trait_id: self.id, location },
391        }
392    }
393
394    /// The rigid `Self` type for this trait: a named generic over `self_type_typevar`, which
395    /// cannot be unified with a concrete type. Every use of `Self` outside of the trait's own
396    /// declaration bookkeeping (method signatures, assumed `Self: CurrentTrait` bounds, the
397    /// elaboration context installed while checking a default method body) must go through
398    /// this accessor rather than wrapping `self_type_typevar` in `Type::TypeVariable` directly.
399    /// That bindable form lets any unification (including an impl search) bind the variable,
400    /// and the binding is then visible to every other use of the trait's `Self` for the rest of
401    /// compilation, since there is exactly one `self_type_typevar` per trait.
402    pub fn self_type(&self) -> Type {
403        let self_type_name = Rc::new(SELF_TYPE_NAME.to_string());
404        self.self_type_typevar.clone().into_named_generic(&self_type_name, None)
405    }
406}

414impl TraitFunction {
415    pub fn arguments(&self) -> &[Type] {
416        match &self.typ {
417            Type::Function(args, _, _, _) => args,
418            Type::Forall(_, typ) => match typ.as_ref() {
419                Type::Function(args, _, _, _) => args,
420                _ => unreachable!("Trait function does not have a function type"),
421            },
422            _ => unreachable!("Trait function does not have a function type"),
423        }
424    }
425
426    pub fn generics(&self) -> &[TypeVariable] {
427        match &self.typ {
428            Type::Function(..) => &[],
429            Type::Forall(generics, _) => generics,
430            _ => unreachable!("Trait function does not have a function type"),
431        }
432    }
433
434    pub fn return_type(&self) -> &Type {
435        match &self.typ {
436            Type::Function(_, return_type, _, _) => return_type,
437            Type::Forall(_, typ) => match typ.as_ref() {
438                Type::Function(_, return_type, _, _) => return_type,
439                _ => unreachable!("Trait function does not have a function type"),
440            },
441            _ => unreachable!("Trait function does not have a function type"),
442        }
443    }
