6/// A type to prevent infinite recursion while traversing types recursively.
7///
8/// Designed to be cloned at branching (e.g. when visiting fields of tuples, or recursing),
9/// so that types at the same level are counted separately. The only type recursion we want
10/// to prevent is cycles.
11#[derive(Clone, Default)]
12pub(crate) struct TypeRecursionContext {
13    depth: u32,
14    data_types: imbl::HashSet<(TypeId, Vec<Type>)>,
15    aliases: imbl::HashSet<(TypeAliasId, Vec<Type>)>,
16}
17
18impl TypeRecursionContext {
19    /// Increases the recursion depth.
20    ///
21    /// Panics if it would go beyond [`TYPE_RECURSION_LIMIT`].
22    pub(crate) fn recur(mut self) -> Self {
23        if self.depth >= TYPE_RECURSION_LIMIT {
24            panic!("Type recursion limit reached - types are too large");
25        }
26        self.depth += 1;
27        self
28    }
29
30    /// Tracks a data type and its generics. Returns whether the data type wasn't already being tracked.
31    pub(crate) fn insert_data_type(&mut self, data_type_id: TypeId, generics: Vec<Type>) -> bool {
32        self.data_types.insert((data_type_id, generics)).is_none()
33    }
34
35    /// Tracks an alias and its generics. Returns whether the alias wasn't already being tracked.
36    pub(crate) fn insert_alias(&mut self, alias_id: TypeAliasId, generics: Vec<Type>) -> bool {
37        self.aliases.insert((alias_id, generics)).is_none()
38    }
39}
40
41impl Type {
42    /// Whether `predicate` holds for this type or for any type reachable from it.
43    ///
44    /// "Reachable" is the shape of a *value* of this type, so the walk expands aliases, follows
45    /// bound type variables, and descends into a data type's fields or variants — a struct whose
46    /// field is a vector contains a vector. It stops at cycles, so a recursive type terminates.
47    ///
48    /// Two positions are deliberately not reachable:
49    ///
50    /// * A function's parameter and return types. Only its environment is carried in a value of
51    ///   the function type; arguments are passed in and the return type is returned.
52    /// * A trait-as-type's generics, which describe the bound rather than the represented value.
53    pub(crate) fn contains_matching(&self, mut predicate: impl FnMut(&Type) -> bool) -> bool {
54        self.contains_matching_helper(&mut predicate, TypeRecursionContext::default())
55    }
56
57    fn contains_matching_helper(
58        &self,
59        predicate: &mut impl FnMut(&Type) -> bool,
60        mut context: TypeRecursionContext,
61    ) -> bool {
62        if predicate(self) {
63            return true;
64        }
65
66        // Every recursive call goes through this, so that a caller cannot descend into one of the
67        // positions above by forgetting which arm it was writing.
68        let mut descend = |typ: &Type, context: TypeRecursionContext| {
69            typ.contains_matching_helper(predicate, context)
70        };
71
72        match self {
73            Type::FieldElement
74            | Type::Integer(..)
75            | Type::Bool
76            | Type::Unit
77            | Type::Constant(..)
78            | Type::Quoted(..)
79            | Type::TraitAsType(..)
80            | Type::Error => false,
81
82            Type::Vector(element) | Type::String(element) | Type::Reference(element, _) => {
83                descend(element, context.recur())
84            }
85            Type::Forall(_, typ) => descend(typ, context.recur()),
86
87            Type::Array(element, length) | Type::FmtString(length, element) => {
88                descend(length, context.clone().recur()) || descend(element, context.recur())
89            }
90            Type::CheckedCast { from, to } => {
91                descend(from, context.clone().recur()) || descend(to, context.recur())
92            }
93            Type::InfixExpr(lhs, _op, rhs, _) => {
94                descend(lhs, context.clone().recur()) || descend(rhs, context.recur())
95            }
96
97            // Only the environment is carried in a value of a function type.
98            Type::Function(_args, _ret, env, _unconstrained) => descend(env, context.recur()),
99
100            Type::Tuple(elements) => {
101                elements.iter().any(|element| descend(element, context.clone().recur()))
102            }
103
104            Type::TypeVariable(type_variable)
105            | Type::NamedGeneric(NamedGeneric { type_var: type_variable, .. }) => {
106                let binding = type_variable.borrow();
107                match &*binding {
108                    TypeBinding::Bound(bound) => descend(bound, context.recur()),
109                    TypeBinding::Unbound(..) => false,
110                }
111            }
112
113            Type::Alias(alias, generics) => {
114                if !context.insert_alias(alias.borrow().id, generics.clone()) {
115                    return false;
116                }
117                let aliased = alias.borrow().get_type(generics);
118                descend(&aliased, context.recur())
119            }
120
121            Type::DataType(definition, generics) => {
122                let definition = definition.borrow();
123                if !context.insert_data_type(definition.id, generics.clone()) {
124                    return false;
125                }
126                if let Some(fields) = definition.get_fields(generics) {
127                    fields.iter().any(|(_, field, _)| descend(field, context.clone().recur()))
128                } else if let Some(variants) = definition.get_variants(generics) {
129                    variants
130                        .iter()
131                        .flat_map(|(_, arguments)| arguments)
132                        .any(|argument| descend(argument, context.clone().recur()))
133                } else {
134                    false
135                }
136            }
137        }
138    }
139}

141#[cfg(test)]
142mod tests {
143    use crate::{Kind, Type, TypeVariable, TypeVariableId};
144
145    fn contains_vector(typ: &Type) -> bool {
146        typ.contains_matching(|typ| matches!(typ, Type::Vector(_)))
147    }
148
149    fn vector() -> Type {
150        Type::Vector(Box::new(Type::FieldElement))
151    }
152
153    #[test]
154    fn matches_the_type_itself() {
155        assert!(contains_vector(&vector()));
156        assert!(!contains_vector(&Type::FieldElement));
157    }
158
159    #[test]
160    fn descends_through_containers() {
161        assert!(contains_vector(&Type::Tuple(vec![Type::Bool, vector()])));
162        assert!(contains_vector(&Type::Array(Box::new(Type::constant_u32(2)), Box::new(vector()))));
163        assert!(contains_vector(&Type::Reference(Box::new(vector()), true)));
164        assert!(!contains_vector(&Type::Tuple(vec![Type::Bool, Type::FieldElement])));
165    }
166
167    #[test]
168    fn follows_bound_type_variables() {
169        let type_variable = TypeVariable::unbound(TypeVariableId(0), Kind::Normal);
170        let typ = Type::TypeVariable(type_variable.clone());
171        assert!(!contains_vector(&typ));
172
173        type_variable.bind(vector());
174        assert!(contains_vector(&typ));
175    }
176
177    /// Only a function's environment is carried in a value of the function type: its arguments are
178    /// passed in and its return type is returned, so neither is part of the value's shape.
179    #[test]
180    fn reads_a_function_type_as_its_environment_only() {
181        let function = |args, ret, env| Type::Function(args, Box::new(ret), Box::new(env), false);
182
183        assert!(!contains_vector(&function(vec![vector()], Type::Bool, Type::Unit)));
184        assert!(!contains_vector(&function(vec![Type::Bool], vector(), Type::Unit)));
185        assert!(contains_vector(&function(vec![Type::Bool], Type::Bool, vector())));
186    }
