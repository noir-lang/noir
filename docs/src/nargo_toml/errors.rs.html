8/// Errors covering situations where a package is either missing, malformed or does not pass semver
9/// validation checks.
10#[derive(Debug, Error)]
11pub enum ManifestError {
12    /// Package doesn't have a manifest file
13    #[error("cannot find a Nargo.toml for {0}")]
14    MissingFile(PathBuf),
15
16    #[error("Cannot read file {0} - does it exist?")]
17    ReadFailed(PathBuf),
18
19    #[error("Cannot write to file {0}")]
20    WriteFailed(PathBuf),
21
22    #[error(
23        "A dependency named `{0}` already exists. Pass `--override` to replace it, or remove it from Nargo.toml first"
24    )]
25    DependencyAlreadyExists(String),
26
27    #[error("The `dependencies` section in {0} is not a table")]
28    InvalidDependenciesTable(PathBuf),
29
30    #[error("Nargo.toml is missing a parent directory")]
31    MissingParent,
32
33    #[error("Missing `type` field in {0}")]
34    MissingPackageType(PathBuf),
35
36    #[error("Cannot use `{1}` for `type` field in {0}")]
37    InvalidPackageType(PathBuf, String),
38
39    /// Package manifest is unreadable.
40    #[error("Nargo.toml is badly formed, could not parse.\n\n{}", .0.message())]
41    MalformedFile(#[from] toml::de::Error),
42
43    /// Package manifest could not be parsed while editing it.
44    #[error("Nargo.toml is badly formed, could not parse.\n\n{0}")]
45    MalformedFileEdit(#[from] toml_edit::TomlError),
46
47    #[error(
48        "Unexpected workspace definition found in {0}. If you're attempting to load this as a dependency, you may need to add a `directory` field to your `Nargo.toml` to show which package within the workspace to use"
49    )]
50    UnexpectedWorkspace(PathBuf),
51
52    #[error("Cannot find file {entry} which was specified as the `entry` field in {toml}")]
53    MissingEntryFile { toml: PathBuf, entry: PathBuf },
54
55    #[error(
56        r#"Cannot find file {entry} which is defaulted due to specifying `type = "{package_type}"` in {toml}"#
57    )]
58    MissingDefaultEntryFile { toml: PathBuf, entry: PathBuf, package_type: PackageType },
59
60    #[error("{} found in {toml}", if name.is_empty() { "Empty package name".into() } else { format!("Invalid package name `{name}`") })]
61    InvalidPackageName { toml: PathBuf, name: String },
62
63    #[error("{} found in {toml}", if name.is_empty() { "Empty dependency name".into() } else { format!("Invalid dependency name `{name}`") })]
64    InvalidDependencyName { toml: PathBuf, name: String },
65
66    #[error("Invalid directory path {directory} in {toml}: It must point to a subdirectory")]
67    InvalidDirectory { toml: PathBuf, directory: PathBuf },
68
69    #[error(transparent)]
70    GitError(#[from] GitError),
71
72    #[error("Selected package `{0}` was not found")]
73    MissingSelectedPackage(CrateName),
74
75    #[error("Default package was not found. Does {0} exist in your workspace?")]
76    MissingDefaultPackage(PathBuf),
77
78    #[error("Package `{0}` has type `bin` but you cannot depend on binary packages")]
79    BinaryDependency(CrateName),
80
81    #[error("No common ancestor between {root} and {current}")]
82    NoCommonAncestor { root: PathBuf, current: PathBuf },
83
84    #[error(transparent)]
85    SemverError(SemverError),
86
87    #[error("Cyclic package dependency found when processing {cycle}")]
88    CyclicDependency { cycle: String },
89}

91/// Errors encountered while downloading a git dependency into the global dependency cache.
92#[derive(Debug, Error)]
93pub enum GitError {
94    #[error("Invalid git dependency URL `{url}`: {source}")]
95    InvalidUrl { url: String, source: url::ParseError },
96
97    #[error("Invalid git dependency URL `{url}`: it must include a host")]
98    MissingHost { url: String },
99
100    #[error(
101        "Cannot download dependency {url} at tag `{tag}`: `git` was not found. Install git and make sure it is on your PATH"
102    )]
103    GitNotFound { url: String, tag: String },
104
105    #[error("Cannot download dependency {url} at tag `{tag}`: failed to run `git`: {source}")]
106    SpawnFailed { url: String, tag: String, source: std::io::Error },
107
108    #[error("Failed to download dependency {url} at tag `{tag}`: `git clone` failed ({status}){}", if stderr.is_empty() { String::new() } else { format!(":\n{stderr}") })]
109    CloneFailed { url: String, tag: String, status: ExitStatus, stderr: String },
110
111    #[error("Failed to lock the git dependency cache: {0}")]
112    Lock(std::io::Error),
113}

115#[allow(clippy::enum_variant_names)]
116#[derive(Error, Debug, PartialEq, Eq, Clone)]
117pub enum SemverError {
118    #[error(
119        "Invalid value for `compiler_version` in package {package_name}. Requirements may only refer to full releases"
120    )]
121    InvalidCompilerVersionRequirement { package_name: CrateName, required_compiler_version: String },
122    #[error(
123        "Incompatible compiler version in package {package_name}. Required compiler version is {required_compiler_version} but the compiler version is {compiler_version_found}.\n Update the compiler_version field in Nargo.toml to >={required_compiler_version} or compile this project with version {required_compiler_version}"
124    )]
125    IncompatibleVersion {
126        package_name: CrateName,
127        required_compiler_version: String,
128        compiler_version_found: String,
129    },
130    #[error(
131        "Could not parse the required compiler version for package {package_name} in Nargo.toml. Error: {error}"
132    )]
133    CouldNotParseRequiredVersion { package_name: String, error: String },
134    #[error(
135        "Could not parse the package version for package {package_name} in Nargo.toml. Error: {error}"
136    )]
137    CouldNotParsePackageVersion { package_name: String, error: String },
138}
