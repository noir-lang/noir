1//! Verify that line comments never cause the formatter to swallow code.
2//!
3//! The AST fuzzer generates a valid program, which is formatted once to establish a normalized
4//! baseline. We then insert a line comment at a fuzz-selected token boundary, format again, and
5//! require the non-comment token stream to remain equal to the baseline. This catches both invalid
6//! formatter output and the more dangerous case where swallowed code still parses.
7
8use arbitrary::Unstructured;
9use color_eyre::eyre;
10use nargo_fmt::{Config as FormatterConfig, format};
11use noir_ast_fuzzer::{DisplayAstAsNoir, arb_program};
12use noirc_errors::Location;
13use noirc_frontend::{
14    lexer::Lexer,
15    parser,
16    token::{FmtStrFragment, Token},
17};
18
19use super::default_config;
20
21pub fn fuzz(u: &mut Unstructured) -> eyre::Result<()> {
22    let config = default_config(u)?;
23    let program = arb_program(u, config)?;
24    let source = DisplayAstAsNoir(&program).to_string();
25    let Some(baseline) = try_format(&source) else { return Ok(()) };
26
27    let mut boundaries = significant_token_boundaries(&baseline);
28    boundaries.pop(); // A comment after the final token cannot swallow code.
29    if boundaries.is_empty() {
30        return Ok(());
31    }
32
33    let boundary = boundaries[u.choose_index(boundaries.len())?];
34    let mut with_comment = baseline.clone();
35    with_comment.insert_str(boundary, " // fuzzed line comment\n");
36
37    // A newline is not valid at every token boundary. Invalid mutations do not exercise the
38    // formatter and are discarded like other unsuitable fuzz inputs.
39    let Some(formatted) = try_format(&with_comment) else { return Ok(()) };
40
41    let expected_tokens = significant_tokens(&baseline);
42    let actual_tokens = significant_tokens(&formatted);
43    if actual_tokens != expected_tokens {
44        eyre::bail!(
45            "formatter changed code tokens after a line comment at byte {boundary}\n\
46             baseline:\n{baseline}\nwith comment:\n{with_comment}\nformatted:\n{formatted}"
47        );
48    }
49
50    Ok(())
51}
52
53fn try_format(source: &str) -> Option<String> {
54    let (parsed_module, errors) = parser::parse_program_with_dummy_file(source);
55    if errors.iter().any(|error| !error.is_warning()) {
56        return None;
57    }
58    Some(format(source, parsed_module, &FormatterConfig::default()))
59}
60
61fn significant_token_boundaries(source: &str) -> Vec<usize> {
62    Lexer::new_with_dummy_file(source)
63        .skip_comments(true)
64        .skip_whitespaces(true)
65        .filter_map(|token| {
66            let token = token.expect("formatter output should lex");
67            (!matches!(token.token(), Token::EOF)).then(|| token.span().end() as usize)
68        })
69        .collect()
70}
71
72fn significant_tokens(source: &str) -> Vec<Token> {
73    let tokens: Vec<_> = Lexer::new_with_dummy_file(source)
74        .skip_comments(true)
75        .skip_whitespaces(true)
76        .map(|token| without_locations(token.expect("formatter output should lex").into_token()))
77        .collect();
78
79    // A comment can make a group multiline, in which case the formatter may add a trailing comma.
80    // Ignore those layout-only tokens while checking that all source code survived.
81    (0..tokens.len())
82        .filter_map(|index| {
83            let token = &tokens[index];
84            let is_layout_comma = *token == Token::Comma
85                && matches!(
86                    tokens.get(index + 1),
87                    Some(Token::RightParen | Token::RightBracket | Token::RightBrace)
88                );
89            (!is_layout_comma).then(|| token.clone())
90        })
91        .collect()
92}
93
94/// Erase the source positions a token carries, so that tokens compare on code alone.
95///
96/// A format string is the one token that records where in the source it was written: the lexer
97/// stores the location of every `{...}` interpolation inside it. Inserting a comment moves the
98/// code that follows, which changes those locations without changing the program.
99fn without_locations(token: Token) -> Token {
100    let Token::FmtStr(fragments, length) = token else {
101        return token;
102    };
103    let fragments = fragments
104        .into_iter()
105        .map(|fragment| match fragment {
106            FmtStrFragment::Interpolation(name, _) => {
107                FmtStrFragment::Interpolation(name, Location::dummy())
108            }
109            fragment @ FmtStrFragment::String(_) => fragment,
110        })
111        .collect();
112    Token::FmtStr(fragments, length)
113}

115#[cfg(test)]
116mod tests {
117    use super::{significant_tokens, try_format};
118
119    /// A line comment shifts every token after it, and a format string carries the source
120    /// location of each of its interpolations. Those locations are not code, so moving one
121    /// must not read as the formatter having changed the program.
122    #[test]
123    fn line_comment_before_a_format_string_is_not_a_change() {
124        let source = "unconstrained fn main() {\n    let x: u32 = 0;\n    println(f\"v{x}\");\n}\n";
125        let baseline = try_format(source).expect("source should format");
126
127        let boundary = baseline.find(';').expect("the let statement should be formatted") + 1;
128        let mut with_comment = baseline.clone();
129        with_comment.insert_str(boundary, " // fuzzed line comment\n");
130        let formatted = try_format(&with_comment).expect("commented source should format");
131
132        assert_eq!(significant_tokens(&formatted), significant_tokens(&baseline));
133    }
134
135    #[test]
136    fn fuzz_with_arbtest() {
137        crate::targets::tests::fuzz_with_arbtest(super::fuzz, 1000);
138    }
