58#[wasm_bindgen(js_name = abiEncode)]
59pub fn abi_encode(
60    abi: JsAbi,
61    inputs: JsInputMap,
62    return_value: Option<JsInputValue>,
63) -> Result<JsWitnessMap, JsAbiError> {
64    console_error_panic_hook::set_once();
65    let abi: Abi =
66        JsValueSerdeExt::into_serde(&JsValue::from(abi)).map_err(|err| err.to_string())?;
67    let inputs: BTreeMap<String, JsonTypes> =
68        JsValueSerdeExt::into_serde(&JsValue::from(inputs)).map_err(|err| err.to_string())?;
69    for value in inputs.values() {
70        reject_unsafe_integers(value)?;
71    }
72    let return_value: Option<InputValue> = return_value
73        .map(|return_value| {
74            let toml_return_value: JsonTypes =
75                JsValueSerdeExt::into_serde(&JsValue::from(return_value))
76                    .expect("could not decode return value");
77            reject_unsafe_integers(&toml_return_value)?;
78            InputValue::try_from_json(
79                toml_return_value,
80                &abi.return_type.as_ref().unwrap().abi_type,
81                MAIN_RETURN_NAME,
82            )
83            .map_err(JsAbiError::from)
84        })
85        .transpose()?;
86
87    let abi_map = abi.to_btree_map();
88    let parsed_inputs: BTreeMap<String, InputValue> =
89        try_btree_map(abi_map, |(arg_name, abi_type)| {
90            // Check that toml contains a value for each argument in the ABI.
91            let value = inputs
92                .get(&arg_name)
93                .ok_or_else(|| InputParserError::MissingArgument(arg_name.clone()))?;
94            InputValue::try_from_json(value.clone(), &abi_type, &arg_name)
95                .map(|input_value| (arg_name, input_value))
96        })?;
97
98    let witness_map = abi.encode(&parsed_inputs, return_value)?;
99
100    Ok(witness_map.into())
101}

103/// The largest integer JavaScript can represent exactly, i.e. `Number.MAX_SAFE_INTEGER` (`2^53 - 1`).
104const MAX_SAFE_INTEGER: i64 = (1 << 53) - 1;
105
106/// Rejects any numeric input whose magnitude exceeds [`MAX_SAFE_INTEGER`].
107///
108/// JavaScript represents `number` as a 64-bit float, so an integer above `2^53 - 1` is silently
109/// rounded before it ever reaches wasm. Accepting such a value would encode a witness for a
110/// different field element than the one written at the call site. Callers must pass large values as
111/// strings, which preserve their exact decimal representation.
112fn reject_unsafe_integers(value: &JsonTypes) -> Result<(), JsAbiError> {
113    match value {
114        JsonTypes::Integer(integer) => {
115            if integer.unsigned_abs() > MAX_SAFE_INTEGER as u64 {
116                return Err(JsAbiError::new(format!(
117                    "Numeric input `{integer}` is not a safe integer (its magnitude exceeds \
118                     2^53 - 1, so JavaScript may have already rounded it); pass it as a string \
119                     instead"
120                )));
121            }
122        }
123        JsonTypes::Array(items) => {
124            for item in items {
125                reject_unsafe_integers(item)?;
126            }
127        }
128        JsonTypes::Table(map) => {
129            for item in map.values() {
130                reject_unsafe_integers(item)?;
131            }
132        }
133        JsonTypes::String(_) | JsonTypes::Bool(_) => {}
134    }
135
136    Ok(())
137}

139#[wasm_bindgen(js_name = abiDecode)]
140pub fn abi_decode(abi: JsAbi, witness_map: JsWitnessMap) -> Result<JsValue, JsAbiError> {
141    console_error_panic_hook::set_once();
142    let abi: Abi =
143        JsValueSerdeExt::into_serde(&JsValue::from(abi)).map_err(|err| err.to_string())?;
144
145    let witness_map = WitnessMap::from(witness_map);
146
147    let (inputs, return_value) = abi.decode(&witness_map)?;
148
149    let abi_types = abi.to_btree_map();
150    let inputs_map: BTreeMap<String, JsonTypes> = try_btree_map(inputs, |(key, value)| {
151        JsonTypes::try_from_input_value(&value, &abi_types[&key]).map(|value| (key, value))
152    })?;
153
154    let return_value = return_value
155        .map(|value| JsonTypes::try_from_input_value(&value, &abi.return_type.unwrap().abi_type))
156        .transpose()?;
157
158    #[derive(Serialize)]
159    struct InputsAndReturn {
160        inputs: BTreeMap<String, JsonTypes>,
161        return_value: Option<JsonTypes>,
162    }
163
164    let return_struct = InputsAndReturn { inputs: inputs_map, return_value };
165    <JsValue as JsValueSerdeExt>::from_serde(&return_struct).map_err(|err| err.to_string().into())
166}

168#[wasm_bindgen(js_name = serializeWitness)]
169pub fn serialize_witness(witness_map: JsWitnessMap) -> Result<Vec<u8>, JsAbiError> {
170    console_error_panic_hook::set_once();
171    let converted_witness: WitnessMap<FieldElement> = witness_map.into();
172    let witness_stack: WitnessStack<FieldElement> = converted_witness.into();
173    let output = witness_stack.serialize();
174    output.map_err(|_| JsAbiError::new("Failed to serialize witness stack".to_string()))
175}

177#[wasm_bindgen(js_name = deserializeWitness)]
178pub fn deserialize_witness(bytes: Vec<u8>) -> Result<JsWitnessMap, JsAbiError> {
179    console_error_panic_hook::set_once();
180    let mut witness_stack =
181        WitnessStack::deserialize(bytes.as_slice()).map_err(|err| err.to_string())?;
182    let witness =
183        witness_stack.pop().expect("Should have at least one witness on the stack").witness;
184    Ok(witness.into())
185}

195#[wasm_bindgen(js_name = abiDecodeError)]
196pub fn abi_decode_error(
197    abi: JsAbi,
198    raw_error: JsRawAssertionPayload,
199) -> Result<JsValue, JsAbiError> {
200    console_error_panic_hook::set_once();
201    let mut abi: Abi =
202        JsValueSerdeExt::into_serde(&JsValue::from(abi)).map_err(|err| err.to_string())?;
203
204    let raw_error: RawAssertionPayload<String> =
205        JsValueSerdeExt::into_serde(&JsValue::from(raw_error)).map_err(|err| err.to_string())?;
206    // `FieldElement` is represented as a string in JS so we must convert these into `FieldElements` manually.
207    let error_data = raw_error
208        .data
209        .iter()
210        .map(|field| FieldElement::from_hex(field))
211        .collect::<Option<Vec<_>>>()
212        .unwrap();
213
214    let error_type = abi.error_types.remove(&raw_error.selector).expect("Missing error type");
215    match error_type {
216        AbiErrorType::FmtString { .. } => {
217            let string = display_abi_error(&error_data, error_type).to_string();
218            Ok(JsValue::from_str(&string))
219        }
220        AbiErrorType::Custom(typ) => {
221            let input_value = decode_value(&mut error_data.into_iter(), &typ, "error")?;
222            let json_types = JsonTypes::try_from_input_value(&input_value, &typ)?;
223            <JsValue as JsValueSerdeExt>::from_serde(&json_types)
224                .map_err(|err| err.to_string().into())
225        }
226        AbiErrorType::String { string } => Ok(JsValue::from_str(&string)),
227    }
